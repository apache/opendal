30pub struct ObjectStoreDeleter {
31    store: Arc<dyn ObjectStore + 'static>,
32}
33
34impl ObjectStoreDeleter {
35    pub(crate) fn new(store: Arc<dyn ObjectStore + 'static>) -> Self {
36        Self { store }
37    }
38}
39
40/// Object store keys cannot end with `/`, and `object_store::path::Path::from`
41/// drops a trailing slash. A directory path therefore has no marker object to
42/// delete, and converting it would target the same-named regular file.
43fn is_directory_path(path: &str) -> bool {
44    path.ends_with('/')
45}
46
47impl oio::BatchDelete for ObjectStoreDeleter {
48    async fn delete_once(&self, path: String, _: OpDelete) -> Result<()> {
49        if is_directory_path(&path) {
50            return Ok(());
51        }
52
53        let object_path = ObjectStorePath::from(path);
54        match self.store.delete(&object_path).await {
55            Ok(()) | Err(object_store::Error::NotFound { .. }) => Ok(()),
56            Err(err) => Err(parse_error(err)),
57        }
58    }
59
60    async fn delete_batch(&self, paths: Vec<(String, OpDelete)>) -> Result<BatchDeleteResult> {
61        let mut files = Vec::new();
62        let mut result_batch = BatchDeleteResult::default();
63        for (path, args) in paths {
64            if is_directory_path(&path) {
65                result_batch.succeeded.push((path, args));
66            } else {
67                files.push((path, args));
68            }
69        }
70
71        if files.is_empty() {
72            return Ok(result_batch);
73        }
74
75        let locations: Vec<_> = files
76            .iter()
77            .map(|(path, _)| Ok(ObjectStorePath::from(path.as_str())))
78            .collect();
79        let mut results = self.store.delete_stream(stream::iter(locations).boxed());
80        for (path, args) in files {
81            let Some(result) = results.next().await else {
82                break;
83            };
84            match result {
85                Ok(_) | Err(object_store::Error::NotFound { .. }) => {
86                    result_batch.succeeded.push((path, args));
87                }
88                Err(
89                    err @ (object_store::Error::AlreadyExists { .. }
90                    | object_store::Error::Precondition { .. }
91                    | object_store::Error::NotModified { .. }
92                    | object_store::Error::PermissionDenied { .. }
93                    | object_store::Error::Unauthenticated { .. }),
94                ) => {
95                    result_batch.failed.push((path, args, parse_error(err)));
96                }
97                // An unattributed bulk error can stand for an entire chunk.
98                Err(err) => return Err(parse_error(err)),
99            }
100        }
101
102        Ok(result_batch)
103    }
104}

106#[cfg(test)]
107mod tests {
108    use object_store::local::LocalFileSystem;
109    use object_store::memory::InMemory;
110
111    use super::*;
112    use crate::ObjectStoreBuilder;
113
114    #[tokio::test]
115    async fn test_delete_directory_path_keeps_same_named_file() {
116        let store: Arc<dyn ObjectStore> = Arc::new(InMemory::new());
117        store
118            .put(&ObjectStorePath::from("foo"), "file".into())
119            .await
120            .unwrap();
121        store
122            .put(&ObjectStorePath::from("foo/child"), "child".into())
123            .await
124            .unwrap();
125        let op = Operator::new(ObjectStoreBuilder::new(store)).unwrap();
126
127        op.delete("foo/").await.unwrap();
128        assert!(op.stat("foo").await.unwrap().is_file());
129        assert!(op.stat("foo/child").await.unwrap().is_file());
130
131        op.delete_iter(["foo/", "missing"]).await.unwrap();
132        assert!(op.stat("foo").await.unwrap().is_file());
133        assert!(op.stat("foo/child").await.unwrap().is_file());
134
135        op.delete_iter(["foo/", "bar/"]).await.unwrap();
136        assert!(op.stat("foo").await.unwrap().is_file());
137        assert!(op.stat("foo/child").await.unwrap().is_file());
138    }
139
140    #[tokio::test]
141    async fn test_delete_directory_and_root_paths_on_local_filesystem() {
142        let root = tempfile::tempdir().unwrap();
143        let store: Arc<dyn ObjectStore> =
144            Arc::new(LocalFileSystem::new_with_prefix(root.path()).unwrap());
145        store
146            .put(&ObjectStorePath::from("foo"), "file".into())
147            .await
148            .unwrap();
149        let op = Operator::new(ObjectStoreBuilder::new(store)).unwrap();
150
151        op.delete("/").await.unwrap();
152        op.delete("foo/").await.unwrap();
153        op.delete_iter(["/", "foo/"]).await.unwrap();
154        assert!(op.stat("foo").await.unwrap().is_file());
155    }
156
157    #[tokio::test]
158    async fn test_local_filesystem_missing_deletes_are_idempotent() {
159        let root = tempfile::tempdir().unwrap();
160        let store = LocalFileSystem::new_with_prefix(root.path()).unwrap();
161        store
162            .put(&ObjectStorePath::from("present.txt"), "content".into())
163            .await
164            .unwrap();
165        store
166            .put(&ObjectStorePath::from("keep"), "keep contents".into())
167            .await
168            .unwrap();
169        let op = Operator::new(ObjectStoreBuilder::new(Arc::new(store))).unwrap();
170
171        op.delete("missing.txt").await.unwrap();
172        op.delete_iter([
173            "/",
174            "keep/",
175            "present.txt",
176            "missing.txt",
177            "also-missing.txt",
178        ])
179        .await
180        .unwrap();
181        assert_eq!(op.read("keep").await.unwrap().to_bytes(), "keep contents");
182        assert_eq!(
183            op.stat("present.txt").await.unwrap_err().kind(),
184            ErrorKind::NotFound
185        );
186    }
