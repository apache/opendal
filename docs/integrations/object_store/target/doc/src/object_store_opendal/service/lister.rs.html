31pub struct ObjectStoreLister {
32    stream: BoxStream<'static, Result<oio::Entry>>,
33}
34
35impl ObjectStoreLister {
36    pub(crate) fn new(
37        store: Arc<dyn ObjectStore + 'static>,
38        path: &str,
39        args: OpList,
40    ) -> Result<Self> {
41        // `OpList::limit` is a per-page size hint. object_store does not expose
42        // a page-size parameter, so the stream must return every result and
43        // ignore the hint rather than truncating the listing.
44        let stream = if args.recursive() {
45            recursive_stream(store, path, args.start_after())
46        } else {
47            non_recursive_stream(store, path, args.start_after())
48        };
49
50        Ok(Self { stream })
51    }
52}
53
54/// Build a recursive listing.
55///
56/// [`ObjectStore::list`] is always recursive. A path without a trailing `/` is
57/// still a prefix, so list its parent and keep only entries that start with the
58/// requested prefix.
59///
60/// `object_store::Path` stores percent-encoded locations, while OpenDAL paths
61/// are decoded. Decode locations before comparing them with the requested
62/// prefix or `start_after`, and return the decoded path in the entry.
63fn recursive_stream(
64    store: Arc<dyn ObjectStore + 'static>,
65    path: &str,
66    start_after: Option<&str>,
67) -> BoxStream<'static, Result<oio::Entry>> {
68    let (directory, prefix) = if path == "/" || path.ends_with('/') {
69        (path, path.trim_start_matches('/').to_string())
70    } else {
71        (get_parent(path), path.to_string())
72    };
73    let start_after = start_after.map(str::to_string);
74
75    // Do not use `list_with_offset`: it compares percent-encoded locations,
76    // whose order can differ from the decoded OpenDAL order. Filter after
77    // decoding so `start_after` keeps OpenDAL semantics.
78    store
79        .list(Some(&ObjectStorePath::from(directory)))
80        .filter_map(move |item| {
81            let entry = match item {
82                Ok(meta) => {
83                    let location = percent_decode_path(meta.location.as_ref());
84                    if !location.starts_with(&prefix)
85                        || start_after
86                            .as_deref()
87                            .is_some_and(|marker| location.as_str() <= marker)
88                    {
89                        return futures::future::ready(None);
90                    }
91
92                    let metadata = format_metadata(&meta);
93                    Ok(oio::Entry::new(&location, metadata))
94                }
95                Err(err) => Err(parse_error(err)),
96            };
97
98            futures::future::ready(Some(entry))
99        })
100        .boxed()
101}

103/// Build a non-recursive listing from [`ObjectStore::list_with_delimiter`].
104///
105/// The delimiter API returns direct children and folds deeper objects into
106/// common prefixes. A path without a trailing `/` is still a prefix, so list
107/// its parent and keep only entries that start with the requested prefix.
108///
109/// Common prefixes and object locations are percent-encoded by
110/// `object_store::Path`, so decode them before returning entries and comparing
111/// them with the OpenDAL prefix or `start_after`.
112fn non_recursive_stream(
113    store: Arc<dyn ObjectStore + 'static>,
114    path: &str,
115    start_after: Option<&str>,
116) -> BoxStream<'static, Result<oio::Entry>> {
117    let (directory, prefix) = if path == "/" || path.ends_with('/') {
118        (path, None)
119    } else {
120        (get_parent(path), Some(path.to_string()))
121    };
122    let directory = ObjectStorePath::from(directory);
123    let start_after = start_after.map(str::to_string);
124
125    stream::once(async move {
126        let result = match store.list_with_delimiter(Some(&directory)).await {
127            Ok(result) => result,
128            Err(err) => return vec![Err(parse_error(err))],
129        };
130
131        let mut entries: Vec<Result<oio::Entry>> =
132            Vec::with_capacity(result.common_prefixes.len() + result.objects.len());
133        for common_prefix in result.common_prefixes {
134            let entry_path = format!("{}/", percent_decode_path(common_prefix.as_ref()));
135            if !keep_entry(&entry_path, prefix.as_deref(), start_after.as_deref()) {
136                continue;
137            }
138            entries.push(Ok(oio::Entry::new(
139                &entry_path,
140                MetadataBuilder::dir().build(),
141            )));
142        }
143        for meta in result.objects {
144            let entry_path = percent_decode_path(meta.location.as_ref());
145            if !keep_entry(&entry_path, prefix.as_deref(), start_after.as_deref()) {
146                continue;
147            }
148            entries.push(Ok(oio::Entry::new(&entry_path, format_metadata(&meta))));
149        }
150        entries
151    })
152    .flat_map(stream::iter)
153    .boxed()
154}
155
156fn keep_entry(path: &str, prefix: Option<&str>, start_after: Option<&str>) -> bool {
157    prefix.is_none_or(|prefix| path.starts_with(prefix))
158        && start_after.is_none_or(|marker| path > marker)
159}

161// ObjectStoreLister is safe to share between threads, because it only has &mut self API calls.
162unsafe impl Sync for ObjectStoreLister {}
163
164impl oio::List for ObjectStoreLister {
165    async fn next(&mut self) -> Result<Option<oio::Entry>> {
166        self.stream.next().await.transpose()
167    }
168}
