fn last_modified(ruby: &Ruby, rb_self: &Self) -> Result<Option<magnus::time::Time>, Error> {
    let Some(timestamp) = rb_self.0.last_modified() else {
        return Ok(None);
    };
    let timestamp = timestamp.into_inner();
    let mut timespec = magnus::time::Timespec {
        tv_sec: timestamp.as_second(),
        tv_nsec: i64::from(timestamp.subsec_nanosecond()),
    };
    // Jiff uses signed fractional seconds, while a timespec uses nonnegative nanoseconds.
    if timespec.tv_nsec < 0 {
        timespec.tv_sec -= 1;
        timespec.tv_nsec += 1_000_000_000;
    }

    ruby.time_timespec_new(timespec, magnus::time::Offset::utc())
        .map(Some)
}

pub fn mode(&self) -> &str {
    match self.0.mode() {
        ocore::EntryMode::FILE => "File",
        ocore::EntryMode::DIR => "Directory",
        ocore::EntryMode::Unknown => "Unknown",
    }
}

fn user_metadata(ruby: &Ruby, rb_self: &Self) -> Option<magnus::RHash> {
        rb_self
            .0
            .user_metadata()
            .map(|metadata| ruby.hash_from_iter(metadata))
    }
}
