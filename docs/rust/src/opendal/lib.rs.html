43/// Install the process-wide defaults provided by the facade crate.
44///
45/// The first call runs [`init_default_registry`], installs the Tokio executor
46/// if `executors-tokio` is enabled, then installs the reqwest transport and
47/// registers the `http` context factory if a `http-transport-reqwest*` feature
48/// is enabled. Later calls leave the installed defaults and factories in place.
49///
50/// The first installed executor and the first installed HTTP transport each
51/// stay in effect as process-wide defaults. Applications that supply their own
52/// can disable `auto-register-services` and install them first with
53/// [`Executor::install_default`] and [`HttpTransporter::install_default`].
54/// They can then call this function to install the remaining facade defaults,
55/// or call [`init_default_registry`] for services and layers alone.
56///
57/// Registration of the reqwest `http` context factory replaces an earlier
58/// factory of the same name, even if a custom process-wide transport is
59/// already installed. To keep a custom `http` context factory, use
60/// [`OperatorRegistry::register_context`] after the first call. To replace
61/// runtime resources on one operator, use [`Operator::with_context`].
62///
63/// When `auto-register-services` is enabled, a process constructor calls this
64/// function before `main`. If the constructor is disabled or unavailable,
65/// call it explicitly to make these defaults available before constructing
66/// operators with [`Operator::from_uri`] or [`Operator::via_iter`].
67pub fn install_default() {
68    static DEFAULTS_INIT: std::sync::Once = std::sync::Once::new();
69    DEFAULTS_INIT.call_once(|| {
70        init_default_registry();
71
72        #[cfg(feature = "executors-tokio")]
73        opendal_executor_tokio::install_default();
74
75        #[cfg(any(
76            feature = "http-transport-reqwest",
77            feature = "http-transport-reqwest-native-tls",
78            feature = "http-transport-reqwest-rustls",
79            feature = "http-transport-reqwest-rustls-no-provider"
80        ))]
81        {
82            opendal_http_transport_reqwest::install_default();
83            opendal_http_transport_reqwest::register_http_context(OperatorRegistry::get());
84        }
85    });
86}

94mod default_executor_tests {
95    #[tokio::test]
96    async fn default_executor_runs_concurrent_reads() -> crate::Result<()> {
97        let op = crate::Operator::via_iter("memory", [])?;
98        op.write("file", "abcdefgh").await?;
99        let content = op
100            .read_with("file")
101            .range(0..8)
102            .chunk(2)
103            .concurrent(2)
104            .await?;
105        assert_eq!(content.to_bytes(), "abcdefgh");
106        Ok(())
107    }

110/// Register the facade's enabled services and configuration-driven layers in
111/// the global [`OperatorRegistry`].
112///
113/// This function registers `memory` and the services enabled by `services-*`
114/// features on the current target. The built-in configuration layer names are
115/// `chaos`, `concurrent_limit`, `logging`, `mime_guess`, `retry`, `throttle`,
116/// `timeout`, and `tracing`, each gated by its corresponding `layers-*`
117/// feature. Registration makes these layers available for selection in
118/// `layers` when loading an operator.
119///
120/// The first call replaces earlier registrations for those service and layer
121/// names. Register your own replacements after initialization to keep them
122/// across subsequent calls.
123///
124/// Use [`install_default`] to also install the process-wide runtime defaults
125/// and the reqwest `http` context factory. The `auto-register-services`
126/// constructor normally calls it before `main`. Some linkers omit constructors
127/// from `staticlib` builds, so call it explicitly if you need those defaults
128/// in such a build.
129pub fn init_default_registry() {
130    static DEFAULT_REGISTRY_INIT: std::sync::Once = std::sync::Once::new();
131    DEFAULT_REGISTRY_INIT.call_once(|| init_default_registry_inner(OperatorRegistry::get()));
132}

876        use crate::services::{GARAGE_SCHEME, Garage, GarageConfig};
877
878        let options = [("endpoint", "http://127.0.0.1:3900"), ("region", "garage")];

880        let from_iter = Operator::via_iter(
881            GARAGE_SCHEME,
882            options
883                .into_iter()
884                .chain([("bucket", "example-bucket"), ("root", "root")])
885                .map(|(key, value)| (key.to_owned(), value.to_owned())),
886        )
887        .unwrap();
888        let mut config = GarageConfig::default();
889        config.bucket = "example-bucket".into();
890        config.root = Some("root".into());
891        config.endpoint = "http://127.0.0.1:3900".into();
892        config.region = "garage".into();
893        let from_config = Operator::from_config(config).unwrap();
894        let from_builder = Operator::new(
895            Garage::default()
896                .bucket("example-bucket")
897                .root("root")
898                .endpoint("http://127.0.0.1:3900")
899                .region("garage"),
900        )
901        .unwrap();
902
903        for op in [from_uri, from_iter, from_config, from_builder] {
904            assert_eq!(op.info().scheme(), GARAGE_SCHEME);
