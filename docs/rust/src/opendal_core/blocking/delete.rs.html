18use crate::Deleter as AsyncDeleter;
19use crate::*;
20
21/// BlockingDeleter is designed to continuously remove content from storage.
22///
23/// It leverages batch deletion capabilities provided by storage services for efficient removal.
24pub struct Deleter {
25    handle: tokio::runtime::Handle,
26    inner: AsyncDeleter,
27}
28
29impl Deleter {
30    pub(crate) fn create(handle: tokio::runtime::Handle, inner: AsyncDeleter) -> Result<Self> {
31        Ok(Self { handle, inner })
32    }
33
34    /// Delete a path with default or per-entry options.
35    ///
36    /// Pass a path directly to use [`options::DeleteOptions::default`]. Pass
37    /// `(String, DeleteOptions)` when this entry needs version or condition
38    /// options.
39    pub fn delete(&mut self, input: impl IntoDeleteInput) -> Result<()> {
40        self.handle.block_on(self.inner.delete(input))
41    }
42
43    /// Delete an infallible iterator of paths.
44    ///
45    /// Also see:
46    ///
47    /// - [`Deleter::delete_try_iter`]: delete a fallible iterator of paths.
48    pub fn delete_iter<I, D>(&mut self, iter: I) -> Result<()>
49    where
50        I: IntoIterator<Item = D>,
51        D: IntoDeleteInput,
52    {
53        for entry in iter {
54            self.delete(entry)?;
55        }
56        Ok(())
57    }
58
59    /// Delete an fallible iterator of paths.
60    ///
61    /// Also see:
62    ///
63    /// - [`Deleter::delete_iter`]: delete an infallible iterator of paths.
64    pub fn delete_try_iter<I, D>(&mut self, try_iter: I) -> Result<()>
65    where
66        I: IntoIterator<Item = Result<D>>,
67        D: IntoDeleteInput,
68    {
69        for entry in try_iter {
70            self.delete(entry?)?;
71        }
72        Ok(())
73    }
74
75    /// Close the deleter, this will flush the deleter and wait until all paths are deleted.
76    pub fn close(&mut self) -> Result<()> {
77        self.handle.block_on(self.inner.close())
78    }
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84    use crate::Entry;
85    use crate::blocking::Operator;
86    use crate::services;
87
88    fn new_runtime() -> tokio::runtime::Runtime {
89        tokio::runtime::Builder::new_multi_thread()
90            .enable_all()
91            .build()
92            .unwrap()
93    }
94
95    fn new_blocking_operator(rt: &tokio::runtime::Runtime) -> Operator {
96        // The guard must not outlive construction, or calls on the operator
97        // would nest a `block_on` inside the entered runtime.
98        let _guard = rt.enter();
99        let op = crate::Operator::new(services::Memory::default()).unwrap();
100        Operator::new(op).unwrap()
101    }
102
103    #[test]
104    fn test_delete_try_iter_accepts_blocking_lister() {
105        let rt = new_runtime();
106        let op = new_blocking_operator(&rt);
107        op.write("dir/a", "a").unwrap();
108        op.write("dir/b", "b").unwrap();
109
110        let mut deleter = op.deleter().unwrap();
111        deleter.delete_try_iter(op.lister("dir/").unwrap()).unwrap();
112        deleter.close().unwrap();
113
114        assert!(op.list("dir/").unwrap().is_empty());
115    }
116
117    #[test]
118    fn test_delete_try_iter_error_drops_blocking_lister() {
119        let rt = new_runtime();
120        let op = new_blocking_operator(&rt);
121        op.write("dir/a", "a").unwrap();
122        op.write("dir/b", "b").unwrap();
123
124        let err = Error::new(ErrorKind::Unexpected, "injected iterator failure");
125        let iter = std::iter::once(Err::<Entry, _>(err)).chain(op.lister("dir/").unwrap());
126
127        let mut deleter = op.deleter().unwrap();
128        let err = deleter.delete_try_iter(iter).unwrap_err();
129        assert_eq!(err.kind(), ErrorKind::Unexpected);
130        deleter.close().unwrap();
131
132        assert_eq!(op.list("dir/").unwrap().len(), 2);
133    }
134
135    #[test]
136    fn test_deleter_options() {
137        let rt = new_runtime();
138        let op = new_blocking_operator(&rt);
139        op.write("dir/a", "a").unwrap();
140        op.write("dir/b", "b").unwrap();
141
142        let mut deleter = op
143            .deleter_options(options::DeleterOptions { concurrent: 0 })
144            .unwrap();
145        deleter.delete_iter(["dir/a", "dir/b"]).unwrap();
146        deleter.close().unwrap();
147
148        assert!(op.list("dir/").unwrap().is_empty());
149    }
150}
