20/// Compiled Unix shell style glob pattern for path-component matching.
21///
22/// Services implementing native [`crate::Capability::list_with_glob`] and
23/// [`crate::layers::SimulateLayer`] share this matcher to evaluate
24/// [`crate::options::ListOptions::glob`] consistently and prune directories
25/// during traversal.
26#[derive(Debug)]
27pub struct GlobPattern {
28    segments: Vec<Segment>,
29    directories_only: bool,
30}
31
32impl GlobPattern {
33    /// Parse and compile a glob pattern.
34    ///
35    /// Returns [`ErrorKind::ConfigInvalid`] if the pattern is empty, contains
36    /// malformed syntax, or uses disallowed literal components such as `.` or
37    /// `..`.
38    pub fn new(pattern: &str) -> Result<Self> {
39        let mut segments = Vec::new();
40        for segment in pattern.split('/').filter(|part| !part.is_empty()) {
41            if segment == "**" {
42                if !matches!(segments.last(), Some(Segment::Recursive)) {
43                    segments.push(Segment::Recursive);
44                }
45                continue;
46            }
47            let matcher = SegmentMatcher::new(segment)?;
48            if let Some(literal) = matcher.literal() {
49                if matches!(literal.as_str(), "." | "..") || literal.contains('\0') {
50                    return Err(invalid("invalid literal path component in glob pattern"));
51                }
52                segments.push(Segment::Literal(literal));
53            } else {
54                segments.push(Segment::Glob(matcher));
55            }
56        }
57        if segments.is_empty() {
58            return Err(invalid("glob pattern must contain a path component"));
59        }
60        Ok(Self {
61            segments,
62            directories_only: pattern.ends_with('/'),
63        })
64    }
65
66    /// Yield leading literal directory components that can be entered directly
67    /// without listing their parent directories.
68    pub fn literal_directories(&self) -> impl Iterator<Item = &str> {
69        let terminal_components = if matches!(self.segments.last(), Some(Segment::Recursive)) {
70            // The directory before a trailing `**` can itself match.
71            2
72        } else {
73            1
74        };
75        self.segments[..self.segments.len().saturating_sub(terminal_components)]
76            .iter()
77            .map_while(|segment| match segment {
78                // Backslashes and drive prefixes can have path semantics on some services.
79                Segment::Literal(value) if !value.contains(['\\', ':']) => Some(value.as_str()),
80                _ => None,
81            })
82    }
83
84    /// Return `true` if the pattern can match entries deeper than one directory level.
85    pub fn spans_directories(&self) -> bool {
86        self.segments.len() > 1 || matches!(self.segments[0], Segment::Recursive)
87    }
88
89    /// Whether a guided walk from a directory at `position` lists every directory below it.
90    pub(crate) fn walks_whole_subtree(&self, position: usize) -> bool {
91        self.segments[position..]
92            .iter()
93            .find(
94                |segment| !matches!(segment, Segment::Glob(matcher) if matcher.matches_any_name()),
95            )
96            .is_some_and(|segment| matches!(segment, Segment::Recursive))
97    }
98
99    /// Advance every viable pattern position against a single path component.
100    ///
101    /// The returned positions are sorted and deduplicated. Start from `&[0]` at
102    /// the base directory, or `&[n]` after skipping `n` leading literal
103    /// directories returned by [`Self::literal_directories`].
104    pub fn advance(&self, positions: &[usize], name: &str) -> Vec<usize> {
105        let mut next = Vec::new();
106        for &position in positions {
107            match self.segments.get(position) {
108                Some(Segment::Recursive) => {
109                    next.push(position);
110                    match self.segments.get(position + 1) {
111                        Some(segment) if segment.matches(name) => next.push(position + 2),
112                        None => next.push(position + 1),
113                        _ => {}
114                    }
115                }
116                Some(segment) if segment.matches(name) => next.push(position + 1),
117                _ => {}
118            }
119        }
120        next.sort_unstable();
121        next.dedup();
122        next
123    }
124
125    /// Return `true` if `positions` represent a complete match for an entry.
126    ///
127    /// Patterns with a trailing `/` require `is_dir` to be `true`.
128    pub fn accepts(&self, positions: &[usize], is_dir: bool) -> bool {
129        if self.directories_only && !is_dir {
130            return false;
131        }
132        let end = self.segments.len();
133        positions.last() == Some(&end)
134            || (is_dir
135                && matches!(self.segments.last(), Some(Segment::Recursive))
136                && positions.last() == Some(&(end - 1)))
137    }
138
139    /// Return `true` if descending into a directory at `positions` can still
140    /// match descendants.
141    pub fn can_descend(&self, positions: &[usize]) -> bool {
142        positions
143            .first()
144            .is_some_and(|&first| first < self.segments.len())
145    }
146
147    /// Return `true` if a relative path matches the pattern.
148    pub fn matches(&self, path: &str, is_dir: bool) -> bool {
149        let mut positions = vec![0];
150        for component in path.trim_end_matches('/').split('/') {
151            positions = self.advance(&positions, component);
152            if positions.is_empty() {
153                return false;
154            }
155        }
156        self.accepts(&positions, is_dir)
157    }
158}

160#[derive(Debug)]
161enum Segment {
162    Literal(String),
163    Glob(SegmentMatcher),
164    Recursive,
165}
166
167impl Segment {
168    fn matches(&self, name: &str) -> bool {
169        match self {
170            Self::Literal(literal) => literal == name,
171            Self::Glob(matcher) => matcher.matches(name),
172            Self::Recursive => unreachable!("consecutive recursive wildcards are collapsed"),
173        }
174    }
175}
176
177#[derive(Debug)]
178struct SegmentMatcher {
179    tokens: Vec<Token>,
180}
181
182#[derive(Debug)]
183enum Token {
184    Literal(char),
185    Any,
186    Star,
187    Class {
188        ranges: Vec<(char, char)>,
189        negated: bool,
190    },
191    Split(usize),
192    Jump(usize),
193}
194
195#[derive(Default)]
196struct Alternation {
197    split: usize,
198    exits: Vec<usize>,
199}
200
201impl SegmentMatcher {
202    fn new(segment: &str) -> Result<Self> {
203        let mut chars = segment.chars().peekable();
204        let mut tokens = Vec::new();
205        let mut alternations = Vec::<Alternation>::new();
206        while let Some(ch) = chars.next() {
207            let token = match ch {
208                '*' => {
209                    if chars.peek() == Some(&'*') {
210                        return Err(invalid("`**` must occupy a whole glob path component"));
211                    }
212                    Token::Star
213                }
214                '?' => Token::Any,
215                '[' => parse_class(&mut chars)?,
216                '\\' => Token::Literal(escaped_char(&mut chars)?),
217                '{' => {
218                    alternations.push(Alternation {
219                        split: tokens.len(),
220                        ..Default::default()
221                    });
222                    Token::Split(0)
223                }
224                ',' if !alternations.is_empty() => {
225                    let group = alternations.last_mut().unwrap();
226                    group.exits.push(tokens.len());
227                    tokens.push(Token::Jump(0));
228                    let next_branch = tokens.len();
229                    tokens[group.split] = Token::Split(next_branch);
230                    group.split = next_branch;
231                    Token::Split(0)
232                }
233                '}' if !alternations.is_empty() => {
234                    let group = alternations.pop().unwrap();
235                    tokens[group.split] = Token::Jump(group.split + 1);
236                    let end = tokens.len();
237                    for exit in group.exits {
238                        tokens[exit] = Token::Jump(end);
239                    }
240                    continue;
241                }
242                _ => Token::Literal(ch),
243            };
244            tokens.push(token);
245        }
246        if !alternations.is_empty() {
247            return Err(invalid("unterminated alternation in glob pattern"));
248        }
249        Ok(Self { tokens })
250    }
251
252    fn matches_any_name(&self) -> bool {
253        matches!(self.tokens.as_slice(), [Token::Star])
254    }
255
256    fn literal(&self) -> Option<String> {
257        self.tokens
258            .iter()
259            .map(|token| match token {
260                Token::Literal(ch) => Some(*ch),
261                _ => None,
262            })
263            .collect()
264    }
265
266    fn matches(&self, name: &str) -> bool {
267        if name.contains('/') {
268            return false;
269        }
270        let mut current = vec![false; self.tokens.len() + 1];
271        let mut next = current.clone();
272        current[0] = true;
273        for ch in name.chars().map(Some).chain(std::iter::once(None)) {
274            // Branch targets only point forward. A single scan resolves all
275            // transitions that consume no characters before advancing the input.
276            for (index, token) in self.tokens.iter().enumerate() {
277                if !current[index] {
278                    continue;
279                }
280                match token {
281                    Token::Star => {
282                        current[index + 1] = true;
283                        next[index] |= ch.is_some();
284                    }
285                    Token::Split(branch) => {
286                        current[index + 1] = true;
287                        current[*branch] = true;
288                    }
289                    Token::Jump(target) => current[*target] = true,
290                    Token::Any => next[index + 1] |= ch.is_some(),
291                    Token::Literal(literal) => next[index + 1] |= ch == Some(*literal),
292                    Token::Class { ranges, negated } => {
293                        next[index + 1] |= ch.is_some_and(|ch| {
294                            ranges.iter().any(|(lo, hi)| *lo <= ch && ch <= *hi) != *negated
295                        });
296                    }
297                }
298            }
299            if ch.is_none() {
300                break;
301            }
302            std::mem::swap(&mut current, &mut next);
303            next.fill(false);
304        }
305        current[self.tokens.len()]
306    }
307}

309fn escaped_char(chars: &mut std::iter::Peekable<std::str::Chars<'_>>) -> Result<char> {
310    chars
311        .next()
312        .ok_or_else(|| invalid("trailing backslash in glob path component"))
313}
314
315fn parse_class(chars: &mut std::iter::Peekable<std::str::Chars<'_>>) -> Result<Token> {
316    let negated = chars.next_if(|ch| matches!(ch, '!' | '^')).is_some();
317    let mut ranges = Vec::new();
318    while let Some(ch) = chars.next() {
319        if ch == ']' && !ranges.is_empty() {
320            return Ok(Token::Class { ranges, negated });
321        }
322        let start = if ch == '\\' { escaped_char(chars)? } else { ch };
323        if chars.next_if_eq(&'-').is_none() {
324            ranges.push((start, start));
325            continue;
326        }
327        let end = match chars.peek() {
328            Some(']') => {
329                ranges.push((start, start));
330                ranges.push(('-', '-'));
331                continue;
332            }
333            Some('\\') => {
334                chars.next();
335                escaped_char(chars)?
336            }
337            _ => escaped_char(chars)?,
338        };
339        if end < start {
340            return Err(invalid("reversed character range in glob pattern"));
341        }
342        ranges.push((start, end));
343    }
344    Err(invalid("unterminated character class in glob pattern"))
345}
346
347fn invalid(message: &str) -> Error {
348    Error::new(ErrorKind::ConfigInvalid, message)
349}

351#[cfg(test)]
352mod tests {
353    use super::*;
354
355    #[test]
356    fn matches_one_component_by_unicode_scalar() {
357        for (pattern, name, expected) in [
358            ("*.jpg", "a.jpg", true),
359            ("*.jpg", "dir/a.jpg", false),
360            ("?.jpg", "猫.jpg", true),
361            ("?.jpg", "猫犬.jpg", false),
362            ("?.jpg", "/.jpg", false),
363            ("[à-é]", "é", true),
364            ("[!a-z]", "é", true),
365            ("[!a-z]", "a", false),
366            ("[]a]", "]", true),
367            ("[**]", "*", true),
368            (r"[a\-z]", "-", true),
369            (r"[a\-z]", "m", false),
370            (r"\*\?\[\\", "*?[\\", true),
371            (r"foo\**", "foo*猫", true),
372            ("*.{jpg,{png,gif}}", "猫.gif", true),
373            ("{,é}.txt", ".txt", true),
374            ("{a,b}{1,2}", "b2", true),
375            ("{a,b}{1,2}", "a", false),
376        ] {
377            assert_eq!(
378                GlobPattern::new(pattern).unwrap().matches(name, false),
379                expected,
380                "{pattern:?} against {name:?}"
381            );
382        }
383    }
384
385    #[test]
386    fn rejects_invalid_patterns() {
387        for pattern in [
388            "",
389            "/",
390            "a**b",
391            "a/***/b",
392            "file\\",
393            "[",
394            "[]",
395            "[z-a]",
396            "{one,two",
397            "../*.jpg",
398            r"\../*.jpg",
399        ] {
400            assert_eq!(
401                GlobPattern::new(pattern).unwrap_err().kind(),
402                ErrorKind::ConfigInvalid,
403                "{pattern:?}"
404            );
405        }
406    }
407
408    #[test]
409    fn only_leading_literal_directories_can_be_skipped() {
410        let pattern = GlobPattern::new(r"a/\[b\]/**/*.json").unwrap();
411        assert_eq!(
412            pattern.literal_directories().collect::<Vec<_>>(),
413            ["a", "[b]"]
414        );
415        assert!(pattern.matches("a/[b]/x.json", false));
416        assert!(pattern.matches("a/[b]/c/x.json", false));
417        assert!(!pattern.matches("a/c/x.json", false));
418    }
