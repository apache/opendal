24/// BatchDelete is used to implement [`oio::Delete`] based on batch delete operation.
25///
26/// OneShotDeleter will perform delete operation while calling `close`.
27pub trait BatchDelete: Send + Sync + Unpin + 'static {
28    /// delete_once delete one path at once.
29    ///
30    /// Implementations should make sure that the data is deleted correctly at once.
31    ///
32    /// BatchDeleter may call this method while there are only one path to delete.
33    fn delete_once(
34        &self,
35        path: String,
36        args: OpDelete,
37    ) -> impl Future<Output = Result<()>> + MaybeSend;
38
39    /// delete_batch delete multiple paths at once.
40    ///
41    /// Implementations must account for every input path in either
42    /// `succeeded` or `failed`; `BatchDeleter` rejects a result that drops
43    /// or invents paths.
44    fn delete_batch(
45        &self,
46        batch: Vec<(String, OpDelete)>,
47    ) -> impl Future<Output = Result<BatchDeleteResult>> + MaybeSend;
48}
49
50/// BatchDeleteResult is the result of batch delete operation.
51#[derive(Default)]
52pub struct BatchDeleteResult {
53    /// Collection of successful deletions, containing tuples of (path, args)
54    pub succeeded: Vec<(String, OpDelete)>,
55    /// Collection of failed deletions, containing tuples of (path, args, error)
56    pub failed: Vec<(String, OpDelete, Error)>,
57}
58
59/// BatchDeleter is used to implement [`oio::Delete`] based on batch delete.
60pub struct BatchDeleter<D: BatchDelete> {
61    inner: D,
62    buffer: HashSet<(String, OpDelete)>,
63    max_batch_size: usize,
64}
65
66impl<D: BatchDelete> BatchDeleter<D> {
67    /// Create a new batch deleter.
68    pub fn new(inner: D, max_batch_size: Option<usize>) -> Self {
69        debug_assert!(
70            max_batch_size.is_some(),
71            "BatchDeleter requires delete_max_size to be configured"
72        );
73        let max_batch_size = max_batch_size.unwrap_or(1);
74
75        Self {
76            inner,
77            buffer: HashSet::default(),
78            max_batch_size,
79        }
80    }
81
82    async fn flush_buffer(&mut self) -> Result<usize> {
83        if self.buffer.is_empty() {
84            return Ok(0);
85        }
86
87        if self.buffer.len() == 1 {
88            let (path, args) = self
89                .buffer
90                .iter()
91                .next()
92                .expect("the delete buffer size must be 1")
93                .clone();
94            self.inner.delete_once(path, args).await?;
95            self.buffer.clear();
96            return Ok(1);
97        }
98
99        let batch_len = self.buffer.len();
100        let batch = self.buffer.iter().cloned().collect();
101        let result = self.inner.delete_batch(batch).await?;
102
103        // The backend must account for every input path. A result that
104        // drops or invents paths is a bug, so fail instead of losing
105        // buffered deletes or retrying forever.
106        if result.succeeded.len() + result.failed.len() != batch_len {
107            return Err(Error::new(
108                ErrorKind::Unexpected,
109                "batch delete result size mismatch",
110            ));
111        }
112
113        for item in result.succeeded {
114            self.buffer.remove(&item);
115        }
116        let deleted = batch_len - self.buffer.len();
117
118        // Failed paths remain queued so a later flush can retry them.
119        let mut first_temporary_error = None;
120        for (path, op, err) in result.failed {
121            let err = err
122                .with_context("path", path)
123                .with_context("version", op.version().unwrap_or("<latest>"));
124            if !err.is_temporary() {
125                return Err(err);
126            }
127            first_temporary_error.get_or_insert(err);
128        }
129
130        if deleted == 0 {
131            return Err(first_temporary_error.unwrap_or_else(|| {
132                Error::new(
133                    ErrorKind::Unexpected,
134                    "batch delete removed zero buffered paths",
135                )
136            }));
137        }
138
139        Ok(deleted)
140    }
141}
142
143impl<D: BatchDelete> oio::Delete for BatchDeleter<D> {
144    async fn delete(&mut self, path: &str, mut args: OpDelete) -> Result<()> {
145        // A concurrent task fills its own handle, then lets close own the flush
146        // and any layer-level retries of that flush.
147        let deferred = args.take_deferred_batch_flush();
148        if deferred && self.buffer.len() >= self.max_batch_size {
149            // An outer layer may advertise a larger batch than this backend accepts.
150            self.flush_buffer().await?;
151        }
152        self.buffer.insert((path.to_string(), args));
153        if !deferred && self.buffer.len() >= self.max_batch_size {
154            self.flush_buffer().await?;
155        }
156
157        Ok(())
158    }
159
160    async fn close(&mut self) -> Result<()> {
161        loop {
162            let deleted = self.flush_buffer().await?;
163
164            if self.buffer.is_empty() {
165                break;
166            }
167
168            if deleted == 0 {
169                return Err(Error::new(
170                    ErrorKind::Unexpected,
171                    "batch delete made no progress while buffer remains",
172                ));
173            }
174        }
175
176        Ok(())
177    }
178}

188    type BatchResponse = Box<dyn Fn(Vec<(String, OpDelete)>) -> BatchDeleteResult + Send + Sync>;
189
190    /// MockBatchDelete replays scripted `delete_batch` responses and records
191    /// calls so tests can assert how `BatchDeleter` retries failed paths.
192    struct MockBatchDelete {
193        delete_batch_calls: Mutex<Vec<Vec<String>>>,
194        delete_once_paths: Mutex<Vec<String>>,
195        responses: Mutex<VecDeque<BatchResponse>>,
196    }
197
198    impl MockBatchDelete {
199        fn new() -> Self {
200            Self {
201                delete_batch_calls: Mutex::new(Vec::new()),
202                delete_once_paths: Mutex::new(Vec::new()),
203                responses: Mutex::new(VecDeque::new()),
204            }
205        }
206
207        fn push_response(
208            &self,
209            respond: impl Fn(Vec<(String, OpDelete)>) -> BatchDeleteResult + Send + Sync + 'static,
210        ) {
211            self.responses.lock().unwrap().push_back(Box::new(respond));
212        }
213    }
214
215    impl BatchDelete for MockBatchDelete {
216        async fn delete_once(&self, path: String, _: OpDelete) -> Result<()> {
217            self.delete_once_paths.lock().unwrap().push(path);
218            Ok(())
219        }
220
221        async fn delete_batch(&self, batch: Vec<(String, OpDelete)>) -> Result<BatchDeleteResult> {
222            let paths = batch.iter().map(|(path, _)| path.clone()).collect();
223            self.delete_batch_calls.lock().unwrap().push(paths);
224            let respond = self
225                .responses
226                .lock()
227                .unwrap()
228                .pop_front()
229                .expect("delete_batch called without a scripted response");
230            Ok(respond(batch))
231        }
232    }
233
234    /// Split a batch by path into `succeeded` and `failed`. Items listed in
235    /// neither are dropped from the result, like a buggy backend would.
236    /// `failed` entries are `(path, kind, temporary)`.
237    fn split(
238        succeeded: Vec<&'static str>,
239        failed: Vec<(&'static str, ErrorKind, bool)>,
240    ) -> impl Fn(Vec<(String, OpDelete)>) -> BatchDeleteResult + Send + Sync {
241        move |batch| {
242            let mut result = BatchDeleteResult::default();
243            for (path, op) in batch {
244                if let Some((_, kind, temporary)) =
245                    failed.iter().find(|(p, ..)| *p == path.as_str())
246                {
247                    result.failed.push((
248                        path,
249                        op,
250                        Error::new(*kind, "delete failed").with_temporary(*temporary),
251                    ));
252                } else if succeeded.iter().any(|p| *p == path) {
253                    result.succeeded.push((path, op));
254                }
255            }
256            result
257        }
258    }

260    /// A batch where every item fails permanently must surface the real
261    /// per-object error, not a masked `Unexpected`.
262    #[tokio::test]
263    async fn test_batch_delete_all_failed_permanent() {
264        let inner = MockBatchDelete::new();
265        inner.push_response(split(
266            vec![],
267            vec![
268                ("a", ErrorKind::ConditionNotMatch, false),
269                ("b", ErrorKind::PermissionDenied, false),
270                ("c", ErrorKind::ConditionNotMatch, false),
271            ],
272        ));
273        let mut deleter = BatchDeleter::new(inner, Some(10));
274
275        for path in ["a", "b", "c"] {
276            deleter.delete(path, OpDelete::new()).await.unwrap();
277        }
278        let err = deleter.close().await.unwrap_err();
279
280        assert!(
281            matches!(
282                err.kind(),
283                ErrorKind::ConditionNotMatch | ErrorKind::PermissionDenied
284            ),
285            "permanent batch failure must keep its original kind: {err:?}"
286        );
287        assert!(!err.is_temporary());
288        let rendered = err.to_string();
289        assert!(rendered.contains("path: "), "missing path context: {err}");
290        assert!(rendered.contains("version: <latest>"));
291    }

293    /// A batch where every item fails with a temporary error must return a
294    /// temporary error and keep all paths queued, so a retry layer can
295    /// replay the same batch on a later `close`.
296    #[tokio::test]
297    async fn test_batch_delete_all_failed_temporary_retries() {
298        let inner = MockBatchDelete::new();
299        inner.push_response(split(
300            vec![],
301            vec![
302                ("a", ErrorKind::RateLimited, true),
303                ("b", ErrorKind::RateLimited, true),
304            ],
305        ));
306        inner.push_response(split(vec!["a", "b"], vec![]));
307        let mut deleter = BatchDeleter::new(inner, Some(10));
308
309        deleter.delete("a", OpDelete::new()).await.unwrap();
310        deleter.delete("b", OpDelete::new()).await.unwrap();
311        let err = deleter.close().await.unwrap_err();
312
313        assert_eq!(err.kind(), ErrorKind::RateLimited);
314        assert!(
315            err.is_temporary(),
316            "temporary batch failure must stay retryable: {err:?}"
317        );
318        assert!(err.to_string().contains("path: "));
319        assert_eq!(deleter.buffer.len(), 2);
320
321        deleter.close().await.unwrap();
322        assert_eq!(deleter.inner.delete_batch_calls.lock().unwrap().len(), 2);
323        assert!(deleter.buffer.is_empty());

343    /// A permanent per-object error surfaces with its original kind even
344    /// when the rest of the batch succeeded.
345    #[tokio::test]
346    async fn test_batch_delete_partial_failure_permanent() {
347        let inner = MockBatchDelete::new();
348        inner.push_response(split(
349            vec!["a", "b"],
350            vec![("c", ErrorKind::ConditionNotMatch, false)],
351        ));
352        let mut deleter = BatchDeleter::new(inner, Some(10));
353
354        for path in ["a", "b", "c"] {
355            deleter.delete(path, OpDelete::new()).await.unwrap();
356        }
357        let err = deleter.close().await.unwrap_err();
358
359        assert_eq!(err.kind(), ErrorKind::ConditionNotMatch);

363        assert!(deleter.buffer.iter().all(|(p, _)| p == "c"));

366    /// Temporary per-object errors alongside progress keep the failed
367    /// paths queued, so `close` retries them until the buffer drains.
368    #[tokio::test]
369    async fn test_batch_delete_partial_failure_temporary_retries() {
370        let inner = MockBatchDelete::new();
371        inner.push_response(split(
372            vec!["a"],
373            vec![
374                ("b", ErrorKind::RateLimited, true),
375                ("c", ErrorKind::RateLimited, true),
376            ],
377        ));
378        inner.push_response(split(vec!["b"], vec![("c", ErrorKind::RateLimited, true)]));
379        let mut deleter = BatchDeleter::new(inner, Some(10));
380
381        for path in ["a", "b", "c"] {
382            deleter.delete(path, OpDelete::new()).await.unwrap();
383        }
384        deleter.close().await.unwrap();
385
386        assert_eq!(deleter.inner.delete_batch_calls.lock().unwrap().len(), 2);
387        // "c" drains through the single-object `delete_once` path.

390            &["c"]
391        );
392        assert!(deleter.buffer.is_empty());
