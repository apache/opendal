22/// Join a normalized root and operation path, omitting the leading `/`.
23///
24/// Pass a root such as `/` or `/abc/def/`. An operation path of `/` addresses
25/// the root itself.
26pub fn build_absolute_path(root: &str, path: &str) -> String {
27    debug_assert!(root.starts_with('/'), "root must start with /");
28    debug_assert!(root.ends_with('/'), "root must end with /");
29
30    let p = root[1..].to_string();
31
32    if path == "/" {
33        p
34    } else {
35        debug_assert!(!path.starts_with('/'), "path must not start with /");
36        p + path
37    }
38}

40/// Join a normalized root and operation path, keeping the leading `/`.
41///
42/// Pass a root such as `/` or `/abc/def/`. An operation path of `/` addresses
43/// the root itself.
44pub fn build_rooted_absolute_path(root: &str, path: &str) -> String {
45    debug_assert!(root.starts_with('/'), "root must start with /");
46    debug_assert!(root.ends_with('/'), "root must end with /");
47
48    let p = root.to_string();
49
50    if path == "/" {
51        p
52    } else {
53        debug_assert!(!path.starts_with('/'), "path must not start with /");
54        p + path
55    }
56}

58/// Strip a normalized root prefix from a path.
59///
60/// Pass a root such as `/` or `/abc/def/`. The path must start with that
61/// prefix and may omit its leading `/`. The returned path is relative to root.
62pub fn build_relative_path(root: &str, path: &str) -> String {
63    debug_assert!(root != path, "get rel path with root is invalid");
64
65    if path.starts_with('/') {
66        debug_assert!(
67            path.starts_with(root),
68            "path {path} doesn't start with root {root}"
69        );
70        path[root.len()..].to_string()
71    } else {
72        debug_assert!(
73            path.starts_with(&root[1..]),
74            "path {path} doesn't start with root {root}"
75        );
76        path[root.len() - 1..].to_string()
77    }
78}

101/// Normalize an operation path relative to the operator's root.
102///
103/// Strip leading `/`, collapse repeated `/`, and remove `.` components.
104/// Preserve all other components verbatim, including whitespace and `..`.
105/// A trailing `/` marks a directory and is preserved. A path with no remaining
106/// components becomes `/`.
107pub fn normalize_path(path: &str) -> String {
108    let path = path.trim_start_matches('/');
109    if path.is_empty() {
110        return "/".to_string();
111    }
112
113    let mut p = path
114        .split('/')
115        .filter(|v| !v.is_empty() && *v != ".")
116        .collect::<Vec<&str>>()
117        .join("/");
118
119    if p.is_empty() {
120        return "/".to_string();
121    }
122    if path.ends_with('/') {
123        p.push('/');
124    }
125
126    p
127}

129/// Normalize a service root to `/` or a path with one leading and trailing `/`.
130///
131/// Collapse repeated `/` and remove `.` components. Preserve all other
132/// components verbatim, including whitespace and `..`.
133pub fn normalize_root(v: &str) -> String {
134    let mut v = v
135        .split('/')
136        .filter(|v| !v.is_empty() && *v != ".")
137        .collect::<Vec<&str>>()
138        .join("/");
139    if !v.starts_with('/') {
140        v.insert(0, '/');
141    }
142    if !v.ends_with('/') {
143        v.push('/')
144    }
145    v
146}

148/// Join an OpenDAL path to a local filesystem root.
149///
150/// This function strips trailing `/` characters, so the OpenDAL root path `/`
151/// resolves to `root`. It returns [`ErrorKind::NotFound`] for parent-directory
152/// components, root-directory components, or Windows path prefixes.
153///
154/// The check is lexical. The filesystem resolves symlinks.
155pub fn confined_join(root: &Path, path: &str) -> Result<PathBuf> {
156    let trimmed = path.trim_end_matches('/');
157    if Path::new(trimmed).components().any(|component| {
158        matches!(
159            component,
160            Component::ParentDir | Component::RootDir | Component::Prefix(_)
161        )
162    }) {
163        return Err(
164            Error::new(ErrorKind::NotFound, "path escapes the configured root")
165                .with_context("path", path),
166        );
167    }
168    Ok(root.join(trimmed))
169}

171/// Get basename from path.
172pub fn get_basename(path: &str) -> &str {
173    // Handle root case
174    if path == "/" {
175        return "/";
176    }
177
178    // Handle file case
179    if !path.ends_with('/') {
180        return path
181            .split('/')
182            .next_back()
183            .expect("file path without name is invalid");
184    }
185
186    // The idx of second `/` if path in reserve order.
187    // - `abc/` => `None`
188    // - `abc/def/` => `Some(3)`
189    let idx = path[..path.len() - 1].rfind('/').map(|v| v + 1);
190
191    match idx {
192        Some(v) => {
193            let (_, name) = path.split_at(v);
194            name
195        }
196        None => path,
197    }
198}

200/// Get parent from path.
201pub fn get_parent(path: &str) -> &str {
202    if path == "/" {
203        return "/";
204    }
205
206    if !path.ends_with('/') {
207        // The idx of first `/` if path in reserve order.
208        // - `abc` => `None`
209        // - `abc/def` => `Some(3)`
210        let idx = path.rfind('/');
211
212        return match idx {
213            Some(v) => {
214                let (parent, _) = path.split_at(v + 1);
215                parent
216            }
217            None => "/",
218        };
219    }
220
221    // The idx of second `/` if path in reserve order.
222    // - `abc/` => `None`
223    // - `abc/def/` => `Some(3)`
224    let idx = path[..path.len() - 1].rfind('/').map(|v| v + 1);
225
226    match idx {
227        Some(v) => {
228            let (parent, _) = path.split_at(v);
229            parent
230        }
231        None => "/",
232    }
233}

235// Sets the size of random generated postfix for random file names
236const RANDOM_TMP_PATH_POSTFIX_LENGTH: usize = 8;
237// Allowed characters for choices in a random-generated char
238const CHARS: &[u8] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
239const CHARS_LENGTH: u64 = CHARS.len() as u64;
240
241/// Build a temporary path of a file path.
242///
243/// `build_tmp_path_of` appends a dot following a random generated postfix.
244/// Don't use it with a path to a folder.
245#[inline]
246pub fn build_tmp_path_of(path: &str) -> String {
247    let name = get_basename(path);
248    let mut buf = String::with_capacity(name.len() + RANDOM_TMP_PATH_POSTFIX_LENGTH);
249    buf.push_str(name);
250    buf.push('.');
251
252    // Uses `std` for the random number generator instead of external crates.
253    //
254    // `RandomState::new` builds a hasher that generates a different random sequence each time.
255    // Calling `RandomState::new` each time produces a `RandomState` from
256    // a per-thread pseudo seed pools that Rust manages.
257    // The default hasher, `SipHasher13`, has some notable properties:
258    //
259    // 1. `fastrand` is roughly 10x faster than `SipHasher13`, but it adds an extra dependency.
260    // 2. While `fastrand` is faster, `SipHasher13` is fast enough for our needs since
261    //    we're only generating a few characters.
262    // 3. This is not a cryptographically secure pseudorandom number generator (CSPRNG).
263    //
264    // If we need stronger randomness in the future, we can:
265    // 1. Increase the output length.
266    // 2. Use the `getrandom` crate to source randomness from the OS.
267    for _ in 0..RANDOM_TMP_PATH_POSTFIX_LENGTH {
268        let random = std::collections::hash_map::RandomState::new()
269            .build_hasher()
270            .finish();
271        let choice: usize = (random % CHARS_LENGTH).try_into().unwrap();
272        buf.push(CHARS[choice] as char);
273    }
274
275    buf
276}

278/// Validate given path is match with given EntryMode.
279#[inline]
280pub fn validate_path(path: &str, mode: EntryMode) -> bool {
281    debug_assert!(!path.is_empty(), "input path should not be empty");
282
283    match mode {
284        EntryMode::FILE => !path.ends_with('/'),
285        EntryMode::DIR => path.ends_with('/'),
286        EntryMode::Unknown => false,
287    }
288}
