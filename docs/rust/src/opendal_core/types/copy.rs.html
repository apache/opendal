29/// `Copier` drives a long-running copy operation one step at a time.
30///
31/// - `Copier` implements `Stream<Item = Result<usize>>`.
32/// - `Some(Ok(n))` means the copy operation made progress by `n` bytes.
33/// - `None` means the copy operation has completed.
34///
35/// A copier completes the copy before it terminates: the inner copier is closed
36/// and its completion response is processed before the stream yields `None`, and
37/// [`Copier::close`] returns the metadata from that response. Aborting or
38/// failing a copier makes it terminal: later [`Copier::next`] calls and stream
39/// polls return `None` without driving the inner copier again, and
40/// [`Copier::close`] fails instead of reporting a result.
41pub struct Copier {
42    /// The current state of this copier.
43    state: State,
44}
45
46/// State of a [`Copier`].
47///
48/// While the copy makes progress the copier alternates between `Idle` and
49/// `InFlight`. `Closed` and `Aborted` end the copy, and `Failed` records that
50/// the copy cannot continue, though the inner copier can still be aborted.
51enum State {
52    /// The inner copier is ready to be advanced.
53    Idle(oio::Copier),
54    /// A step is in flight and owns the inner copier.
55    ///
56    /// The step owns the inner copier, so a canceled stream poll never drops
57    /// progress the inner copier already made.
58    InFlight(BoxedStaticFuture<Step>),
59    /// The copy completed and `close_dyn` returned its metadata.
60    Closed(Metadata),
61    /// The copy was aborted by the caller.
62    Aborted,
63    /// The copy, or the final `close_dyn`, failed.
64    ///
65    /// The copier is kept so the caller can still abort a partially created
66    /// server-side copy.
67    Failed(oio::Copier),
68}
69
70/// The outcome of one [`advance`] step.
71enum Step {
72    /// The copy made progress by this many bytes and can be advanced again.
73    Progress(oio::Copier, usize),
74    /// The copy completed and the inner copier has been closed.
75    Closed(Metadata),
76    /// Advancing or closing the copy failed.
77    ///
78    /// The copier is kept so the caller can still abort a partially created
79    /// server-side copy.
80    Failed(oio::Copier, Error),
81}
82
83/// Drive one step of `copier`: advance the copy and close it once it completes.
84async fn advance(mut copier: oio::Copier) -> Step {
85    match copier.next_dyn().await {
86        Ok(Some(n)) => Step::Progress(copier, n),
87        Ok(None) => match copier.close_dyn().await {
88            Ok(metadata) => Step::Closed(metadata),
89            Err(err) => Step::Failed(copier, err),
90        },
91        Err(err) => Step::Failed(copier, err),
92    }
93}

95impl fmt::Debug for Copier {
96    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
97        let state = match &self.state {
98            State::Idle(_) => "idle",
99            State::InFlight(_) => "in_flight",
100            State::Closed(_) => "closed",
101            State::Aborted => "aborted",
102            State::Failed(_) => "failed",
103        };
104
105        f.debug_struct("Copier").field("state", &state).finish()
106    }
107}

114impl Copier {
115    /// Create a new copier.
116    pub(crate) fn create(
117        ctx: OperationContext,
118        srv: Servicer,
119        from: &str,
120        to: &str,
121        args: OpCopy,
122    ) -> Result<Self> {
123        let copier = srv.copy(&ctx, from, to, args)?;
124
125        Ok(Self {
126            state: State::Idle(copier),
127        })
128    }
129
130    /// Advance the copy operation by one step.
131    ///
132    /// Returns `None` once the copy completed or the copier became terminal.
133    pub async fn next(&mut self) -> Result<Option<usize>> {
134        match &self.state {
135            // Terminal copiers never drive the inner copier again.
136            State::Closed(_) | State::Aborted | State::Failed(_) => Ok(None),
137            // A step started by a canceled stream poll is awaited instead of
138            // dropped, so the progress it already made is not lost.
139            State::Idle(_) | State::InFlight(_) => self.step().await,
140        }
141    }
142
143    /// Abort the pending copy operation.
144    ///
145    /// Aborting a completed copier is a no-op. After a successful abort, later
146    /// [`Copier::next`] calls and stream polls return `None`, and
147    /// [`Copier::close`] fails.
148    pub async fn abort(&mut self) -> Result<()> {
149        match &self.state {
150            // The copy already completed or was aborted, there is nothing left
151            // to abort.
152            State::Closed(_) | State::Aborted => return Ok(()),
153            State::Idle(_) | State::InFlight(_) | State::Failed(_) => {}
154        }
155
156        if matches!(self.state, State::InFlight(_)) {
157            // Await the in-flight step before aborting: the step owns the inner
158            // copier, so aborting without it would silently do nothing. The
159            // step error is not returned here because `abort` reports whether
160            // the server-side copy was cleaned up, and callers retry `abort` on
161            // its own error.
162            let _ = self.step().await;
163        }
164
165        // A failed close keeps the copier, so both states can still hold a copy
166        // that needs to be aborted.
167        let copier = match &mut self.state {
168            State::Idle(copier) | State::Failed(copier) => copier,
169            // The step completed the copy, so there is nothing left to abort.
170            _ => return Ok(()),
171        };
172
173        // Keep the copier when the abort fails: the inner copier is retryable,
174        // so the next `abort` call can clean up the same server-side copy.
175        copier.abort_dyn().await?;
176        self.state = State::Aborted;
177        Ok(())
178    }
179
180    /// Close the copier and return metadata from the server-side completion response.
181    ///
182    /// This drives the copy to completion when it has not finished yet.
183    pub async fn close(&mut self) -> Result<Metadata> {
184        match &self.state {
185            State::Closed(metadata) => return Ok(metadata.clone()),
186            State::Aborted => {
187                return Err(Error::new(
188                    ErrorKind::Unexpected,
189                    "copier has already been aborted and can't be closed",
190                ));
191            }
192            State::Failed(_) => {
193                return Err(Error::new(
194                    ErrorKind::Unexpected,
195                    "copier has already failed and can't be closed",
196                ));
197            }
198            State::Idle(_) | State::InFlight(_) => {}
199        }
200
201        while self.next().await?.is_some() {}
202
203        match &self.state {
204            State::Closed(metadata) => Ok(metadata.clone()),
205            _ => Ok(MetadataBuilder::unknown().build()),
206        }
207    }
208
209    /// Run one step of the copy and record its outcome.
210    ///
211    /// The step lives in the state before it is awaited, so canceling the
212    /// caller's future never drops progress the inner copier already made.
213    async fn step(&mut self) -> Result<Option<usize>> {
214        self.start_step();
215
216        // Terminal copiers have no step left to run.
217        let State::InFlight(fut) = &mut self.state else {
218            return Ok(None);
219        };
220        let step = fut.await;
221
222        self.apply_step(step)
223    }
224
225    /// Start a step unless one is in flight or the copier is terminal.
226    fn start_step(&mut self) {
227        // The step takes ownership of the inner copier, so the state has to move
228        // out before the next state can be built from it. `State::Aborted` only
229        // fills the slot during that move, and no await point is crossed here.
230        self.state = match std::mem::replace(&mut self.state, State::Aborted) {
231            State::Idle(copier) => State::InFlight(Box::pin(advance(copier))),
232            // An in-flight step is resumed, and terminal copiers have no step
233            // left to start.
234            state => state,
235        };
236    }
237
238    /// Record the outcome of one step and transition to the matching state.
239    fn apply_step(&mut self, step: Step) -> Result<Option<usize>> {
240        match step {
241            Step::Progress(copier, n) => {
242                self.state = State::Idle(copier);
243                Ok(Some(n))
244            }
245            Step::Closed(metadata) => {
246                self.state = State::Closed(metadata);
247                Ok(None)
248            }
249            Step::Failed(copier, err) => {
250                self.state = State::Failed(copier);
251                Err(err)
252            }
253        }
254    }
255}
256
257impl Stream for Copier {
258    type Item = Result<usize>;
259
260    fn poll_next(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
261        let this = self.get_mut();
262        this.start_step();
263
264        // Terminal copiers have no step left to run.
265        let State::InFlight(fut) = &mut this.state else {
266            return Poll::Ready(None);
267        };
268        let step = ready!(fut.as_mut().poll(cx));
269
270        match this.apply_step(step) {
271            Ok(Some(n)) => Poll::Ready(Some(Ok(n))),
272            Ok(None) => Poll::Ready(None),
273            Err(err) => Poll::Ready(Some(Err(err))),
274        }
275    }
276}

291    /// Content length returned by [`MockCopy::close`].
292    const CLOSE_CONTENT_LENGTH: u64 = 42;
293
294    /// Call counters shared with the mock copier, so tests can assert how often
295    /// each inner operation ran.
296    #[derive(Default)]
297    struct MockCalls {
298        next: AtomicUsize,
299        close: AtomicUsize,
300        abort: AtomicUsize,
301    }
302
303    impl MockCalls {
304        fn next(&self) -> usize {
305            self.next.load(Ordering::Relaxed)
306        }
307
308        fn close(&self) -> usize {
309            self.close.load(Ordering::Relaxed)
310        }
311
312        fn abort(&self) -> usize {
313            self.abort.load(Ordering::Relaxed)
314        }
315    }
316
317    struct MockCopy {
318        /// Results returned by `next`, in order. A missing entry completes the copy.
319        steps: VecDeque<Result<Option<usize>>>,
320        /// Whether the first `next` call returns `Poll::Pending` once.
321        pending_once: bool,
322        /// Whether `close` fails instead of returning metadata.
323        close_fails: bool,
324        calls: Arc<MockCalls>,
325    }
326
327    impl MockCopy {
328        fn new(
329            calls: &Arc<MockCalls>,
330            steps: impl IntoIterator<Item = Result<Option<usize>>>,
331        ) -> Self {
332            Self {
333                steps: steps.into_iter().collect(),
334                pending_once: false,
335                close_fails: false,
336                calls: calls.clone(),
337            }
338        }
339    }
340
341    impl oio::Copy for MockCopy {
342        async fn next(&mut self) -> Result<Option<usize>> {
343            self.calls.next.fetch_add(1, Ordering::Relaxed);
344
345            if self.pending_once {
346                self.pending_once = false;
347                yield_once().await;
348            }
349
350            self.steps.pop_front().unwrap_or(Ok(None))
351        }
352
353        async fn close(&mut self) -> Result<Metadata> {
354            self.calls.close.fetch_add(1, Ordering::Relaxed);
355
356            if self.close_fails {
357                return Err(Error::new(ErrorKind::Unexpected, "mock close failed"));
358            }
359
360            let mut builder = MetadataBuilder::unknown();
361            builder.set_file(CLOSE_CONTENT_LENGTH);
362            Ok(builder.build())
363        }
364
365        async fn abort(&mut self) -> Result<()> {
366            self.calls.abort.fetch_add(1, Ordering::Relaxed);
367            Ok(())
368        }
369    }
370
371    /// Return `Poll::Pending` on the first poll and `Poll::Ready` afterwards.
372    ///
373    /// Lets a test leave a `next` step in flight and cancel the stream poll.
374    async fn yield_once() {
375        let mut yielded = false;
376
377        std::future::poll_fn(move |cx| {
378            if yielded {
379                return Poll::Ready(());
380            }
381
382            yielded = true;
383            cx.waker().wake_by_ref();
384            Poll::Pending
385        })
386        .await;
387    }
388
389    fn mock_copier(mock: MockCopy) -> Copier {
390        Copier {
391            state: State::Idle(Box::new(mock)),
392        }
393    }

395    #[tokio::test]
396    async fn test_copier_stream_closes_inner_copier_once() {
397        let calls = Arc::new(MockCalls::default());
398        let copier = mock_copier(MockCopy::new(&calls, [Ok(Some(4)), Ok(Some(6)), Ok(None)]));
399
400        // Driving the copier as a stream must complete the copy, not only end
401        // the stream: `close_dyn` is the step that finalizes it.
402        let sizes = copier.try_collect::<Vec<_>>().await.unwrap();
403
404        assert_eq!(sizes, vec![4, 6]);
405        assert_eq!(calls.next(), 3);
406        assert_eq!(calls.close(), 1);
407    }

409    #[tokio::test]
410    async fn test_copier_stream_repoll_after_close() {
411        let calls = Arc::new(MockCalls::default());
412        let mut copier = mock_copier(MockCopy::new(&calls, [Ok(Some(4)), Ok(None)]));
413
414        let sizes = (&mut copier).try_collect::<Vec<_>>().await.unwrap();
415        assert_eq!(sizes, vec![4]);
416        assert_eq!(calls.next(), 2);
417        assert_eq!(calls.close(), 1);
418        assert!(matches!(copier.state, State::Closed(_)));
419
420        // A completed stream stays completed: it returns `None` without driving

426        assert!(copier.next().await.unwrap().is_none());
427        assert_eq!(calls.next(), 2);
428        assert_eq!(calls.close(), 1);
429
430        // `close` reports the metadata returned by `close_dyn`.
431        assert_eq!(
432            copier.close().await.unwrap().content_length(),
433            CLOSE_CONTENT_LENGTH
434        );
435        assert_eq!(calls.close(), 1);
436    }

463    #[tokio::test]
464    async fn test_copier_abort_after_cancelled_poll() {
465        let calls = Arc::new(MockCalls::default());
466        let mut copier = mock_copier(MockCopy {
467            pending_once: true,
468            ..MockCopy::new(&calls, [Ok(Some(4))])
469        });
470
471        // Cancel the first poll while the `next` step is in flight.
472        assert!(poll!(StreamExt::next(&mut copier)).is_pending());
473        assert!(matches!(copier.state, State::InFlight(_)));
474
475        // `abort` must reach the inner copier instead of returning early
476        // because the in-flight step owns it.
477        copier.abort().await.unwrap();
478        assert_eq!(calls.abort(), 1);
479        assert!(matches!(copier.state, State::Aborted));
480
481        // The aborted copier is terminal.
482        assert!(copier.next().await.unwrap().is_none());
483        assert!(matches!(
484            poll!(StreamExt::next(&mut copier)),
485            Poll::Ready(None)
486        ));
487        assert!(copier.close().await.is_err());
488        assert_eq!(calls.next(), 1);
489        assert_eq!(calls.close(), 0);
490
491        // Aborting again is a no-op.
492        copier.abort().await.unwrap();
493        assert_eq!(calls.abort(), 1);
494    }
