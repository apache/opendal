18use std::fmt::Debug;
19use std::fmt::Formatter;
20use std::future::Future;
21use std::sync::Arc;
22use std::sync::OnceLock;
23use std::time::Duration;
24
25use super::*;
26use crate::Result;
27use crate::raw::BoxedStaticFuture;
28use crate::raw::MaybeSend;
29
30static DEFAULT_EXECUTOR: OnceLock<Executor> = OnceLock::new();
31
32/// Executor that runs futures in background.
33///
34/// Executor is created by users and used by opendal. So it's by design that Executor only
35/// expose constructor methods.
36///
37/// Executor will run futures in background and return a `Task` as handle to the future. Users
38/// can call `task.await` to wait for the future to complete or drop the `Task` to cancel it.
39#[derive(Clone)]
40pub struct Executor {
41    executor: Arc<dyn Execute>,
42}
43
44impl Debug for Executor {
45    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
46        write!(f, "Executor")
47    }
48}
49
50impl Default for Executor {
51    fn default() -> Self {
52        Self::new()
53    }
54}
55
56impl Executor {
57    /// Create an executor that uses the process-wide default for each task.
58    ///
59    /// Concurrent I/O returns an error if no default has been installed.
60    pub fn new() -> Self {
61        Self::with(DefaultExecutor)
62    }
63
64    /// Create a new executor with given execute impl.
65    pub fn with(exec: impl Execute) -> Self {
66        Self {
67            executor: Arc::new(exec),
68        }
69    }
70
71    /// Install the process-wide default executor.
72    ///
73    /// The first installed executor wins. Later calls are ignored.
74    /// The `opendal` facade installs Tokio when its `executors-tokio` and
75    /// `auto-register-services` features are both enabled.
76    pub fn install_default(exec: impl Execute) {
77        let _ = DEFAULT_EXECUTOR.set(Self::with(exec));
78    }
79
80    /// Return the inner executor.
81    pub fn into_inner(self) -> Arc<dyn Execute> {
82        self.executor
83    }
84
85    /// Return a reference to the inner executor.
86    pub fn inner(&self) -> &dyn Execute {
87        self.executor.as_ref()
88    }
89
90    /// Return the configured timeout for an individual I/O task.
91    pub(crate) fn timeout(&self) -> Option<BoxedStaticFuture<()>> {
92        self.executor.timeout()
93    }
94
95    /// Run given future in background immediately.
96    pub(crate) fn execute<F>(&self, f: F) -> Result<Task<F::Output>>
97    where
98        F: Future + MaybeSend + 'static,
99        F::Output: MaybeSend + 'static,
100    {
101        Task::submit(self.inner(), f)
102    }
103}
104
105struct DefaultExecutor;
106
107impl Execute for DefaultExecutor {
108    fn execute(&self, f: BoxedStaticFuture<()>) {
109        match DEFAULT_EXECUTOR.get() {
110            Some(executor) => executor.inner().execute(f),
111            None => panic!("default executor is not installed"),
112        }
113    }
114
115    fn try_execute(&self, f: BoxedStaticFuture<()>) -> Result<()> {
116        match DEFAULT_EXECUTOR.get() {
117            Some(executor) => executor.inner().try_execute(f),
118            None => ().try_execute(f),
119        }
120    }
121
122    fn timeout(&self) -> Option<BoxedStaticFuture<()>> {
123        DEFAULT_EXECUTOR.get()?.timeout()
124    }
125
126    fn sleep(&self, duration: Duration) -> Option<BoxedStaticFuture<()>> {
127        DEFAULT_EXECUTOR.get()?.inner().sleep(duration)
128    }
129}
