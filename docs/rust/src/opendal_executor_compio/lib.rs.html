24use std::sync::Arc;
25use std::time::Duration;
26
27use compio::dispatcher::Dispatcher;
28use futures_timer::Delay;
29use opendal_core::Error;
30use opendal_core::ErrorKind;
31use opendal_core::Execute;
32use opendal_core::Result;
33use opendal_core::raw::BoxedStaticFuture;
34
35/// Runs OpenDAL background tasks on a Compio [`Dispatcher`].
36///
37/// Timers run independently of the dispatcher, including with a single worker
38/// configured to run tasks sequentially.
39///
40/// Use [`Execute::try_execute`] to receive submission errors. The infallible
41/// [`Execute::execute`] method logs submission errors.
42#[derive(Clone)]
43pub struct CompioExecutor {
44    dispatcher: Arc<Dispatcher>,
45}
46
47impl CompioExecutor {
48    /// Create an executor backed by the given dispatcher.
49    pub fn new(dispatcher: Arc<Dispatcher>) -> Self {
50        Self { dispatcher }
51    }
52}
53
54impl Execute for CompioExecutor {
55    fn execute(&self, f: BoxedStaticFuture<()>) {
56        if let Err(err) = self.try_execute(f) {
57            log::error!("{err}");
58        }
59    }
60
61    fn try_execute(&self, f: BoxedStaticFuture<()>) -> Result<()> {
62        let _completion = self
63            .dispatcher
64            .dispatch(move || f)
65            .map_err(|_| Error::new(ErrorKind::Unexpected, "compio dispatcher is unavailable"))?;
66        Ok(())
67    }
68
69    fn sleep(&self, duration: Duration) -> Option<BoxedStaticFuture<()>> {
70        Some(Box::pin(Delay::new(duration)))
71    }
72}
