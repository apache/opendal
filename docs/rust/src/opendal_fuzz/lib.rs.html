18//! Shared fuzz input generation for the fuzz targets.
19//!
20//! This crate intentionally does not depend on `libfuzzer-sys` so that its
21//! unit tests can run with the standard test harness.
22
23use arbitrary::Arbitrary;
24use arbitrary::Unstructured;
25use opendal::Operator;
26use opendal::Result;
27use opendal::tests::WriteAction;
28use opendal::tests::WriteChecker;
29
30/// Upper bound of the total bytes one `FuzzWriterInput` may write across all
31/// its write actions.
32const MAX_TOTAL_BYTES: usize = 4 * 1024 * 1024;
33/// Upper bound of the total bytes produced when an explicit chunk size is
34/// set, expressed as the number of chunks. A tiny `buffer` must not turn the
35/// byte budget into millions of underlying writes (OSS-Fuzz issue
36/// 530185493, apache/opendal#7848).
37const MAX_CHUNKS: usize = 256;
38
39/// Structured input for the `fuzz_writer` target.
40#[derive(Debug, Clone)]
41pub struct FuzzWriterInput {
42    /// Write actions to perform, in order.
43    pub actions: Vec<WriteAction>,
44    /// Optional chunk size passed to `Writer::chunk`.
45    pub buffer: Option<usize>,
46    /// Optional concurrency passed to `Writer::concurrent`.
47    pub concurrent: Option<usize>,
48}
49
50impl Arbitrary<'_> for FuzzWriterInput {
51    fn arbitrary(u: &mut Unstructured<'_>) -> arbitrary::Result<Self> {
52        let mut actions = vec![];
53        let buffer = if u.int_in_range(0..=1)? == 1 {
54            Some(u.int_in_range(1..=8 * 1024 * 1024)?)
55        } else {
56            None
57        };
58        let concurrent = if u.int_in_range(0..=1)? == 1 {
59            Some(u.int_in_range(0..=16)?)
60        } else {
61            None
62        };
63
64        // Each action produces at least one underlying write of `buffer`
65        // bytes, so an explicit chunk size additionally caps the total
66        // bytes at `buffer * MAX_CHUNKS`.
67        let mut remaining = buffer.map_or(MAX_TOTAL_BYTES, |chunk: usize| {
68            chunk.saturating_mul(MAX_CHUNKS).min(MAX_TOTAL_BYTES)
69        });
70
71        let count = u.int_in_range(1..=256)?;
72
73        for _ in 0..count {
74            if remaining == 0 {
75                break;
76            }
77            let size = u.int_in_range(1..=remaining)?;
78            remaining -= size;
79            actions.push(WriteAction::Write(size));
80        }
81
82        Ok(FuzzWriterInput {
83            actions,
84            buffer,
85            concurrent,
86        })
87    }
88}

90/// Write the generated actions through `op`'s writer, then read the object
91/// back and verify its content.
92pub async fn fuzz_writer(op: Operator, input: FuzzWriterInput) -> Result<()> {
93    let path = uuid::Uuid::new_v4().to_string();
94
95    let total_size = input
96        .actions
97        .iter()
98        .map(|a| match a {
99            WriteAction::Write(size) => *size,
100        })
101        .collect();
102
103    let checker = WriteChecker::new(total_size);
104
105    let mut writer = op.writer_with(&path);
106    if let Some(buffer) = input.buffer {
107        writer = writer.chunk(buffer);
108    } else if let Some(min_size) = op.info().capability().write_multi_min_size {
109        writer = writer.chunk(min_size);
110    }
111    if let Some(concurrent) = input.concurrent {
112        writer = writer.concurrent(concurrent);
113    }
114
115    let mut writer = writer.await?;
116
117    for chunk in checker.chunks() {
118        writer.write(chunk.clone()).await?;
119    }
120
121    writer.close().await?;
122
123    let result = op.read(&path).await?.to_bytes();
124
125    checker.check(&result);
126
127    op.delete(&path).await?;
128    Ok(())
129}

131#[cfg(test)]
132mod tests {
133    use super::*;
134    use opendal::tests::TEST_RUNTIME;
135
136    fn test_operator() -> Operator {
137        let root = std::env::temp_dir().join(format!("opendal-fuzz-test-{}", uuid::Uuid::new_v4()));
138        std::fs::create_dir_all(&root).expect("create test root dir must succeed");
139        Operator::new(opendal::services::Fs::default().root(&root.to_string_lossy()))
140            .expect("operator init must succeed")
141    }
142
143    fn parse(input: &[u8]) -> FuzzWriterInput {
144        FuzzWriterInput::arbitrary(&mut Unstructured::new(input)).expect("input must parse")
145    }
146
147    fn total_bytes(input: &FuzzWriterInput) -> usize {
148        input
149            .actions
150            .iter()
151            .map(|a| match a {
152                WriteAction::Write(size) => *size,
153            })
154            .sum()
155    }
156
157    /// Regression test for OSS-Fuzz issue 530185493 (apache/opendal#7848):
158    /// `buffer == 1` must cap the total written bytes at `MAX_CHUNKS`
159    /// instead of generating millions of one-byte writes.
160    #[test]
161    fn chunk_size_one_respects_chunk_budget() {
162        // buffer flag = 1, buffer = 1, concurrent flag = 0, count = 6,
163        // then sizes 101, 101, 53 which exhaust the 256-byte budget.
164        let input = parse(&[0x01, 0, 0, 0, 0x00, 0x05, 100, 100, 52]);
165        assert_eq!(input.buffer, Some(1));
166        assert!(total_bytes(&input) <= MAX_CHUNKS);
167
168        TEST_RUNTIME
169            .block_on(fuzz_writer(test_operator(), input))
170            .expect("fuzz writer must succeed");
171    }
172
173    /// A chunk-size-one input must not generate more writes than its byte
174    /// budget even when the requested action count is large.
175    #[test]
176    fn chunk_size_one_truncates_excess_actions() {
177        // buffer = 1, no concurrent, count = 256: the first action takes
178        // all 255 bytes it may, the second takes the last byte, and the
179        // remaining 254 requested actions are dropped.
180        let input = parse(&[0x01, 0, 0, 0, 0x00, 0xFF, 0xFE]);
181        assert_eq!(input.buffer, Some(1));
182        assert_eq!(input.actions.len(), 2);
183        assert!(total_bytes(&input) <= MAX_CHUNKS);
184    }
185
186    /// Without an explicit chunk size, actions are still bounded by
187    /// `MAX_TOTAL_BYTES` in total.
188    #[test]
189    fn total_bytes_respects_budget_without_chunk() {
190        // No buffer, no concurrent, count = 4, the first action takes
191        // 4 MiB - 1 and the second takes the last byte.
192        let input = parse(&[0x00, 0x00, 0x03, 0x3F, 0xFF, 0xFE, 0xFF]);
193        assert_eq!(input.buffer, None);
194        assert_eq!(input.actions.len(), 2);
195        assert_eq!(total_bytes(&input), MAX_TOTAL_BYTES);
196
197        TEST_RUNTIME
198            .block_on(fuzz_writer(test_operator(), input))
199            .expect("fuzz writer must succeed");
200    }
201
202    /// The shape of the 9-byte OSS-Fuzz reproducer: `buffer == 1` plus a
203    /// trailing short size section parses into a bounded input and still
204    /// completes the write -> close -> readback path.
205    #[test]
206    fn oss_fuzz_reproducer_shape_is_bounded() {
207        let input = parse(&[0x01, 0, 0, 0, 0x00, 0x05, 0, 0, 0]);
208        assert_eq!(input.buffer, Some(1));
209        assert!(total_bytes(&input) <= MAX_CHUNKS);
210
211        TEST_RUNTIME
212            .block_on(fuzz_writer(test_operator(), input))
213            .expect("fuzz writer must succeed");
214    }
215}
