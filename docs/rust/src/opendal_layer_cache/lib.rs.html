41/// The source path supplied to the read cache policy.
42#[non_exhaustive]
43#[derive(Debug, Clone, Copy)]
44pub struct CacheReadRequest<'a> {
45    path: &'a str,
46}
47
48impl<'a> CacheReadRequest<'a> {
49    /// Return the normalized source path.
50    pub fn path(&self) -> &'a str {
51        self.path
52    }
53}
54
55/// Select how the cache handles a reader.
56#[non_exhaustive]
57#[derive(Debug, Clone, Copy, PartialEq, Eq)]
58pub enum CacheDecision {
59    /// Bypass the cache completely for this reader.
60    Bypass,
61    /// Use the cache with the layer's default TTL.
62    Cache,
63    /// Use the cache with this metadata TTL.
64    /// Zero revalidates before serving. The maximum disables TTL revalidation.
65    CacheFor(Duration),
66}
67
68/// Decide by source path whether to cache reads and how long their metadata remains fresh.
69pub trait CachePolicy: Send + Sync + 'static {
70    /// Decide once for the reader, before any I/O.
71    fn on_read(&self, req: &CacheReadRequest<'_>) -> CacheDecision;
72}
73
74impl<F> CachePolicy for F
75where
76    F: Fn(&CacheReadRequest<'_>) -> CacheDecision + Send + Sync + 'static,
77{
78    fn on_read(&self, req: &CacheReadRequest<'_>) -> CacheDecision {
79        self(req)
80    }
81}

83const DEFAULT_CHUNK_SIZE: usize = 1024 * 1024;
84
85/// Cache aligned source read chunks in another OpenDAL operator.
86///
87/// The default chunk size is 1 MiB, capped by the cache operator's advertised
88/// maximum total write size. The default metadata TTL is 60 seconds.
89/// Use a cache backend with capacity eviction to reclaim chunks from old object
90/// identities. Use separate cache roots or instances for different tenants,
91/// endpoints, or credentials because fresh cache hits skip source I/O.
92///
93/// Add source retry layers before this layer so they retry individual source
94/// requests. If a cached stream detects that the source changed after yielding
95/// bytes, restart the whole read and discard the partial result.
96///
97/// Sources must expose a verifiable identity (an immutable `version` other than
98/// `"null"`, a strong `etag`, or `last_modified` + `content_length`) to be
99/// cached; reads without one pass through to the source. When a source exposes
100/// only `last_modified` and `content_length`, equal-length overwrites within
101/// the backend's timestamp resolution (such as 1-second mtime granularity) can
102/// keep the same identity and serve cached chunks across `ttl` boundaries.
103///
104/// Single-path mutations invalidate cached metadata for affected paths on
105/// completion. Reads explicitly selecting a non-immutable placeholder version
106/// such as `"null"` bypass the cache so local and external overwrites remain
107/// visible. For unversioned reads, external writes, recursive deletes,
108/// cancelled mutations, and cross-instance read/write races are bounded by
109/// `ttl` (subject to weak `last_modified` granularity above). Reads selecting
110/// an explicit immutable `version` skip `ttl` revalidation once cached until
111/// invalidated or evicted, even if the version is deleted externally. A reused `Reader` caches its
112/// resolved metadata pointer locally for up to `min(ttl, 1s)` across range
113/// reads, so completed cross-instance invalidations become visible to an
114/// already-reused `Reader` within `min(ttl, 1s)` and immediately to a newly
115/// opened `Reader` (or immediately when `ttl` is [`Duration::ZERO`]).
116#[derive(Clone)]
117pub struct CacheLayer {
118    state: Arc<State>,
119    policy: Arc<dyn CachePolicy>,
120    chunk_size: usize,
121    max_chunk_size: usize,
122    ttl: Duration,
123}
124
125impl fmt::Debug for CacheLayer {
126    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
127        f.debug_struct("CacheLayer")
128            .field("chunk_size", &self.chunk_size)
129            .field("ttl", &self.ttl)
130            .finish_non_exhaustive()
131    }
132}
133
134impl CacheLayer {
135    /// Create a layer backed by `cache`.
136    ///
137    /// Returns [`ErrorKind::ConfigInvalid`] if the cache operator's full
138    /// capabilities do not include read, write, and delete, or if its
139    /// advertised maximum total write size is zero.
140    pub fn new(cache: Operator) -> Result<Self> {
141        let capability = cache.info().capability();
142        if !(capability.read && capability.write && capability.delete) {
143            return Err(Error::new(
144                ErrorKind::ConfigInvalid,
145                "cache must support read, write, and delete",
146            ));
147        }
148        let max_chunk_size = capability.write_total_max_size.unwrap_or(usize::MAX);
149        if max_chunk_size == 0 {
150            return Err(Error::new(
151                ErrorKind::ConfigInvalid,
152                "cache maximum total write size must be positive",
153            ));
154        }
155        Ok(Self {
156            state: Arc::new(State::new(cache)),
157            policy: Arc::new(|_: &CacheReadRequest<'_>| CacheDecision::Cache),
158            chunk_size: DEFAULT_CHUNK_SIZE.min(max_chunk_size),
159            max_chunk_size,
160            ttl: Duration::from_secs(60),
161        })
162    }
163
164    /// Set the aligned chunk size in bytes.
165    ///
166    /// Zero restores the default of 1 MiB. Positive sizes are used without
167    /// rounding. The cache operator's advertised maximum total write size
168    /// caps both the default and explicit sizes.
169    pub fn with_chunk_size(mut self, size: usize) -> Self {
170        let size = if size == 0 { DEFAULT_CHUNK_SIZE } else { size };
171        self.chunk_size = size.min(self.max_chunk_size);
172        self
173    }
174
175    /// Set the default metadata freshness window.
176    ///
177    /// [`Duration::ZERO`] revalidates before serving a reader.
178    /// [`Duration::MAX`] disables periodic revalidation. Reads of an explicitly
179    /// selected immutable version skip periodic revalidation regardless of TTL.
180    pub fn with_ttl(mut self, ttl: Duration) -> Self {
181        self.ttl = ttl;
182        self
183    }
184
185    /// Set a custom per-read admission and TTL policy.
186    ///
187    /// The default policy returns [`CacheDecision::Cache`] for all reads.
188    pub fn with_policy(mut self, policy: impl CachePolicy) -> Self {
189        self.policy = Arc::new(policy);
190        self
191    }
192}
193
194impl Layer for CacheLayer {
195    fn apply_service(&self, inner: Servicer) -> Servicer {
196        Arc::new(CacheService {
197            inner,
198            layer: self.clone(),
199        })
200    }
