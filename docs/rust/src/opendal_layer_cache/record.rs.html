26#[derive(Clone, Debug)]
27pub(crate) struct Pointer {
28    pub(crate) metadata: Metadata,
29    pub(crate) identity: Identity,
30    pub(crate) observed_at: Timestamp,
31}
32
33// The record version is independent of the cache namespace. Keep the ordered
34// metadata fields and their encodings stable for caches shared by processes.
35const RECORD_VERSION: u8 = 1;
36
37// The MetadataBuilder compact field block is limited to u16::MAX bytes.
38// Allow room for the fixed pointer header, field lengths, and checksum framing.
39pub(crate) const MAX_RECORD_LEN: usize = u16::MAX as usize + 128;
40
41const STRINGS: [fn(&Metadata) -> Option<&str>; 8] = [
42    Metadata::cache_control,
43    Metadata::content_disposition,
44    Metadata::content_md5,
45    Metadata::content_type,
46    Metadata::content_encoding,
47    Metadata::content_language,
48    Metadata::etag,
49    Metadata::version,
50];
51
52impl Pointer {
53    pub(crate) fn from_metadata(metadata: Metadata, observed_at: Timestamp) -> Option<Self> {
54        let identity = Identity::from_metadata(&metadata)?;
55        Some(Self {
56            metadata,
57            identity,
58            observed_at,
59        })
60    }
61
62    pub(crate) fn is_fresh(&self, ttl: Duration, version: Option<&str>) -> bool {
63        ttl == Duration::MAX
64            || version.is_some_and(is_immutable_version)
65            || (ttl != Duration::ZERO
66                && SystemTime::from(Timestamp::now())
67                    .duration_since(self.observed_at.into())
68                    .unwrap_or_default()
69                    < ttl)
70    }
71
72    pub(crate) fn encode(&self) -> Buffer {
73        let mut record = Vec::new();
74        record.put_u8(RECORD_VERSION);
75        record.put_u8(self.identity.tier as u8);
76        record.extend_from_slice(&self.identity.token);
77        write_timestamp(&mut record, self.observed_at);
78        let metadata = &self.metadata;
79        record.put_u8(match metadata.mode() {
80            EntryMode::Unknown => 0,
81            EntryMode::FILE => 1,
82            EntryMode::DIR => 2,
83        });
84        record.put_u64(metadata.content_length());
85        record.put_u8(match metadata.is_current() {
86            None => 0,
87            Some(false) => 1,
88            Some(true) => 2,
89        });
90        record.put_u8(u8::from(metadata.is_deleted()));
91        record.put_u8(u8::from(metadata.last_modified().is_some()));
92        if let Some(time) = metadata.last_modified() {
93            write_timestamp(&mut record, time);
94        }
95        for field in STRINGS {
96            match field(metadata) {
97                Some(value) => write_string(&mut record, value),
98                None => record.put_u16(u16::MAX),
99            }
100        }
101        if let Some(values) = metadata.user_metadata() {
102            record.put_u16(values.len() as u16);
103            for (key, value) in values {
104                write_string(&mut record, key);
105                write_string(&mut record, value);
106            }
107        } else {
108            record.put_u16(u16::MAX);
109        }
110        if let Some(checksum) = metadata.checksum()
111            && let Some(tag) = checksum_algo_tag(checksum.algo)
112        {
113            record.put_u8(tag);
114            record.put_u16(checksum.value.len() as u16);
115            record.extend_from_slice(&checksum.value);
116        } else {
117            record.put_u8(0);
118        }
119        record.into()
120    }
121
122    pub(crate) fn decode(record: Buffer) -> Option<Self> {
123        if record.len() > MAX_RECORD_LEN {
124            return None;
125        }
126        let bytes = record.to_vec();
127        let mut input = bytes.as_slice();
128        if input.read_u8()? != RECORD_VERSION {
129            return None;
130        }
131        let tier = match input.read_u8()? {
132            1 => Tier::Version,
133            2 => Tier::Etag,
134            3 => Tier::Modified,
135            _ => return None,
136        };
137        let token = *input.read_array::<16>()?;
138        let identity = Identity { tier, token };
139        let observed_at = input.read_timestamp()?;
140        let mode = input.read_u8()?;
141        let length = u64::from_be_bytes(*input.read_array()?);
142        let mut metadata = match (mode, length) {
143            (0, 0) => MetadataBuilder::unknown(),
144            (1, length) => MetadataBuilder::file(length),
145            (2, 0) => MetadataBuilder::dir(),
146            _ => return None,
147        };
148        metadata.is_current(match input.read_u8()? {
149            0 => None,
150            1 => Some(false),
151            2 => Some(true),
152            _ => return None,
153        });
154        metadata.is_deleted(input.read_bool()?);
155        if input.read_bool()? {
156            metadata.last_modified(input.read_timestamp()?);
157        }
158        // CompactValues stores a bitmap and an end offset per present field.
159        let mut compact_len = 2;
160        let setters: [fn(&mut MetadataBuilder, String) -> &mut MetadataBuilder; STRINGS.len()] = [
161            MetadataBuilder::cache_control,
162            MetadataBuilder::content_disposition,
163            MetadataBuilder::content_md5,
164            MetadataBuilder::content_type,
165            MetadataBuilder::content_encoding,
166            MetadataBuilder::content_language,
167            MetadataBuilder::etag,
168            MetadataBuilder::version,
169        ];
170        for set in setters {
171            let len = input.read_u16()?;
172            if len != u16::MAX {
173                compact_len += 2 + usize::from(len);
174                set(&mut metadata, input.read_string(len)?.to_owned());
175            }
176        }
177        let count = input.read_u16()?;
178        if count != u16::MAX {
179            compact_len += 4;
180            let mut values = Vec::new();
181            for _ in 0..count {
182                let key_len = input.read_u16()?;
183                let key = input.read_string(key_len)?;
184                let value_len = input.read_u16()?;
185                let value = input.read_string(value_len)?;
186                compact_len += 4 + usize::from(key_len) + usize::from(value_len);
187                if values
188                    .last()
189                    .is_some_and(|(previous, _): &(String, String)| previous.as_str() >= key)
190                {
191                    return None;
192                }
193                values.push((key.to_owned(), value.to_owned()));
194            }
195            metadata.user_metadata(values);
196        }
197        let checksum_tag = input.read_u8()?;
198        if checksum_tag != 0 {
199            let algo = checksum_algo_from_tag(checksum_tag)?;
200            let len = input.read_u16()?;
201            if usize::from(len) != algo.digest_len() {
202                return None;
203            }
204            let value = input.read_bytes(len)?;
205            // CompactValues stores an end offset plus 1-byte algorithm tag and digest.
206            compact_len += 2 + 1 + usize::from(len);
207            metadata.checksum(Checksum::new(algo, value));
208        }
209        if !input.is_empty() || compact_len > u16::MAX as usize {
210            return None;
211        }
212        let metadata = metadata.build();
213        if Identity::from_metadata(&metadata) != Some(identity) {
214            return None;
215        }
216        Some(Self {
217            metadata,
218            identity,
219            observed_at,
220        })
221    }
222}
223
224fn checksum_algo_tag(algo: ChecksumAlgo) -> Option<u8> {
225    match algo {
226        ChecksumAlgo::Crc64Nvme => Some(1),
227        ChecksumAlgo::Crc32c => Some(2),
228        ChecksumAlgo::Md5 => Some(3),
229        ChecksumAlgo::Sha256 => Some(4),
230        ChecksumAlgo::Crc64Ecma => Some(5),
231        _ => None,
232    }
233}
234
235fn checksum_algo_from_tag(tag: u8) -> Option<ChecksumAlgo> {
236    match tag {
237        1 => Some(ChecksumAlgo::Crc64Nvme),
238        2 => Some(ChecksumAlgo::Crc32c),
239        3 => Some(ChecksumAlgo::Md5),
240        4 => Some(ChecksumAlgo::Sha256),
241        5 => Some(ChecksumAlgo::Crc64Ecma),
242        _ => None,
243    }
244}
245
246fn write_timestamp(output: &mut Vec<u8>, time: Timestamp) {
247    let time = time.into_inner();
248    output.put_i64(time.as_second());
249    output.put_i32(time.subsec_nanosecond());
250}
251
252fn write_string(output: &mut Vec<u8>, value: &str) {
253    output.put_u16(value.len() as u16);
254    output.extend_from_slice(value.as_bytes());
255}

257trait RecordInput<'a> {
258    fn read_array<const N: usize>(&mut self) -> Option<&'a [u8; N]>;
259    fn read_u8(&mut self) -> Option<u8>;
260    fn read_bool(&mut self) -> Option<bool>;
261    fn read_u16(&mut self) -> Option<u16>;
262    fn read_timestamp(&mut self) -> Option<Timestamp>;
263    fn read_bytes(&mut self, len: u16) -> Option<&'a [u8]>;
264    fn read_string(&mut self, len: u16) -> Option<&'a str>;
265}
266
267impl<'a> RecordInput<'a> for &'a [u8] {
268    fn read_array<const N: usize>(&mut self) -> Option<&'a [u8; N]> {
269        let (value, rest) = self.split_first_chunk()?;
270        *self = rest;
271        Some(value)
272    }
273
274    fn read_u8(&mut self) -> Option<u8> {
275        Some(self.read_array::<1>()?[0])
276    }
277
278    fn read_bool(&mut self) -> Option<bool> {
279        match self.read_u8()? {
280            0 => Some(false),
281            1 => Some(true),
282            _ => None,
283        }
284    }
285
286    fn read_u16(&mut self) -> Option<u16> {
287        Some(u16::from_be_bytes(*self.read_array()?))
288    }
289
290    fn read_timestamp(&mut self) -> Option<Timestamp> {
291        let seconds = i64::from_be_bytes(*self.read_array()?);
292        let nanos = i32::from_be_bytes(*self.read_array()?);
293        Timestamp::new(seconds, nanos).ok()
294    }
295
296    fn read_bytes(&mut self, len: u16) -> Option<&'a [u8]> {
297        let (value, rest) = self.split_at_checked(usize::from(len))?;
298        *self = rest;
299        Some(value)
300    }
301
302    fn read_string(&mut self, len: u16) -> Option<&'a str> {
303        std::str::from_utf8(self.read_bytes(len)?).ok()
304    }
305}

311    #[test]
312    fn pointer_preserves_source_metadata_including_checksum_and_rejects_truncated_records() {
313        let mut metadata = MetadataBuilder::file(42);
314        metadata
315            .version("v1")
316            .etag("\"abc\"")
317            .content_encoding("gzip")
318            .content_language("en")
319            .last_modified(Timestamp::new(-42, -123).unwrap())
320            .is_current(Some(false))
321            .user_metadata([("key".to_owned(), "value".to_owned())])
322            .checksum(Checksum::new(
323                ChecksumAlgo::Crc32c,
324                [0xde, 0xad, 0xbe, 0xef],
325            ));
326        let pointer =
327            Pointer::from_metadata(metadata.build(), Timestamp::new(123, 123).unwrap()).unwrap();
328        let encoded = pointer.encode();
329        let decoded = Pointer::decode(encoded.clone()).unwrap();
330        assert_eq!(decoded.metadata, pointer.metadata);
331        assert_eq!(decoded.metadata.checksum(), pointer.metadata.checksum());
332        assert_eq!(decoded.identity, pointer.identity);
333        assert_eq!(decoded.observed_at, pointer.observed_at);
334        for len in 0..encoded.len() {
335            assert!(Pointer::decode(encoded.slice(..len)).is_none());
336        }
337        // Reject corrupted checksum digest length.
338        let mut bad_checksum = encoded.to_vec();
339        let last = bad_checksum.len();
340        bad_checksum[last - 5] = 3; // wrong digest len for Crc32c
341        bad_checksum.pop();
342        assert!(Pointer::decode(Buffer::from(bad_checksum)).is_none());
343    }

345    #[test]
346    fn pointer_preserves_crc64_ecma_checksum() {
347        let checksum = Checksum::new(
348            ChecksumAlgo::Crc64Ecma,
349            0x995d_c9bb_df19_39fa_u64.to_be_bytes(),
350        );
351        let mut metadata = MetadataBuilder::file(9);
352        metadata.etag("\"abc\"").checksum(checksum.clone());
353        let pointer =
354            Pointer::from_metadata(metadata.build(), Timestamp::new(100, 200).unwrap()).unwrap();
355        let decoded = Pointer::decode(pointer.encode()).unwrap();
356        assert_eq!(decoded.metadata.checksum(), Some(checksum));
357    }

359    #[test]
360    fn pointer_v1_wire_format_is_stable_and_accepts_max_metadata() {
361        let mut metadata = MetadataBuilder::file(10);
362        metadata
363            .etag("\"abc\"")
364            .checksum(Checksum::new(ChecksumAlgo::Crc64Nvme, [7; 8]));
365        let pointer =
366            Pointer::from_metadata(metadata.build(), Timestamp::new(100, 200).unwrap()).unwrap();
367        let encoded = pointer.encode().to_vec();
368        let expected: Vec<u8> = [
369            &[1u8, 2u8][..], // RECORD_VERSION=1, Tier::Etag=2
370            &[
371                0xc2, 0x76, 0x01, 0xd6, 0x28, 0x12, 0x2b, 0x57, 0x7c, 0xa2, 0xdb, 0x46, 0x2e, 0x27,
372                0xd2, 0xf4,
373            ], // sha256("etag", "\"abc\"")[..16]
374            &100i64.to_be_bytes(),
375            &200i32.to_be_bytes(),
376            &[1u8], // EntryMode::FILE
377            &10u64.to_be_bytes(),
378            &[0u8, 0u8, 0u8], // is_current=None, is_deleted=false, has_last_modified=false
379            &[0xff, 0xff].repeat(6), // first 6 string fields absent
380            &5u16.to_be_bytes(),
381            b"\"abc\"",
382            &[0xff, 0xff], // version absent
383            &[0xff, 0xff], // user_metadata absent
384            &[1u8],        // ChecksumAlgo::Crc64Nvme
385            &8u16.to_be_bytes(),
386            &[7u8; 8],
387        ]
388        .concat();
389        assert_eq!(encoded, expected);
390        assert_eq!(
391            Pointer::decode(Buffer::from(expected)).unwrap().metadata,
392            pointer.metadata
393        );
394
395        // Max-size Metadata (u16::MAX compact values + fixed header) still round-trips.
396        let mut large = MetadataBuilder::file(1);
397        large
398            .etag("\"e\"")
399            .last_modified(Timestamp::new(1, 0).unwrap())
400            .content_type("a".repeat(u16::MAX as usize - 9));
401        let large_ptr = Pointer::from_metadata(large.build(), Timestamp::new(1, 0).unwrap())
402            .expect("valid large pointer");
403        let large_encoded = large_ptr.encode();
404        assert!(large_encoded.len() > u16::MAX as usize + 64);
405        assert!(large_encoded.len() <= MAX_RECORD_LEN);
406        assert_eq!(
407            Pointer::decode(large_encoded).unwrap().metadata,
408            large_ptr.metadata
409        );
410    }

412    #[test]
413    fn null_version_still_respects_ttl_revalidation() {
414        let mut meta = MetadataBuilder::file(4);
415        meta.etag("\"e1\"").version("null");
416        let pointer = Pointer::from_metadata(meta.build(), Timestamp::new(1, 0).unwrap()).unwrap();
417
418        assert!(!pointer.is_fresh(Duration::ZERO, Some("null")));
419        assert!(!pointer.is_fresh(Duration::ZERO, Some("")));
420        assert!(!pointer.is_fresh(Duration::from_secs(60), Some("null")));
421        assert!(pointer.is_fresh(Duration::ZERO, Some("v1")));
422        assert!(pointer.is_fresh(Duration::MAX, Some("null")));
423    }
