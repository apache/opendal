39/// `FastraceLayer` traces OpenDAL operations with
40/// [fastrace](https://docs.rs/fastrace/).
41///
42/// It creates spans for service calls and deferred operation bodies such as
43/// readers, writers, listers, deleters, copiers, and composers. It also traces outbound
44/// HTTP requests through response body consumption and propagates trace context
45/// to tasks spawned through the operation's executor.
46///
47/// # Examples
48///
49/// ## Basic Setup
50///
51/// ```no_run
52/// # use opendal_core::services;
53/// # use opendal_core::Operator;
54/// # use opendal_core::Result;
55/// # use opendal_layer_fastrace::FastraceLayer;
56/// #
57/// # fn main() -> Result<()> {
58/// let _ = Operator::new(services::Memory::default())?
59///     .layer(FastraceLayer::new());
60/// # Ok(())
61/// # }
62/// ```
63///
64/// ## Real usage
65///
66/// ```no_run
67/// # use anyhow::Result;
68/// # use fastrace::prelude::*;
69/// # use opendal_core::services;
70/// # use opendal_core::Operator;
71/// # use opendal_layer_fastrace::FastraceLayer;
72/// #
73/// # fn main() -> Result<()> {
74/// let reporter = fastrace_jaeger::JaegerReporter::new("127.0.0.1:6831".parse()?, "opendal").unwrap();
75/// fastrace::set_reporter(reporter, fastrace::collector::Config::default());
76///
77/// {
78///     let root = Span::root("op", SpanContext::random());
79///     let runtime = tokio::runtime::Runtime::new()?;
80///     runtime.block_on(
81///         async {
82///             let _ = dotenvy::dotenv();
83///             let op = Operator::new(services::Memory::default())?
84///                 .layer(FastraceLayer::new());
85///             op.write("test", "0".repeat(16 * 1024 * 1024).into_bytes())
86///                 .await?;
87///             op.stat("test").await?;
88///             op.read("test").await?;
89///             Ok::<(), opendal_core::Error>(())
90///         }
91///         .in_span(Span::enter_with_parent("test", &root)),
92///     )?;
93/// }
94///
95/// fastrace::flush();
96/// # Ok(())
97/// # }
98/// ```
99///
100/// # Output
101///
102/// OpenDAL is using [`fastrace`](https://docs.rs/fastrace/latest/fastrace/) for tracing internally.
103///
104/// To enable fastrace output, initialize a reporter supported by `fastrace`.
105///
106/// For example:
107///
108/// ```no_run
109/// # use anyhow::Result;
110/// #
111/// # fn main() -> Result<()> {
112/// let reporter = fastrace_jaeger::JaegerReporter::new("127.0.0.1:6831".parse()?, "opendal").unwrap();
113/// fastrace::set_reporter(reporter, fastrace::collector::Config::default());
114/// # Ok(())
115/// # }
116/// ```
117///
118/// For real-world usage, take a look at [`fastrace-datadog`](https://crates.io/crates/fastrace-datadog) or [`fastrace-jaeger`](https://crates.io/crates/fastrace-jaeger).
119#[derive(Clone, Debug, Default)]
120#[non_exhaustive]
121pub struct FastraceLayer {}

130impl Layer for FastraceLayer {
131    fn apply_service(&self, inner: Servicer) -> Servicer {
132        Arc::new(FastraceService { inner })
133    }
134
135    fn apply_context(&self, _service: Servicer, inner: OperationContext) -> OperationContext {
136        let (transport, executor) = inner.into_parts();
137        OperationContext::from_parts(
138            HttpTransporter::new(FastraceHttpTransport { inner: transport }),
139            Executor::with(FastraceExecutor {
140                inner: executor.into_inner(),
141            }),
142        )
143    }
144}

146struct FastraceHttpTransport {
147    inner: HttpTransporter,
148}
149
150impl HttpTransport for FastraceHttpTransport {
151    async fn fetch(&self, req: http::Request<Buffer>) -> Result<http::Response<HttpBody>> {
152        let span = Span::enter_with_local_parent("http::fetch");
153        let mut fetch = pin!(self.inner.fetch(req));
154        // Enter the span for each poll so the response body can take ownership of it.
155        let response = poll_fn(|cx| {
156            let _guard = span.set_local_parent();
157            fetch.as_mut().poll(cx)
158        })
159        .await?;
160
161        let (parts, body) = response.into_parts();
162        let body = body.map_inner(|inner| {
163            Box::new(FastraceHttpStream {
164                inner,
165                span: Some(span),
166            })
167        });
168        Ok(http::Response::from_parts(parts, body))
169    }
170}

172struct FastraceHttpStream<S> {
173    inner: S,
174    span: Option<Span>,
175}
176
177impl<S: Stream<Item = Result<Buffer>> + Unpin> Stream for FastraceHttpStream<S> {
178    type Item = Result<Buffer>;
179
180    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
181        let result = {
182            let _guard = self.span.as_ref().map(Span::set_local_parent);
183            Pin::new(&mut self.inner).poll_next(cx)
184        };
185        if matches!(result, Poll::Ready(None | Some(Err(_)))) {
186            self.span.take();
187        }
188        result
189    }
190}

192struct FastraceExecutor {
193    inner: Arc<dyn Execute>,
194}
195
196impl Execute for FastraceExecutor {
197    fn execute(&self, f: BoxedStaticFuture<()>) {
198        let span = Span::enter_with_local_parent("executor::execute");
199        self.inner.execute(Box::pin(f.in_span(span)));
200    }
201
202    fn try_execute(&self, f: BoxedStaticFuture<()>) -> Result<()> {
203        let span = Span::enter_with_local_parent("executor::execute");
204        self.inner.try_execute(Box::pin(f.in_span(span)))
205    }
206
207    fn timeout(&self) -> Option<BoxedStaticFuture<()>> {
208        self.inner.timeout()
209    }
210
211    fn sleep(&self, duration: Duration) -> Option<BoxedStaticFuture<()>> {
212        self.inner.sleep(duration)
213    }
214}

350// Keep the operation span with the returned body so later body methods can
351// attach child spans without relying on local span state.
352struct FastraceWrapper<R> {
353    span: Arc<Span>,
354    inner: R,
355}
356
357impl<R> FastraceWrapper<R> {
358    fn new(span: Span, inner: R) -> Self {
359        Self {
360            span: Arc::new(span),
361            inner,
362        }
363    }
364
365    fn with_span(span: Arc<Span>, inner: R) -> Self {
366        Self { span, inner }
367    }
368}

370impl<R: oio::ReadStream> oio::ReadStream for FastraceWrapper<R> {
371    fn read(&mut self) -> impl Future<Output = Result<Buffer>> + MaybeSend {
372        let span = Span::enter_with_parent(Operation::Read.into_static(), &self.span);
373        let _guard = span.set_local_parent();
374        self.inner.read().in_span(span)
375    }
376}

378impl<R: oio::Read> oio::Read for FastraceWrapper<R> {
379    fn open(
380        &self,
381        range: BytesRange,
382    ) -> impl Future<Output = Result<(RpRead, Box<dyn oio::ReadStreamDyn>)>> + MaybeSend {
383        let span = Span::enter_with_parent(Operation::Read.into_static(), &self.span);
384        let _guard = span.set_local_parent();
385        let reader_span = self.span.clone();
386        let open = self.inner.open(range).in_span(span);
387        async move {
388            let (rp, stream) = open.await?;
389            Ok((
390                rp,
391                Box::new(FastraceWrapper::with_span(reader_span, stream))
392                    as Box<dyn oio::ReadStreamDyn>,
393            ))
394        }
395    }
396
397    fn read(
398        &self,
399        range: BytesRange,
400    ) -> impl Future<Output = Result<(RpRead, Buffer)>> + MaybeSend {
401        let span = Span::enter_with_parent(Operation::Read.into_static(), &self.span);
402        let _guard = span.set_local_parent();
403        self.inner.read(range).in_span(span)
404    }
405}
