31/// Builder for TailCutLayer.
32///
33/// Use this to configure the layer, then call `build()` to create a layer
34/// that can be cloned and shared across multiple operators.
35/// Set the deadline bounds in either order. The minimum deadline must be less
36/// than or equal to the maximum deadline when calling [`Self::build`].
37///
38/// # Examples
39///
40/// ```no_run
41/// # use std::time::Duration;
42/// #
43/// # use opendal_core::services;
44/// # use opendal_core::Operator;
45/// # use opendal_core::Result;
46/// # use opendal_layer_tail_cut::TailCutLayer;
47/// #
48/// # fn main() -> Result<()> {
49/// let layer = TailCutLayer::builder()
50///     .percentile(95)
51///     .window(Duration::from_secs(60))
52///     .build();
53///
54/// let op = Operator::new(services::Memory::default())?
55///     .layer(layer);
56/// # Ok(())
57/// # }
58/// ```
59#[derive(Clone)]
60pub struct TailCutLayerBuilder {
61    percentile: u8,
62    safety_factor: f64,
63    window: Duration,
64    min_samples: usize,
65    min_deadline: Duration,
66    max_deadline: Duration,
67}
68
69impl Default for TailCutLayerBuilder {
70    fn default() -> Self {
71        Self {
72            percentile: 95,
73            safety_factor: 1.3,
74            window: Duration::from_secs(60),
75            min_samples: 200,
76            min_deadline: Duration::from_millis(500),
77            max_deadline: Duration::from_secs(30),
78        }
79    }
80}

82impl TailCutLayerBuilder {
83    /// Create a new [`TailCutLayerBuilder`] with default settings.
84    pub fn new() -> Self {
85        Self::default()
86    }
87
88    /// Set the percentile threshold (e.g., 95 for P95, 99 for P99).
89    ///
90    /// Requests slower than this percentile × safety_factor will be cancelled.
91    ///
92    /// Default: 95
93    ///
94    /// # Panics
95    ///
96    /// Panics if percentile is not between 50 and 99.
97    pub fn percentile(mut self, percentile: u8) -> Self {
98        assert!(
99            (50..=99).contains(&percentile),
100            "percentile must be between 50 and 99"
101        );
102        self.percentile = percentile;
103        self
104    }
105
106    /// Set the safety factor multiplier.
107    ///
108    /// The actual deadline is calculated as: P{percentile} × safety_factor.
109    /// A higher value reduces false positives but may miss some long tails.
110    ///
111    /// Default: 1.3 (30% buffer)
112    ///
113    /// # Panics
114    ///
115    /// Panics if factor is not between 1.0 and 5.0.
116    pub fn safety_factor(mut self, factor: f64) -> Self {
117        assert!(
118            (1.0..=5.0).contains(&factor),
119            "safety_factor must be between 1.0 and 5.0"
120        );
121        self.safety_factor = factor;
122        self
123    }
124
125    /// Set the sliding window duration for statistics collection.
126    ///
127    /// Longer windows provide more stable statistics but react slower to changes.
128    /// Shorter windows adapt faster but may be more noisy.
129    ///
130    /// Default: 60 seconds
131    ///
132    /// # Panics
133    ///
134    /// Panics if window is greater than 120 seconds.
135    pub fn window(mut self, window: Duration) -> Self {
136        assert!(
137            window <= Duration::from_millis(MAX_WINDOW_MS),
138            "window must be <= {} seconds",
139            MAX_WINDOW_MS / 1000
140        );
141        self.window = window;
142        self
143    }
144
145    /// Set the minimum number of samples required before enabling adaptive cancellation.
146    ///
147    /// During cold start (when sample count < min_samples), the layer will not
148    /// cancel any requests to avoid false positives.
149    ///
150    /// Default: 200
151    pub fn min_samples(mut self, min_samples: usize) -> Self {
152        self.min_samples = min_samples;
153        self
154    }
155
156    /// Set the minimum deadline (floor).
157    ///
158    /// Even if calculated deadline is shorter, it will be clamped to this value.
159    /// This prevents overly aggressive cancellation on very fast backends.
160    ///
161    /// Default: 500ms
162    pub fn min_deadline(mut self, deadline: Duration) -> Self {
163        self.min_deadline = deadline;
164        self
165    }
166
167    /// Set the maximum deadline (ceiling).
168    ///
169    /// Even if calculated deadline is longer, it will be clamped to this value.
170    /// This acts as a safety fallback timeout.
171    ///
172    /// Default: 30s
173    pub fn max_deadline(mut self, deadline: Duration) -> Self {
174        self.max_deadline = deadline;
175        self
176    }
177
178    /// Build the layer.
179    ///
180    /// The returned layer can be cloned to share statistics across operators.
181    ///
182    /// # Panics
183    ///
184    /// Panics if `min_deadline` is greater than `max_deadline`.
185    ///
186    /// # Examples
187    ///
188    /// ```no_run
189    /// # use std::time::Duration;
190    /// #
191    /// # use opendal_core::services;
192    /// # use opendal_core::Operator;
193    /// # use opendal_core::Result;
194    /// # use opendal_layer_tail_cut::TailCutLayer;
195    /// #
196    /// # fn main() -> Result<()> {
197    /// let layer = TailCutLayer::builder()
198    ///     .percentile(95)
199    ///     .window(Duration::from_secs(60))
200    ///     .build();
201    ///
202    /// // Share the layer across operators
203    /// let op1 = Operator::new(services::Memory::default())?
204    ///     .layer(layer.clone());
205    ///
206    /// let op2 = Operator::new(services::Memory::default())?
207    ///     .layer(layer.clone());
208    /// // op1 and op2 share the same statistics
209    /// # Ok(())
210    /// # }
211    /// ```
212    pub fn build(self) -> TailCutLayer {
213        assert!(
214            self.min_deadline <= self.max_deadline,
215            "min_deadline ({:?}) must be <= max_deadline ({:?})",
216            self.min_deadline,
217            self.max_deadline
218        );
219
220        TailCutLayer {
221            config: Arc::new(TailCutConfig {
222                percentile: self.percentile,
223                safety_factor: self.safety_factor,
224                window: self.window,
225                min_samples: self.min_samples,
226                min_deadline: self.min_deadline,
227                max_deadline: self.max_deadline,
228            }),
229            stats: Arc::new(TailCutStats::new()),
230        }
231    }
232}

245/// `TailCutLayer` monitors request latency and cancels requests that run
246/// significantly slower than the historical baseline (for example, slower than
247/// P95).
248///
249/// Active deadlines use the operation context's executor timer. The layer
250/// falls back to Tokio timers when [`Execute::sleep`] returns `None`.
251///
252/// Create this layer with [`TailCutLayer::builder()`]. Clone it to share
253/// statistics across multiple operators.
254///
255/// # Examples
256///
257/// ```no_run
258/// # use std::time::Duration;
259/// #
260/// # use opendal_core::services;
261/// # use opendal_core::Operator;
262/// # use opendal_core::Result;
263/// # use opendal_layer_tail_cut::TailCutLayer;
264/// #
265/// # fn main() -> Result<()> {
266/// let layer = TailCutLayer::builder()
267///     .percentile(95)
268///     .safety_factor(1.3)
269///     .window(Duration::from_secs(60))
270///     .build();
271///
272/// let op = Operator::new(services::Memory::default())?
273///     .layer(layer);
274/// # Ok(())
275/// # }
276/// ```
277#[derive(Clone)]
278pub struct TailCutLayer {
279    config: Arc<TailCutConfig>,
280    stats: Arc<TailCutStats>,
281}

336impl TailCutService {
337    /// Calculate the deadline for a given operation and size.
338    fn calculate_deadline(&self, op: Operation, size: Option<u64>) -> Option<Duration> {
339        let op_stats = self.stats.stats_for(op);
340
341        if op_stats.total_samples(size, self.config.window) < self.config.min_samples {
342            return None;
343        }
344
345        let q = self.config.percentile as f64 / 100.0;
346        let pctl = op_stats.quantile(size, q, self.config.window)?;
347
348        let deadline = Duration::from_secs_f64(pctl.as_secs_f64() * self.config.safety_factor);
349        Some(deadline.clamp(self.config.min_deadline, self.config.max_deadline))
350    }
351
352    async fn with_deadline<F, T>(
353        &self,
354        ctx: &OperationContext,
355        op: Operation,
356        size: Option<u64>,
357        fut: F,
358    ) -> Result<T>
359    where
360        F: Future<Output = Result<T>>,
361    {
362        let start = Instant::now();
363
364        let result = if let Some(deadline) = self.calculate_deadline(op, size) {
365            match run_with_deadline(ctx.executor().inner(), deadline, fut).await {
366                Some(res) => res,
367                None => Err(Error::new(ErrorKind::Unexpected, "cancelled by tail cut")
368                    .with_operation(op)
369                    .with_context("percentile", format!("P{}", self.config.percentile))
370                    .with_context("deadline", format!("{:?}", deadline))
371                    .set_temporary()),
372            }
373        } else {
374            fut.await
375        };
376
377        if result.is_ok() {
378            let latency = start.elapsed();
379            self.stats.stats_for(op).record(size, latency);
380        }
381
382        result
383    }
384}

541/// Body wrapper that applies tail-cut deadlines to streaming I/O calls.
542struct TailCutWrapper<R> {
543    inner: R,
544    config: Arc<TailCutConfig>,
545    stats: Arc<TailCutStats>,
546    executor: Executor,
547}
548
549impl<R> TailCutWrapper<R> {
550    fn new(
551        inner: R,
552        config: Arc<TailCutConfig>,
553        stats: Arc<TailCutStats>,
554        executor: Executor,
555    ) -> Self {
556        Self {
557            inner,
558            config,
559            stats,
560            executor,
561        }
562    }
563
564    fn calculate_deadline(&self, key: impl Into<StatKey>) -> Option<Duration> {
565        self.calculate_deadline_for(key, None)
566    }
567
568    fn calculate_deadline_for(
569        &self,
570        key: impl Into<StatKey>,
571        size: Option<u64>,
572    ) -> Option<Duration> {
573        let op_stats = self.stats.stats_for(key);
574
575        if op_stats.total_samples(size, self.config.window) < self.config.min_samples {
576            return None;
577        }
578
579        let q = self.config.percentile as f64 / 100.0;
580        let pctl = op_stats.quantile(size, q, self.config.window)?;
581
582        let deadline = Duration::from_secs_f64(pctl.as_secs_f64() * self.config.safety_factor);
583        Some(deadline.clamp(self.config.min_deadline, self.config.max_deadline))
584    }
585
586    #[inline]
587    async fn with_io_deadline<F, T>(
588        executor: &Executor,
589        deadline: Option<Duration>,
590        percentile: u8,
591        stats: &Arc<TailCutStats>,
592        size: Option<u64>,
593        key: impl Into<StatKey>,
594        fut: F,
595    ) -> Result<T>
596    where
597        F: std::future::Future<Output = Result<T>>,
598    {
599        let key = key.into();
600        let start = Instant::now();
601
602        let result = if let Some(dl) = deadline {
603            match run_with_deadline(executor.inner(), dl, fut).await {
604                Some(res) => res,
605                None => Err(
606                    Error::new(ErrorKind::Unexpected, "io cancelled by tail cut")
607                        .with_operation(key.operation())
608                        .with_context("percentile", format!("P{}", percentile))
609                        .with_context("deadline", format!("{:?}", dl))
610                        .set_temporary(),
611                ),
612            }
613        } else {
614            fut.await
615        };
616
617        if result.is_ok() {
618            let latency = start.elapsed();
619            stats.stats_for(key).record(size, latency);
620        }
621
622        result
623    }
624
625    /// Like [`Self::with_io_deadline`], but only records a latency sample when
626    /// the inner future ever returns `Poll::Pending`.
627    ///
628    /// Operations like `Delete::delete` and `Compose::compose` are synchronous
629    /// buffer pushes on batching backends (`BatchDeleter`, `GcsComposer`), and
630    /// `List::next`, `Write::write`, and `ReadStream::read` complete
631    /// synchronously when they serve buffered pages, buffered chunks, or
632    /// in-memory payloads. Sampling those instant completions would flood the
633    /// lowest histogram bucket, pass `min_samples` before any real I/O ran,
634    /// and collapse the deadline toward `min_deadline`. The deadline itself is
635    /// still enforced so real async work (page fetches, mid-batch flushes,
636    /// network reads, and intermediate requests) cannot hang.
637    #[inline]
638    async fn with_buffered_io_deadline<F, T>(
639        executor: &Executor,
640        deadline: Option<Duration>,
641        percentile: u8,
642        stats: &Arc<TailCutStats>,
643        size: Option<u64>,
644        key: impl Into<StatKey>,
645        fut: F,
646    ) -> Result<T>
647    where
648        F: std::future::Future<Output = Result<T>>,
649    {
650        let key = key.into();
651        let start = Instant::now();
652        let tracked = YieldTrackingFuture::new(fut);
653
654        let (result, yielded) = if let Some(dl) = deadline {
655            match run_with_deadline(executor.inner(), dl, tracked).await {
656                Some((res, yielded)) => (res, yielded),
657                None => (
658                    Err(
659                        Error::new(ErrorKind::Unexpected, "io cancelled by tail cut")
660                            .with_operation(key.operation())
661                            .with_context("percentile", format!("P{}", percentile))
662                            .with_context("deadline", format!("{:?}", dl))
663                            .set_temporary(),
664                    ),
665                    true,
666                ),
667            }
668        } else {
669            let (res, yielded) = tracked.await;
670            (res, yielded)
671        };
672
673        if yielded && result.is_ok() {
674            let latency = start.elapsed();
675            stats.stats_for(key).record(size, latency);
676        }
677
678        result
679    }
680}

682async fn run_with_deadline<F: Future>(
683    executor: &dyn Execute,
684    duration: Duration,
685    future: F,
686) -> Option<F::Output> {
687    let Some(mut timer) = executor.sleep(duration) else {
688        return tokio::time::timeout(duration, future).await.ok();
689    };
690
691    let mut future = std::pin::pin!(future);
692    std::future::poll_fn(|cx| match future.as_mut().poll(cx) {
693        Poll::Ready(output) => Poll::Ready(Some(output)),
694        Poll::Pending => timer.as_mut().poll(cx).map(|()| None),
695    })
696    .await
697}

699/// Future wrapper that reports whether the inner future ever returned
700/// `Poll::Pending`, so buffered operations only sample real async work.
701struct YieldTrackingFuture<F> {
702    inner: Pin<Box<F>>,
703    yielded: bool,
704}
705
706impl<F> YieldTrackingFuture<F> {
707    fn new(inner: F) -> Self {
708        Self {
709            inner: Box::pin(inner),
710            yielded: false,
711        }
712    }
713}
714
715impl<F: Future> Future for YieldTrackingFuture<F> {
716    type Output = (F::Output, bool);
717
718    fn poll(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
719        match self.inner.as_mut().poll(cx) {
720            Poll::Ready(output) => Poll::Ready((output, self.yielded)),
721            Poll::Pending => {
722                self.yielded = true;
723                Poll::Pending
724            }
725        }
726    }
727}

1095const SLICE_DURATION_MS: u64 = 10_000; // 10 seconds per slice
1096const MAX_WINDOW_MS: u64 = 120_000; // 120 seconds, the largest supported window
1097// One spare slice ensures a slice is only recycled after it has fallen
1098// outside `window_ms + SLICE_DURATION_MS` for the largest supported window.
1099const NUM_SLICES: usize = (MAX_WINDOW_MS / SLICE_DURATION_MS) as usize + 1;
1100const NUM_BUCKETS: usize = 17; // 17 buckets covering 1ms to 64s
1101
1102/// Windowed histogram using lock-free atomic operations.
1103struct WindowedHistogram {
1104    slices: Box<[TimeSlice; NUM_SLICES]>,
1105    current_idx: AtomicUsize,
1106    last_rotate: AtomicU64,
1107    /// Monotonic anchor for slice timestamps. Slices store elapsed
1108    /// milliseconds since this instant, so clock steps cannot move them
1109    /// backwards.
1110    start: Instant,
1111}
1112
1113impl WindowedHistogram {
1114    fn new() -> Self {
1115        let start = Instant::now();
1116        let now = Self::now_ms_from(start);
1117        Self {
1118            slices: Box::new(std::array::from_fn(|index| {
1119                let start_epoch_ms = if index == 0 { now } else { 0 };
1120                TimeSlice::new(start_epoch_ms)
1121            })),
1122            current_idx: AtomicUsize::new(0),
1123            last_rotate: AtomicU64::new(now),
1124            start,
1125        }
1126    }
1127
1128    /// Milliseconds elapsed since this histogram was created.
1129    ///
1130    /// The value starts at 1 so that `start_epoch_ms == 0` can exclusively
1131    /// mark a slice that is hidden from snapshots.
1132    fn now_ms(&self) -> u64 {
1133        Self::now_ms_from(self.start)
1134    }
1135
1136    fn now_ms_from(start: Instant) -> u64 {
1137        let elapsed = u64::try_from(start.elapsed().as_millis()).unwrap_or(u64::MAX);
1138        elapsed.saturating_add(1)
1139    }
1140
1141    fn record(&self, latency: Duration) {
1142        self.maybe_rotate();
1143
1144        let bucket_idx = Self::latency_to_bucket(latency);
1145        let slice_idx = self.current_idx.load(Ordering::Acquire);
1146
1147        self.slices[slice_idx].buckets[bucket_idx].fetch_add(1, Ordering::Relaxed);
1148    }
1149
1150    fn quantile(&self, q: f64, window: Duration) -> Option<Duration> {
1151        debug_assert!((0.0..=1.0).contains(&q), "quantile must be in [0, 1]");
1152
1153        let snapshot = self.snapshot(window);
1154        let total: u64 = snapshot.iter().sum();
1155
1156        if total == 0 {
1157            return None;
1158        }
1159
1160        let target = (total as f64 * q).ceil() as u64;
1161        let mut cumsum = 0u64;
1162
1163        for (bucket_idx, &count) in snapshot.iter().enumerate() {
1164            cumsum += count;
1165            if cumsum >= target {
1166                return Some(Self::bucket_to_latency(bucket_idx));
1167            }
1168        }
1169
1170        Some(Self::bucket_to_latency(NUM_BUCKETS - 1))
1171    }
1172
1173    fn total_samples(&self, window: Duration) -> usize {
1174        self.snapshot(window).iter().map(|&v| v as usize).sum()
1175    }
1176
1177    fn snapshot(&self, window: Duration) -> [u64; NUM_BUCKETS] {
1178        let mut result = [0u64; NUM_BUCKETS];
1179        let now_ms = self.now_ms();
1180        let window_ms = window.as_millis() as u64;
1181
1182        for slice in self.slices.iter() {
1183            let start = slice.start_epoch_ms.load(Ordering::Acquire);
1184
1185            if start > 0 && now_ms.saturating_sub(start) < window_ms + SLICE_DURATION_MS {
1186                for (i, bucket) in slice.buckets.iter().enumerate() {
1187                    result[i] += bucket.load(Ordering::Relaxed);
1188                }
1189            }
1190        }
1191
1192        result
1193    }
1194
1195    fn maybe_rotate(&self) {
1196        let now = self.now_ms();
1197        let last_rotate = self.last_rotate.load(Ordering::Relaxed);
1198
1199        // A stale `last_rotate`, for example from a concurrent rotation or a
1200        // clock that moved backwards, can be newer than `now`. Leave the ring
1201        // untouched instead of underflowing or moving `last_rotate` backwards.
1202        let Some(elapsed) = now.checked_sub(last_rotate) else {
1203            return;
1204        };
1205        if elapsed < SLICE_DURATION_MS {
1206            return;
1207        }
1208
1209        if self
1210            .last_rotate
1211            .compare_exchange(last_rotate, now, Ordering::Release, Ordering::Relaxed)
1212            .is_err()
1213        {
1214            return;
1215        }
1216
1217        self.rotate_slice(now);
1218    }
1219
1220    fn rotate_slice(&self, now: u64) {
1221        let old_idx = self.current_idx.load(Ordering::Acquire);
1222        let new_idx = (old_idx + 1) % NUM_SLICES;
1223
1224        let new_slice = &self.slices[new_idx];
1225        // Hide the slice before clearing it, so a concurrent `snapshot`
1226        // cannot pair the new timestamp with bucket counts from its previous
1227        // use.
1228        new_slice.start_epoch_ms.store(0, Ordering::Release);
1229        for bucket in &new_slice.buckets {
1230            bucket.store(0, Ordering::Relaxed);
1231        }
1232        new_slice.start_epoch_ms.store(now, Ordering::Release);
1233
1234        self.current_idx.store(new_idx, Ordering::Release);
1235    }
1236
1237    fn latency_to_bucket(latency: Duration) -> usize {
1238        let ms = latency.as_millis() as u64;
1239
1240        if ms == 0 {
1241            return 0;
1242        }
1243
1244        let bucket = 64 - ms.leading_zeros();
1245        (bucket as usize).min(NUM_BUCKETS - 1)
1246    }
1247
1248    fn bucket_to_latency(bucket_idx: usize) -> Duration {
1249        if bucket_idx == 0 {
1250            Duration::from_millis(1)
1251        } else if bucket_idx >= NUM_BUCKETS - 1 {
1252            Duration::from_secs(64)
1253        } else {
1254            Duration::from_millis(1u64 << bucket_idx)
1255        }
1256    }
1257}

1891    fn test_rotation_hides_slice_before_clearing_buckets() {
1892        const STALE: u64 = 1_000_000;
1893        const ROUNDS: usize = 2_000;
1894
1895        let hist = Arc::new(WindowedHistogram::new());
1896        let pause = Arc::new(AtomicBool::new(false));
1897        let paused = Arc::new(AtomicBool::new(false));
1898        let stop = Arc::new(AtomicBool::new(false));
1899        let finished = Arc::new(AtomicBool::new(false));
1900
1901        let observer = {
1902            let hist = hist.clone();
1903            let pause = pause.clone();
1904            let paused = paused.clone();
1905            let stop = stop.clone();
1906            let finished = finished.clone();
1907
1908            std::thread::spawn(move || {
1909                let mut observed_stale = false;
1910                loop {
1911                    if stop.load(Ordering::Acquire) {
1912                        break;
1913                    }
1914                    if pause.load(Ordering::Acquire) {
1915                        paused.store(true, Ordering::Release);
1916                        while pause.load(Ordering::Acquire) && !stop.load(Ordering::Acquire) {
1917                            std::hint::spin_loop();
1918                        }
1919                        paused.store(false, Ordering::Release);
1920                        continue;
1921                    }
1922
1923                    let snapshot = hist.snapshot(Duration::from_secs(120));
1924                    if snapshot.iter().any(|&count| count >= STALE) {
1925                        observed_stale = true;
1926                        break;
1927                    }
1928                }
1929                finished.store(true, Ordering::Release);
1930                observed_stale
1931            })
1932        };
1933
1934        for _ in 0..ROUNDS {
1935            if finished.load(Ordering::Acquire) {
1936                break;
1937            }
1938
1939            pause.store(true, Ordering::Release);
1940            while !paused.load(Ordering::Acquire) && !finished.load(Ordering::Acquire) {
1941                std::hint::spin_loop();
1942            }
1943            if finished.load(Ordering::Acquire) {
1944                break;
1945            }
1946
1947            // Reset every slice while the observer is quiescent, so preparing
1948            // the next target can never race with a snapshot.
1949            for slice in hist.slices.iter() {
1950                slice.start_epoch_ms.store(0, Ordering::Release);
1951                for bucket in &slice.buckets {
1952                    bucket.store(0, Ordering::Relaxed);
1953                }
1954            }
1955
1956            pause.store(false, Ordering::Release);
1957            while paused.load(Ordering::Acquire) && !finished.load(Ordering::Acquire) {
1958                std::hint::spin_loop();
1959            }
1960            if finished.load(Ordering::Acquire) {
1961                break;
1962            }
1963
1964            for _ in 0..(NUM_SLICES - 1) {
1965                let next = (hist.current_idx.load(Ordering::Acquire) + 1) % NUM_SLICES;
1966                for bucket in &hist.slices[next].buckets {
1967                    bucket.store(STALE, Ordering::Relaxed);
1968                }
1969
1970                hist.rotate_slice(hist.now_ms());
1971            }
1972        }
1973
1974        stop.store(true, Ordering::Release);
1975        pause.store(true, Ordering::Release);
1976        while !finished.load(Ordering::Acquire) {
1977            std::hint::spin_loop();
1978        }
1979
1980        let observed_stale = observer.join().unwrap();
1981        assert!(
1982            !observed_stale,
1983            "snapshot observed bucket counts from a slice being reused"
1984        );

1991        pages: usize,

1996    impl oio::PageList for AsyncPageList {
1997        async fn next_page(&self, ctx: &mut oio::PageContext) -> Result<()> {
1998            let page = self.fetched.fetch_add(1, Ordering::Relaxed);
1999            if page >= self.pages {
2000                ctx.done = true;
2001                return Ok(());
2002            }
2003
2004            tokio::time::sleep(self.delay).await;
2005

2007                let path = format!("page-{page}/entry-{index}");
2008                ctx.entries
2009                    .push_back(oio::Entry::new(&path, MetadataBuilder::file(0).build()));
2010            }
2011            ctx.done = page + 1 >= self.pages;
2012            Ok(())
2013        }
