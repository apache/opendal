257impl AzfileBuilder {
258    fn build_backend(self) -> Result<AzfileBackend> {
259        debug!("backend build started: {:?}", self);
260
261        let root = normalize_root(&self.config.root.unwrap_or_default());
262        debug!("backend use root {root}");
263
264        let endpoint = match &self.config.endpoint {
265            Some(endpoint) => Ok(endpoint.trim_end_matches('/').to_string()),
266            None => Err(Error::new(ErrorKind::ConfigInvalid, "endpoint is empty")
267                .with_operation("Builder::build")
268                .with_context("service", AZFILE_SCHEME)),
269        }?;
270        debug!("backend use endpoint {}", endpoint);
271
272        let account_name = self
273            .config
274            .account_name
275            .clone()
276            .or_else(|| azure_account_name_from_endpoint(endpoint.as_str()));
277
278        let ctx = Context::new().with_file_read(TokioFileRead).with_env(OsEnv);
279        #[cfg(not(target_arch = "wasm32"))]
280        let ctx = ctx.with_command_execute(TokioCommandExecute);
281
282        let tenant_id = self.config.tenant_id.as_deref().filter(|v| !v.is_empty());
283        let client_id = self.config.client_id.as_deref().filter(|v| !v.is_empty());
284        let client_secret_value = self
285            .config
286            .client_secret
287            .as_deref()
288            .filter(|v| !v.is_empty());
289        let authority_host = self
290            .config
291            .authority_host
292            .as_deref()
293            .filter(|v| !v.is_empty());
294
295        let mut client_secret = ClientSecretCredentialProvider::new();
296        if let Some(v) = tenant_id {
297            client_secret = client_secret.with_tenant_id(v);
298        }
299        if let Some(v) = client_id {
300            client_secret = client_secret.with_client_id(v);
301        }
302        if let Some(v) = client_secret_value {
303            client_secret = client_secret.with_client_secret(v);
304        }
305        if let Some(v) = authority_host {
306            client_secret = client_secret.with_authority_host(v);
307        }
308
309        let has_explicit_client_secret =
310            tenant_id.is_some() && client_id.is_some() && client_secret_value.is_some();
311        let mut credential_providers = if has_explicit_client_secret {
312            ProvideCredentialChain::new().push(client_secret).push(
313                DefaultCredentialProvider::builder()
314                    .no_client_secret()
315                    .build(),
316            )
317        } else {
318            ProvideCredentialChain::new().push(
319                DefaultCredentialProvider::builder()
320                    .client_secret(client_secret)
321                    .build(),
322            )
323        };
324
325        if let (Some(account_name), Some(account_key)) = (
326            account_name.as_deref().filter(|v| !v.is_empty()),
327            self.config.account_key.as_deref().filter(|v| !v.is_empty()),
328        ) {
329            credential_providers = credential_providers.push_front(
330                StaticCredentialProvider::new_shared_key(account_name, account_key),
331            );
332        }
333
334        let sas_token = normalize_sas_token(self.config.sas_token.as_deref());
335        if let Some(sas_token) = sas_token {
336            credential_providers =
337                credential_providers.push_front(StaticCredentialProvider::new_sas_token(sas_token));
338        }
339
340        if let Some(customized_credential_chain) = self.credential_providers {
341            credential_providers = customized_credential_chain;
342        }
343
344        let sign_ctx = ctx;
345        let credential_provider = Arc::new(credential_providers);
346        let request_signer = Arc::new(RequestSigner::new());
347
348        let info = ServiceInfo::new(AZFILE_SCHEME, &root, "");
349        let capability = Capability {
350            stat: true,
351
352            read: true,
353
354            write: true,
355            write_with_user_metadata: true,
356            // Azure Files limits each Put Range update to 4 MiB.
357            // ref: <https://learn.microsoft.com/en-us/rest/api/storageservices/put-range>
358            write_total_max_size: Some(4 * 1024 * 1024),
359
360            create_dir: true,
361            delete: true,
362            rename: true,
363
364            list: true,
365
366            shared: true,
367
368            ..Default::default()
369        };
370
371        Ok(AzfileBackend {
372            core: Arc::new(AzfileCore {
373                info,
374                capability,
375                root,
376                endpoint,
377                sign_ctx,
378                credential_provider,
379                request_signer,
380                signers: HttpTransportCache::new(),
381                share_name: self.config.share_name.clone(),
382                skip_signature: self.config.skip_signature,
383            }),
384        })
385    }
386}

394impl Service for AzfileBackend {
395    type Reader = oio::StreamReader<AzfileReader>;
396    type Writer = AzfileWriters;
397    type Lister = oio::PageLister<AzfileLister>;
398    type Deleter = oio::OneShotDeleter<AzfileDeleter>;
399    type Copier = ();
400    type Composer = ();
401
402    fn info(&self) -> ServiceInfo {
403        self.core.info.clone()
404    }
405
406    fn capability(&self) -> Capability {
407        self.core.capability
408    }
409
410    async fn create_dir(
411        &self,
412        ctx: &OperationContext,
413        path: &str,
414        _: OpCreateDir,
415    ) -> Result<RpCreateDir> {
416        self.core.ensure_parent_dir_exists(ctx, path).await?;
417        let resp = self.core.azfile_create_dir(ctx, path).await?;
418        let status = resp.status();
419
420        match status {
421            StatusCode::CREATED => Ok(RpCreateDir::default()),
422            _ => {
423                // we cannot just check status code because 409 Conflict has two meaning:
424                // 1. If a directory by the same name is being deleted when Create Directory is called, the server returns status code 409 (Conflict)
425                // 2. If a directory or file with the same name already exists, the operation fails with status code 409 (Conflict).
426                // but we just need case 2 (already exists)
427                // ref: https://learn.microsoft.com/en-us/rest/api/storageservices/create-directory
428                if resp
429                    .headers()
430                    .get("x-ms-error-code")
431                    .map(|value| value.to_str().unwrap_or(""))
432                    .unwrap_or_else(|| "")
433                    == "ResourceAlreadyExists"
434                {
435                    Ok(RpCreateDir::default())
436                } else {
437                    Err(parse_error(
438                        ErrorContext::new(ServiceOperation("CreateDirectory")),
439                        resp,
440                    ))
441                }
442            }
443        }
444    }
445
446    async fn stat(&self, ctx: &OperationContext, path: &str, _: OpStat) -> Result<RpStat> {
447        let resp = if path.ends_with('/') {
448            self.core.azfile_get_directory_properties(ctx, path).await?
449        } else {
450            self.core.azfile_get_file_properties(ctx, path).await?
451        };
452
453        let status = resp.status();
454        match status {
455            StatusCode::OK => {
456                let headers = resp.headers();
457                let mut meta = parse_into_metadata(path, headers)?.into_builder();
458                let user_meta = parse_prefixed_headers(headers, X_MS_META_PREFIX);
459                if !user_meta.is_empty() {
460                    meta.user_metadata(user_meta);
461                }
462                Ok(RpStat::new(meta.build()))
463            }
464            _ => Err(parse_error(
465                ErrorContext::new(if path.ends_with('/') {
466                    ServiceOperation("GetDirectoryProperties")
467                } else {
468                    ServiceOperation("GetFileProperties")
469                }),
470                resp,
471            )),
472        }
473    }
474    fn read(&self, ctx: &OperationContext, path: &str, args: OpRead) -> Result<Self::Reader> {
475        let output: oio::StreamReader<AzfileReader> = {
476            Ok(oio::StreamReader::new(AzfileReader::new(
477                self.clone(),
478                ctx.clone(),
479                path,
480                args,
481            )))
482        }?;
483
484        Ok(output)
485    }
486
487    fn write(&self, ctx: &OperationContext, path: &str, args: OpWrite) -> Result<Self::Writer> {
488        let output: AzfileWriters = {
489            let w = AzfileWriter::new(
490                self.core.clone(),
491                ctx.clone(),
492                args.clone(),
493                path.to_string(),
494            );
495            let w = if args.append() {
496                AzfileWriters::Two(oio::AppendWriter::new(w))
497            } else {
498                AzfileWriters::One(oio::OneShotWriter::new(w))
499            };
500            Ok(w)
501        }?;
502
503        Ok(output)
504    }
505
506    fn delete(&self, ctx: &OperationContext) -> Result<Self::Deleter> {
507        let output: oio::OneShotDeleter<AzfileDeleter> = {
508            Ok(oio::OneShotDeleter::new(AzfileDeleter::new(
509                self.core.clone(),
510                ctx.clone(),
511            )))
512        }?;
513
514        Ok(output)
515    }
516
517    fn list(&self, ctx: &OperationContext, path: &str, args: OpList) -> Result<Self::Lister> {
518        let output: oio::PageLister<AzfileLister> = {
519            let l = AzfileLister::new(
520                self.core.clone(),
521                ctx.clone(),
522                path.to_string(),
523                args.limit(),
524            );
525
526            Ok(oio::PageLister::new(l))
527        }?;
528
529        Ok(output)
530    }
531
532    fn copy(
533        &self,
534        _ctx: &OperationContext,
535        _from: &str,
536        _to: &str,
537        _args: OpCopy,
538    ) -> Result<Self::Copier> {
539        Err(Error::new(
540            ErrorKind::Unsupported,
541            "operation is not supported",
542        ))
543    }
544
545    async fn rename(
546        &self,
547        ctx: &OperationContext,
548        from: &str,
549        to: &str,
550        _: OpRename,
551    ) -> Result<RpRename> {
552        self.core.ensure_parent_dir_exists(ctx, to).await?;
553        let resp = self.core.azfile_rename(ctx, from, to).await?;
554        let status = resp.status();
555        match status {
556            StatusCode::OK => Ok(RpRename::default()),
557            _ => Err(parse_error(
558                ErrorContext::new(ServiceOperation("Rename")),
559                resp,
560            )),
561        }
562    }

565        &self,
566        _ctx: &OperationContext,
567        _path: &str,
568        _args: OpPresign,
569    ) -> Result<RpPresign> {
570        Err(Error::new(
571            ErrorKind::Unsupported,
572            "operation is not supported",
573        ))
574    }
