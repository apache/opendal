45/// Drops the leading `?` of a SAS token and treats an empty token as unset.
46pub(crate) fn normalize_sas_token(token: Option<&str>) -> Option<&str> {
47    token
48        .map(|token| token.trim_start_matches('?'))
49        .filter(|token| !token.is_empty())
50}

95    pub async fn sign<T>(&self, ctx: &OperationContext, req: Request<T>) -> Result<Request<T>> {
96        let (mut parts, body) = req.into_parts();
97
98        // Insert x-ms-version header for normal requests.
99        parts.headers.insert(
100            HeaderName::from_static(X_MS_VERSION),
101            // consistent with azdls and azblob
102            HeaderValue::from_static("2022-11-02"),
103        );
104
105        // x-ms-version must be set even when signing is skipped.
106        if self.skip_signature {
107            return Ok(Request::from_parts(parts, body));
108        }
109
110        self.signer(ctx)
111            .sign(&mut parts, None)
112            .await
113            .map_err(|e| new_request_sign_error(e.into()))?;
114
115        Ok(Request::from_parts(parts, body))
116    }

129        ctx: &OperationContext,
130        path: &str,
131        range: BytesRange,
132    ) -> Result<Response<HttpBody>> {
133        let p = build_absolute_path(&self.root, path);
134
135        let url = format!(

137            self.endpoint,
138            self.share_name,
139            percent_encode_path(&p)
140        );
141
142        let mut req = Request::get(&url);
143
144        if !range.is_full() {
145            req = req.header(RANGE, range.to_header());
146        }
147
148        let req = req
149            .extension(Operation::Read)
150            .extension(ServiceOperation("GetFile"));
151
152        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
153        let req = self.sign(ctx, req).await?;
154        ctx.http_transport().fetch(req).await
155    }

157    pub async fn azfile_create_file(
158        &self,
159        ctx: &OperationContext,
160        path: &str,
161        size: usize,
162        args: &OpWrite,
163    ) -> Result<Response<Buffer>> {
164        let p = build_absolute_path(&self.root, path)
165            .trim_start_matches('/')
166            .to_string();
167        let url = format!(
168            "{}/{}/{}",
169            self.endpoint,
170            self.share_name,
171            percent_encode_path(&p)
172        );
173
174        let mut req = Request::put(&url);
175
176        // x-ms-content-length specifies the maximum size for the file, up to 4 tebibytes (TiB)
177        // https://learn.microsoft.com/en-us/rest/api/storageservices/create-file
178        req = req.header(X_MS_CONTENT_LENGTH, size);
179
180        req = req.header(X_MS_TYPE, "file");
181
182        // Content length must be 0 for create request.
183        req = req.header(CONTENT_LENGTH, 0);
184
185        if let Some(ty) = args.content_type() {
186            req = req.header(CONTENT_TYPE, ty);
187        }
188
189        if let Some(pos) = args.content_disposition() {
190            req = req.header(CONTENT_DISPOSITION, pos);
191        }
192
193        // Set user metadata headers.
194        if let Some(user_metadata) = args.user_metadata() {
195            for (key, value) in user_metadata {
196                req = req.header(format!("{X_MS_META_PREFIX}{key}"), value);
197            }
198        }
199
200        let req = req
201            .extension(Operation::Write)
202            .extension(ServiceOperation("CreateFile"));
203
204        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
205        let req = self.sign(ctx, req).await?;
206        self.send(ctx, req).await
207    }

297    pub async fn azfile_rename_request(
298        &self,
299        ctx: &OperationContext,
300        path: &str,
301        new_path: &str,
302    ) -> Result<Request<Buffer>> {
303        let p = build_absolute_path(&self.root, path)
304            .trim_start_matches('/')
305            .to_string();
306
307        let new_p = build_absolute_path(&self.root, new_path)
308            .trim_start_matches('/')
309            .to_string();
310
311        let url = if path.ends_with('/') {
312            format!(
313                "{}/{}/{}?restype=directory&comp=rename",
314                self.endpoint,
315                self.share_name,
316                percent_encode_path(&new_p)
317            )
318        } else {
319            format!(
320                "{}/{}/{}?comp=rename",
321                self.endpoint,
322                self.share_name,
323                percent_encode_path(&new_p)
324            )
325        };
326
327        let mut req = Request::put(&url);
328
329        req = req.header(CONTENT_LENGTH, 0);
330
331        // x-ms-file-rename-source specifies the file or directory to be renamed.
332        // the value must be a URL style path
333        // the official document does not mention the URL style path
334        // find the solution from the community FAQ and implementation of the Java-SDK
335        // ref: https://learn.microsoft.com/en-us/answers/questions/799611/azure-file-service-rest-api(rename)?page=1
336        let source_url = format!(
337            "{}/{}/{}",
338            self.endpoint,
339            self.share_name,
340            percent_encode_path(&p)
341        );
342        let mut source = build_header_value(&source_url)?;
343        if !self.skip_signature {
344            // Only a SAS credential changes the URI; shared key and bearer
345            // token authorize the source through the rename request.
346            let (mut parts, ()) = Request::get(&source_url)
347                .body(())
348                .map_err(new_request_build_error)?
349                .into_parts();
350            let unsigned = parts.uri.clone();
351            self.signer(ctx)
352                .sign(&mut parts, None)
353                .await
354                .map_err(|e| new_request_sign_error(e.into()))?;
355            if parts.uri != unsigned {
356                source = build_header_value(&parts.uri.to_string())?;
357                source.set_sensitive(true);
358            }
359        }
360
361        req = req.header(X_MS_FILE_RENAME_SOURCE, source);
362
363        req = req.header(X_MS_FILE_RENAME_REPLACE_IF_EXISTS, "true");
364
365        req.extension(Operation::Rename)
366            .extension(ServiceOperation("Rename"))
367            .body(Buffer::new())
368            .map_err(new_request_build_error)
369    }

465    pub async fn azfile_list(
466        &self,
467        ctx: &OperationContext,
468        path: &str,
469        limit: &Option<usize>,
470        continuation: &str,
471    ) -> Result<Response<Buffer>> {
472        let p = build_absolute_path(&self.root, path)
473            .trim_start_matches('/')
474            .to_string();
475
476        let url = format!(
477            "{}/{}/{}",
478            self.endpoint,
479            self.share_name,
480            percent_encode_path(&p),
481        );
482
483        let mut url = QueryPairsWriter::new(&url)
484            .push("restype", "directory")
485            .push("comp", "list")
486            .push("include", "Timestamps,ETag");
487
488        if !continuation.is_empty() {
489            url = url.push("marker", &percent_encode_path(continuation));
490        }
491
492        if let Some(limit) = limit {
493            url = url.push("maxresults", &limit.to_string());
494        }
495
496        let req = Request::get(url.finish());
497
498        let req = req
499            .extension(Operation::List)
500            .extension(ServiceOperation("ListDirectoriesAndFiles"));
501
502        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
503        let req = self.sign(ctx, req).await?;
504        self.send(ctx, req).await
505    }

507    pub async fn ensure_parent_dir_exists(&self, ctx: &OperationContext, path: &str) -> Result<()> {
508        let mut dirs = VecDeque::default();
509        // azure file service does not support recursive directory creation
510        let mut p = path;
511        while p != "/" {
512            p = get_parent(p);
513            dirs.push_front(p);
514        }
515
516        let mut pop_dir_count = dirs.len();
517        for dir in dirs.iter().rev() {
518            let resp = self.azfile_get_directory_properties(ctx, dir).await?;
519            if resp.status() == StatusCode::NOT_FOUND {
520                pop_dir_count -= 1;
521                continue;
522            }
523            break;
524        }
525
526        for dir in dirs.iter().skip(pop_dir_count) {
527            let resp = self.azfile_create_dir(ctx, dir).await?;
528
529            if resp.status() == StatusCode::CREATED {
530                continue;
531            }
532
533            if resp
534                .headers()
535                .get("x-ms-error-code")
536                .map(|value| value.to_str().unwrap_or(""))
537                .unwrap_or_else(|| "")
538                == "ResourceAlreadyExists"
539            {
540                continue;
541            }
542
543            return Err(parse_error(
544                ErrorContext::new(ServiceOperation("CreateDirectory")),
545                resp,
546            ));
547        }
548
549        Ok(())
550    }

559/// AzfileError is the error returned by azure file service.
560#[derive(Default, Deserialize)]
561#[serde(default, rename_all = "PascalCase")]
562struct AzfileError {
563    code: String,
564    message: String,
565    query_parameter_name: String,
566    query_parameter_value: String,
567    reason: String,
568}
569
570impl Debug for AzfileError {
571    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
572        let mut de = f.debug_struct("AzfileError");
573        de.field("code", &self.code);
574        // replace `\n` to ` ` for better reading.
575        de.field("message", &self.message.replace('\n', " "));
576
577        if !self.query_parameter_name.is_empty() {
578            de.field("query_parameter_name", &self.query_parameter_name);
579        }
580        if !self.query_parameter_value.is_empty() {
581            de.field("query_parameter_value", &self.query_parameter_value);
582        }
583        if !self.reason.is_empty() {
584            de.field("reason", &self.reason);
585        }
586
587        de.finish()
588    }
589}

591/// Context needed to classify an error from this service.
592#[derive(Clone, Copy, Debug)]
593pub(crate) struct ErrorContext {
594    service_operation: ServiceOperation,
595}
596
597impl ErrorContext {
598    pub(crate) const fn new(service_operation: ServiceOperation) -> Self {
599        Self { service_operation }
600    }
601}
602
603/// Parse an error response using its service request context.
604pub(crate) fn parse_error(ctx: ErrorContext, resp: Response<Buffer>) -> Error {
605    let (parts, body) = resp.into_parts();
606    let bs = body.to_bytes();
607
608    let mut azfile_error = de::from_reader::<_, AzfileError>(bs.clone().reader()).ok();
609    if azfile_error.as_ref().is_none_or(|err| err.code.is_empty())
610        && let Some(code) = parts
611            .headers
612            .get("x-ms-error-code")
613            .and_then(|value| value.to_str().ok())
614    {
615        azfile_error.get_or_insert_with(AzfileError::default).code = code.to_string();
616    }
617
618    let (mut kind, mut retryable) = match parts.status {
619        StatusCode::NOT_FOUND => (ErrorKind::NotFound, false),
620        StatusCode::FORBIDDEN => (ErrorKind::PermissionDenied, false),
621        StatusCode::RANGE_NOT_SATISFIABLE => (ErrorKind::RangeNotSatisfied, false),
622        StatusCode::TOO_MANY_REQUESTS => (ErrorKind::RateLimited, true),
623        StatusCode::INTERNAL_SERVER_ERROR
624        | StatusCode::BAD_GATEWAY
625        | StatusCode::SERVICE_UNAVAILABLE
626        | StatusCode::GATEWAY_TIMEOUT => (ErrorKind::Unexpected, true),
627        _ => (ErrorKind::Unexpected, false),
628    };
629
630    if let Some(azfile_error) = &azfile_error {
631        match azfile_error.code.as_str() {
632            "ParentNotFound" => {
633                (kind, retryable) = (ErrorKind::NotFound, false);
634            }
635            "ResourceAlreadyExists" => {
636                (kind, retryable) = (ErrorKind::AlreadyExists, false);
637            }
638            "DeletePending" | "ShareBeingDeleted"
639                if ctx.service_operation == ServiceOperation("CreateDirectory") =>
640            {
641                (kind, retryable) = (ErrorKind::Conflict, true);
642            }
643            "CannotDeleteFileOrDirectory"
644            | "ContainerQuotaDowngradeNotAllowed"
645            | "DeletePending"
646            | "DeleteShareWhenSnapshotLeased"
647            | "DirectoryNotEmpty"
648            | "FileGenerationMismatch"
649            | "FileLockConflict"
650            | "FileOpenBySmbClient"
651            | "LeaseAcquireDuringShareDelete"
652            | "LeaseIdMismatchWithFileLeaseOperation"
653            | "LeaseIdMismatchWithFileOperation"
654            | "LeaseIdMismatchWithFileShareLeaseOperation"
655            | "LeaseIdMismatchWithFileShareOperation"
656            | "LeaseIdMissingWithFileOperation"
657            | "LeaseIdMissingWithFileShareOperation"
658            | "LeaseLostWithFileOperation"
659            | "LeaseLostWithFileShareOperation"
660            | "LeaseNotPresentWithFileLeaseOperation"
661            | "LeaseNotPresentWithFileOperation"
662            | "LeaseNotPresentWithFileShareLeaseOperation"
663            | "LeaseNotPresentWithFileShareOperation"
664            | "PreviousSnapshotNotFound"
665            | "PreviousSnapshotOperationNotSupported"
666            | "ReadOnlyAttribute"
667            | "RenameCannotOverwriteDestinationFile"
668            | "RenameCycle"
669            | "RenameDirectoryHasOpenFiles"
670            | "ShareAlreadyExists"
671            | "ShareBeingDeleted"
672            | "ShareHasSnapshots"
673            | "ShareSnapshotInProgress"
674            | "SharingViolation" => {
675                (kind, retryable) = (ErrorKind::Conflict, false);
676            }
677            _ if matches!(
678                parts.status,
679                StatusCode::CONFLICT | StatusCode::PRECONDITION_FAILED
680            ) =>
681            {
682                (kind, retryable) = (ErrorKind::Unexpected, false);
683            }
684            _ => {}
685        }
686    }
687
688    let message = azfile_error
689        .map(|err| format!("{err:?}"))
690        .unwrap_or_else(|| String::from_utf8_lossy(&bs).into_owned());
691
692    let mut err = Error::new(kind, &message);
693
694    err = err.with_context("service_operation", ctx.service_operation.0);
695    err = with_azure_error_response_context(err, parts);
696
697    if retryable {
698        err = err.set_temporary();
699    }
700
701    err
702}

744    fn parse_native_error(operation: &'static str, status: StatusCode, code: &str) -> Error {
745        let resp = Response::builder()
746            .status(status)
747            .header("x-ms-error-code", code)
748            .body(Buffer::new())
749            .expect("response must build");
750        parse_error(ErrorContext::new(ServiceOperation(operation)), resp)
751    }
