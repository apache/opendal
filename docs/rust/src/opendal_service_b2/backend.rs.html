66            None
67        } else {
68            Some(root.to_string())
69        };
70

72    }

77            None
78        } else {
79            Some(application_key_id.to_string())
80        };

82        self
83    }

88            None
89        } else {
90            Some(application_key.to_string())

116    /// Builds the backend and returns the result of B2Backend.
117    fn build(self) -> Result<impl Service> {
118        debug!("backend build started: {:?}", self);
119
120        let root = normalize_root(&self.config.root.clone().unwrap_or_default());
121        debug!("backend use root {}", root);
122
123        // Handle bucket.
124        if self.config.bucket.is_empty() {
125            return Err(Error::new(ErrorKind::ConfigInvalid, "bucket is empty")
126                .with_operation("Builder::build")
127                .with_context("service", B2_SCHEME));
128        }
129
130        debug!("backend use bucket {}", self.config.bucket);
131
132        // Handle bucket_id.
133        if self.config.bucket_id.is_empty() {
134            return Err(Error::new(ErrorKind::ConfigInvalid, "bucket_id is empty")
135                .with_operation("Builder::build")
136                .with_context("service", B2_SCHEME));
137        }
138
139        debug!("backend bucket_id {}", self.config.bucket_id);
140
141        let application_key_id = match &self.config.application_key_id {
142            Some(application_key_id) => Ok(application_key_id.clone()),
143            None => Err(
144                Error::new(ErrorKind::ConfigInvalid, "application_key_id is empty")
145                    .with_operation("Builder::build")
146                    .with_context("service", B2_SCHEME),
147            ),
148        }?;
149
150        let application_key = match &self.config.application_key {
151            Some(key_id) => Ok(key_id.clone()),
152            None => Err(
153                Error::new(ErrorKind::ConfigInvalid, "application_key is empty")
154                    .with_operation("Builder::build")
155                    .with_context("service", B2_SCHEME),
156            ),
157        }?;
158
159        let signer = B2Signer {
160            application_key_id,
161            application_key,
162            ..Default::default()
163        };
164
165        Ok(B2Backend {
166            core: Arc::new(B2Core {
167                info: ServiceInfo::new(B2_SCHEME, &root, ""),
168                capability: Capability {
169                    stat: true,
170                    stat_with_version: true,
171                    stat_with_override_cache_control: true,
172                    stat_with_override_content_disposition: true,
173                    stat_with_override_content_type: true,
174
175                    read: true,
176                    read_with_version: true,
177                    read_with_override_cache_control: true,
178                    read_with_override_content_disposition: true,
179                    read_with_override_content_type: true,
180                    read_with_suffix: true,
181
182                    write: true,
183                    write_can_empty: true,
184                    write_can_multi: true,
185                    write_with_content_type: true,
186                    write_with_user_metadata: true,
187                    write_with_checksum: true,
188                    // The min multipart size of b2 is 5 MiB.
189                    //
190                    // ref: <https://www.backblaze.com/docs/cloud-storage-large-files>
191                    write_multi_min_size: Some(5 * 1024 * 1024),
192                    // The max multipart size of b2 is 5 Gb.
193                    //
194                    // ref: <https://www.backblaze.com/docs/cloud-storage-large-files>
195                    write_multi_max_size: if cfg!(target_pointer_width = "64") {
196                        Some(5 * 1024 * 1024 * 1024)
197                    } else {
198                        Some(usize::MAX)
199                    },
200
201                    delete: true,
202                    copy: true,
203                    restore: true,
204                    restore_with_version: true,
205
206                    list: true,
207                    list_with_limit: true,
208                    list_with_start_after: true,
209                    list_with_recursive: true,
210                    list_with_versions: true,
211                    list_with_deleted: true,
212
213                    presign: true,
214                    presign_read: true,
215                    presign_write: true,
216                    presign_stat: true,
217
218                    shared: true,
219
220                    ..Default::default()
221                },
222                signer: Arc::new(RwLock::new(signer)),
223                root,
224
225                bucket: self.config.bucket.clone(),
226                bucket_id: self.config.bucket_id.clone(),
227            }),
228        })
229    }

266    async fn stat(&self, ctx: &OperationContext, path: &str, args: OpStat) -> Result<RpStat> {
267        if let Some(version) = args.version() {
268            if path.ends_with('/') {
269                return Err(Error::new(
270                    ErrorKind::NotFound,
271                    "no such data version for this path",
272                ));
273            }
274            if version.is_empty() {
275                return Err(Error::new(
276                    ErrorKind::ConfigInvalid,
277                    "empty stat version ID",
278                ));
279            }
280
281            let key = build_absolute_path(&self.core.root, path);
282            let body = self
283                .core
284                .get_data_version(ctx, &key, version, Operation::Stat)
285                .await?;
286            let file =
287                serde_json::from_reader(body.reader()).map_err(new_json_deserialize_error)?;
288            return Ok(RpStat::new(parse_file_info(&file).build()));
289        }
290
291        if path == "/" {
292            return Ok(RpStat::new(MetadataBuilder::dir().build()));
293        }
294
295        // Listing by name returns the current file's metadata without first resolving its ID.
296
297        let delimiter = if path.ends_with('/') { Some("/") } else { None };
298
299        let file_info = self.core.get_file_info(ctx, path, delimiter).await?;
300        let meta = parse_file_info(&file_info).build();
301        Ok(RpStat::new(meta))
302    }
303    fn read(&self, ctx: &OperationContext, path: &str, args: OpRead) -> Result<Self::Reader> {
304        if let Some(version) = args.version() {
305            if version.is_empty() {
306                return Err(Error::new(
307                    ErrorKind::ConfigInvalid,
308                    "empty read version ID",
309                ));
310            }
311            if args.override_cache_control().is_some()
312                || args.override_content_disposition().is_some()
313                || args.override_content_type().is_some()
314            {
315                return Err(Error::new(
316                    ErrorKind::Unsupported,
317                    "B2 does not support response overrides for version-selected reads",
318                ));
319            }
320        }
321        let output: oio::StreamReader<B2Reader> = {
322            Ok(oio::StreamReader::new(B2Reader::new(
323                self.clone(),
324                ctx.clone(),
325                path,
326                args,
327            )))
328        }?;
329
330        Ok(output)
331    }
332
333    fn write(&self, ctx: &OperationContext, path: &str, args: OpWrite) -> Result<Self::Writer> {
334        let concurrent = args.concurrent();
335        let writer = B2Writer::new(self.core.clone(), ctx.clone(), path, args)?;
336        Ok(oio::MultipartWriter::new(
337            ctx.executor().clone(),
338            writer,
339            concurrent,
340        ))
341    }
342
343    fn delete(&self, ctx: &OperationContext) -> Result<Self::Deleter> {
344        let output: oio::OneShotDeleter<B2Deleter> = {
345            Ok(oio::OneShotDeleter::new(B2Deleter::new(
346                self.core.clone(),
347                ctx.clone(),
348            )))
349        }?;
350
351        Ok(output)
352    }

354    fn list(&self, ctx: &OperationContext, path: &str, args: OpList) -> Result<Self::Lister> {
355        if args.versions() || args.deleted() {
356            Ok(TwoWays::Two(B2VersionLister::new(
357                self.core.clone(),
358                ctx.clone(),
359                path,
360                &args,
361            )))
362        } else {
363            Ok(TwoWays::One(oio::PageLister::new(B2Lister::new(
364                self.core.clone(),
365                ctx.clone(),
366                path,
367                args.recursive(),
368                args.limit(),
369                args.start_after(),
370            ))))
371        }
372    }

374    fn copy(
375        &self,
376        ctx: &OperationContext,
377        from: &str,
378        to: &str,
379        _args: OpCopy,
380    ) -> Result<Self::Copier> {
381        let core = self.core.clone();
382        let ctx = ctx.clone();
383        let from = from.to_string();
384        let to = to.to_string();
385
386        Ok(oio::OneShotCopier::new(async move {
387            if from.ends_with('/') {
388                return Err(Error::new(ErrorKind::IsADirectory, "is a directory"));
389            }
390
391            let file_info = core.get_file_info(&ctx, &from, None).await?;
392            let source_file_id = file_info.file_id;
393            let source_content_length = file_info.content_length;
394
395            let Some(source_file_id) = source_file_id else {
396                return Err(Error::new(ErrorKind::IsADirectory, "is a directory"));
397            };
398
399            let req = core
400                .copy_file_request(&ctx, &source_file_id, &to, Operation::Copy)
401                .await?;
402            let resp = core.send(&ctx, req).await?;
403
404            let status = resp.status();
405
406            match status {
407                StatusCode::OK => {
408                    let metadata = MetadataBuilder::file(source_content_length);
409                    Ok(metadata.build())
410                }
411                _ => Err(parse_error(
412                    ErrorContext::new(ServiceOperation("CopyFile")),
413                    resp,
414                )),
415            }
416        }))
417    }

419    async fn restore(
420        &self,
421        ctx: &OperationContext,
422        path: &str,
423        args: OpRestore,
424    ) -> Result<RpRestore> {
425        args.check(&self.capability())?;
426
427        let key = build_absolute_path(&self.core.root, path);
428        if let Some(version) = args.version() {
429            if version.is_empty() {
430                return Err(Error::new(
431                    ErrorKind::ConfigInvalid,
432                    "empty restore version ID",
433                ));
434            }
435            self.core
436                .get_data_version(ctx, &key, version, Operation::Restore)
437                .await?;
438
439            let req = self
440                .core
441                .copy_file_request(ctx, version, path, Operation::Restore)
442                .await?;
443            // A lost copy response can otherwise cause a second promotion.
444            let resp = self
445                .core
446                .send(ctx, req)
447                .await
448                .map_err(Error::set_permanent)?;
449            return match resp.status() {
450                StatusCode::OK => Ok(RpRestore::new()),
451                status => Err(
452                    parse_error(ErrorContext::new(ServiceOperation("CopyFile")), resp)
453                        .with_permanent(status.is_server_error()),
454                ),
455            };
456        }
457
458        let mut start_file_id = None;
459        let mut current_marker = None;
460        let marker = 'history: loop {
461            let request = ListFileVersionsRequest {
462                bucket_id: &self.core.bucket_id,
463                prefix: &key,
464                max_file_count: Some(2),
465                start_file_name: start_file_id.as_ref().map(|_| key.as_str()),
466                start_file_id: start_file_id.as_deref(),
467                delimiter: None,
468            };
469            let page: ListFileVersionsResponse<FileVersion> = self
470                .core
471                .list_file_versions(ctx, request, Operation::Restore)
472                .await?;
473            for file in page.files {
474                if file.file_name != key {
475                    break 'history None;
476                }
477                match file.action {
478                    FileVersionAction::Upload | FileVersionAction::Copy => {
479                        if current_marker.is_none() {
480                            return Ok(RpRestore::new());
481                        }
482                        break 'history current_marker;
483                    }
484                    FileVersionAction::Hide if current_marker.is_none() => {
485                        current_marker = Some(file.file_id);
486                    }
487                    FileVersionAction::Hide | FileVersionAction::Start => {}
488                }
489            }
490
491            // A hide marker can remain after all data versions are gone.
492            if page.next_file_name.as_deref() != Some(&key) {
493                break None;
494            }
495            start_file_id = Some(
496                page.next_file_id
497                    .filter(|next| !next.is_empty() && Some(next) != start_file_id.as_ref())
498                    .ok_or_else(|| {
499                        Error::new(ErrorKind::Unexpected, "version listing did not advance")
500                    })?,
501            );
502        };
503        let Some(marker) = marker else {
504            return Err(Error::new(
505                ErrorKind::NotFound,
506                "no live or recoverable version exists",
507            ));
508        };
509        if marker.is_empty() {
510            return Err(Error::new(
511                ErrorKind::Unexpected,
512                "current hide marker has no version ID",
513            ));
514        }
515
516        let req = self
517            .core
518            .delete_file_version_request(ctx, &key, &marker)
519            .await?;
520        // Replaying discovery after a lost delete response can remove another marker.
521        let resp = self
522            .core
523            .send(ctx, req)
524            .await
525            .map_err(Error::set_permanent)?;
526        match resp.status() {
527            StatusCode::OK => Ok(RpRestore::new()),
528            status => Err(parse_error(
529                ErrorContext::new(ServiceOperation("DeleteFileVersion")),
530                resp,
531            )
532            .with_permanent(status.is_server_error())),
533        }
534    }

549    async fn presign(
550        &self,
551        ctx: &OperationContext,
552        path: &str,
553        args: OpPresign,
554    ) -> Result<RpPresign> {
555        match args.operation() {
556            PresignOperation::Stat(stat) => {
557                if stat.version().is_some() {
558                    return Err(Error::new(
559                        ErrorKind::Unsupported,
560                        "B2 does not support presigning stat for a selected version",
561                    ));
562                }
563                self.core
564                    .presign_download(
565                        ctx,
566                        path,
567                        args.expire(),
568                        BytesRange::default(),
569                        DownloadResponseOverrides {
570                            content_type: stat.override_content_type(),
571                            content_disposition: stat.override_content_disposition(),
572                            cache_control: stat.override_cache_control(),
573                        },
574                    )
575                    .await
576            }
577            PresignOperation::Read(range, read) => {
578                if read.version().is_some() {
579                    return Err(Error::new(
580                        ErrorKind::Unsupported,
581                        "B2 does not support presigning reads for a selected version",
582                    ));
583                }
584                self.core
585                    .presign_download(
586                        ctx,
587                        path,
588                        args.expire(),
589                        *range,
590                        DownloadResponseOverrides {
591                            content_type: read.override_content_type(),
592                            content_disposition: read.override_content_disposition(),
593                            cache_control: read.override_cache_control(),
594                        },
595                    )
596                    .await
597            }
598            PresignOperation::Write(write) => {
599                if write.checksum().is_some() {
600                    return Err(Error::new(
601                        ErrorKind::Unsupported,
602                        "B2 does not support presigning writes with a checksum intent",
603                    ));
604                }
605                let resp = self.core.get_upload_url(ctx).await?;
606
607                let mut req = Request::post(&resp.upload_url);
608
609                req = req.header(http::header::AUTHORIZATION, resp.authorization_token);
610                req = req.header(
611                    "X-Bz-File-Name",
612                    percent_encode_path(&build_absolute_path(&self.core.root, path)),
613                );
614                let content_type = write.content_type().unwrap_or("b2/x-auto");
615                req = req.header(http::header::CONTENT_TYPE, content_type);
616                req = req.header(constants::X_BZ_CONTENT_SHA1, "do_not_verify");
617
618                let file_info_options = [
619                    (
620                        constants::CACHE_CONTROL_FILE_INFO_KEY,
621                        write.cache_control(),
622                    ),
623                    (
624                        constants::CONTENT_DISPOSITION_FILE_INFO_KEY,
625                        write.content_disposition(),
626                    ),
627                    (
628                        constants::CONTENT_ENCODING_FILE_INFO_KEY,
629                        write.content_encoding(),
630                    ),
631                    (
632                        constants::CONTENT_LANGUAGE_FILE_INFO_KEY,
633                        write.content_language(),
634                    ),
635                ];
636                for (key, value) in file_info_options {
637                    if let Some(value) = value {
638                        req = req.header(
639                            format!("{}{key}", constants::X_BZ_INFO_PREFIX),
640                            percent_encode_path(value),
641                        );
642                    }
643                }
644
645                if let Some(user_metadata) = write.user_metadata() {
646                    for (key, value) in user_metadata {
647                        if file_info_options.iter().any(|(reserved_key, option)| {
648                            option.is_some() && key.eq_ignore_ascii_case(reserved_key)
649                        }) {
650                            continue;
651                        }
652                        req = req.header(
653                            format!("{}{}", constants::X_BZ_INFO_PREFIX, key),
654                            percent_encode_path(value),
655                        );
656                    }
657                }
658
659                let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
660                // We don't need this request anymore, consume it directly.
661                let (parts, _) = req.into_parts();
662
663                Ok(RpPresign::new(PresignedRequest::new(
664                    parts.method,
665                    parts.uri,
666                    parts.headers,
667                )))
668            }
669            PresignOperation::Delete(_) => Err(Error::new(
670                ErrorKind::Unsupported,
671                "operation is not supported",
672            )),
673            _ => Err(Error::new(
674                ErrorKind::Unsupported,
675                "operation is not supported",
676            )),
677        }
678    }

734
735        impl HttpTransport for VersionTransport {
736            async fn fetch(&self, request: Request<Buffer>) -> Result<Response<HttpBody>> {
737                let (status, body) = match request.uri().path() {
738                    "/b2api/v4/b2_authorize_account" => (
739                        StatusCode::OK,
740                        r#"{"authorizationToken":"token","apiInfo":{"storageApi":{"apiUrl":"https://api.example","downloadUrl":"https://download.example"}}}"#,
741                    ),
742                    "/b2api/v2/b2_get_file_info" => {
743                        assert_eq!(request.method(), http::Method::POST);
744                        assert_eq!(
745                            request.extensions().get::<Operation>(),
746                            Some(&self.operation)
747                        );
748                        assert_eq!(request.body().to_bytes().as_ref(), br#"{"fileId":"old"}"#);
749                        (self.version_status, self.version_body)
750                    }
751                    "/b2api/v2/b2_copy_file" if self.operation == Operation::Restore => {
752                        assert_eq!(
753                            request.extensions().get::<Operation>(),
754                            Some(&Operation::Restore)
755                        );
756                        let body: serde_json::Value =
757                            serde_json::from_reader(request.body().clone().reader()).unwrap();
758                        assert_eq!(body["sourceFileId"], "old");
759                        assert_eq!(body["fileName"], "root/file");
760                        (
761                            StatusCode::FORBIDDEN,
762                            r#"{"status":403,"code":"unauthorized","message":"copy denied"}"#,
763                        )
764                    }
765                    "/b2api/v2/b2_list_file_names" => {
766                        let query = request.uri().query().unwrap();
767                        let body = if query.contains("prefix=root/dir/") {
768                            r#"{"files":[{"fileName":"root/dir/","contentLength":0}]}"#
769                        } else {
770                            assert!(query.contains("prefix=root/file"));
771                            r#"{"files":[{
772                                "fileName":"root/file","fileId":"new","contentLength":13,
773                                "fileInfo":{"content-disposition":"inline","cache-control":"no-store"}
774                            }]}"#
775                        };
776                        (StatusCode::OK, body)
777                    }
778                    other => panic!("unexpected path: {other}"),
779                };
780                let buffer = Buffer::from(body);
781                Response::builder()
782                    .status(status)
783                    .body(HttpBody::new(
784                        stream::iter([Ok(buffer)]),
785                        Some(body.len() as u64),
786                    ))
787                    .map_err(new_request_build_error)
788            }
789        }
790
791        let op = Operator::new(
792            B2Builder::default()
793                .root("root")
794                .bucket("test-bucket")
795                .bucket_id("bucket-id")
796                .application_key_id("test-key")
797                .application_key("test-secret"),
798        )?;
799        let transport = HttpTransporter::new(VersionTransport {
800            operation,
801            version_status,
802            version_body,
803        });
804        Ok(op.with_context(OperationContext::new().with_http_transport(transport)))
805    }
