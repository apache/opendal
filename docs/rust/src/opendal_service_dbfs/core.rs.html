48impl DbfsCore {
49    pub async fn dbfs_create_dir(
50        &self,
51        ctx: &OperationContext,
52        path: &str,
53    ) -> Result<Response<Buffer>> {
54        let url = format!("{}/api/2.0/dbfs/mkdirs", self.endpoint);
55        let mut req = Request::post(&url);
56
57        let auth_header_content = format!("Bearer {}", self.token);
58        req = req.header(header::AUTHORIZATION, auth_header_content);
59
60        let p = build_rooted_abs_path(&self.root, path)
61            .trim_end_matches('/')
62            .to_string();
63
64        let req_body = &json!({
65            "path": percent_encode_path(&p),
66        });
67        let body = Buffer::from(Bytes::from(req_body.to_string()));
68
69        let req = req
70            .extension(Operation::CreateDir)
71            .extension(ServiceOperation("Mkdirs"))
72            .body(body)
73            .map_err(new_request_build_error)?;
74
75        ctx.http_transport().send(req).await
76    }

165    pub fn dbfs_create_file_request(&self, path: &str, body: Bytes) -> Result<Request<Buffer>> {
166        let url = format!("{}/api/2.0/dbfs/put", self.endpoint);
167
168        let contents = BASE64_STANDARD.encode(body);
169        let mut req = Request::post(&url);
170
171        let auth_header_content = format!("Bearer {}", self.token);
172        req = req.header(header::AUTHORIZATION, auth_header_content);

174        let req_body = &json!({
175            "path": path,
176            "contents": contents,
177            "overwrite": true,
178        });
179
180        let body = Buffer::from(Bytes::from(req_body.to_string()));
181
182        req.extension(Operation::Write)
183            .extension(ServiceOperation("Put"))

217    pub async fn dbfs_ensure_parent_path(&self, ctx: &OperationContext, path: &str) -> Result<()> {
218        let resp = self.dbfs_get_status(ctx, path).await?;
219
220        match resp.status() {
221            StatusCode::OK => return Ok(()),
222            StatusCode::NOT_FOUND => {
223                self.dbfs_create_dir(ctx, path).await?;
224            }
225            _ => return Err(parse_error(resp)),
226        }
227        Ok(())
228    }

231mod error {
232    use std::fmt::Debug;
233
234    use http::Response;
235    use http::StatusCode;
236    use serde::Deserialize;
237
238    use opendal_core::raw::*;
239    use opendal_core::*;
240
241    /// DbfsError is the error returned by DBFS service.
242    #[derive(Default, Deserialize)]
243    struct DbfsError {
244        error_code: String,
245        message: String,
246    }
247
248    impl Debug for DbfsError {
249        fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
250            f.debug_struct("DbfsError")
251                .field("error_code", &self.error_code)
252                // replace `\n` to ` ` for better reading.
253                .field("message", &self.message.replace('\n', " "))
254                .finish()
255        }
256    }
257
258    pub(crate) fn parse_error(resp: Response<Buffer>) -> Error {
259        let (parts, body) = resp.into_parts();
260        let bs = body.to_bytes();
261
262        let (kind, retryable) = match parts.status {
263            StatusCode::NOT_FOUND => (ErrorKind::NotFound, false),
264            StatusCode::UNAUTHORIZED | StatusCode::FORBIDDEN => {
265                (ErrorKind::PermissionDenied, false)
266            }
267            StatusCode::PRECONDITION_FAILED => (ErrorKind::ConditionNotMatch, false),
268            StatusCode::INTERNAL_SERVER_ERROR
269            | StatusCode::BAD_GATEWAY
270            | StatusCode::SERVICE_UNAVAILABLE
271            | StatusCode::GATEWAY_TIMEOUT => (ErrorKind::Unexpected, true),
272            _ => (ErrorKind::Unexpected, false),
273        };
274
275        let message = match serde_json::from_slice::<DbfsError>(&bs) {
276            Ok(dbfs_error) => format!("{:?}", dbfs_error.message),
277            Err(_) => String::from_utf8_lossy(&bs).into_owned(),
278        };
279
280        let mut err = Error::new(kind, message);
281
282        err = with_error_response_context(err, parts);
283
284        if retryable {
285            err = err.set_temporary();
286        }
287
288        err
289    }
290}
