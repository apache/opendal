31const EXT4_SUPER_MAGIC: u32 = 0xef53;
32const XFS_SUPER_MAGIC: u32 = 0x5846_5342;
33const WRITE_BATCH_SIZE: usize = 1024 * 1024;
34
35#[derive(Clone, Copy)]
36pub(super) struct DirectIoAlignment {
37    mem_align: usize,
38    pub(super) offset_align: usize,
39}
40
41impl DirectIoAlignment {
42    pub(super) fn open(
43        path: &Path,
44        options: OpenOptions,
45        operation: ServiceOperation,
46    ) -> Result<(File, Self)> {
47        let file = Self::open_raw(path, options, operation)?;
48        if file.metadata().map_err(new_std_io_error)?.is_dir() {
49            return Err(
50                Error::new(OpendalErrorKind::IsADirectory, "read path is a directory")
51                    .with_context("service_operation", operation.0),
52            );
53        }
54        Self::discover_or_cleanup(path, file, operation, false)
55    }
56
57    pub(super) fn open_for_write(
58        path: &Path,
59        operation: ServiceOperation,
60        create_new: bool,
61    ) -> Result<(File, Self)> {
62        if create_new {
63            let mut options = OpenOptions::new();
64            options.write(true).create_new(true);
65            let file = Self::open_raw(path, options, operation)?;
66            return Self::discover_or_cleanup(path, file, operation, true);
67        }
68
69        for _ in 0..8 {
70            // Try opening an existing file first without O_CREAT so a failed
71            // O_DIRECT check never removes or truncates a pre-existing file.
72            let mut existing = OpenOptions::new();
73            existing.write(true);
74            match Self::open_raw(path, existing, operation) {
75                Ok(file) => {
76                    let (file, alignment) =
77                        Self::discover_or_cleanup(path, file, operation, false)?;
78                    file.set_len(0).map_err(new_std_io_error)?;
79                    return Ok((file, alignment));
80                }
81                Err(err) if err.kind() == OpendalErrorKind::NotFound => {}
82                Err(err) => return Err(err),
83            }
84
85            // The target did not exist; create it exclusively so we know we
86            // created the file and can remove it if direct I/O discovery fails.
87            let mut create = OpenOptions::new();
88            create.write(true).create_new(true);
89            match Self::open_raw(path, create, operation) {
90                Ok(file) => return Self::discover_or_cleanup(path, file, operation, true),
91                Err(err) if err.kind() == OpendalErrorKind::ConditionNotMatch => {
92                    // O_CREAT|O_EXCL fails with EEXIST on a dangling symlink while
93                    // opening without O_CREAT fails with ENOENT.
94                    if std::fs::symlink_metadata(path)
95                        .is_ok_and(|meta| meta.file_type().is_symlink())
96                    {
97                        break;
98                    }
99                }
100                Err(err) => return Err(err),
101            }
102        }
103
104        let mut fallback = OpenOptions::new();
105        fallback.write(true).create(true);
106        let file = Self::open_raw(path, fallback, operation)?;
107        let (file, alignment) = Self::discover_or_cleanup(path, file, operation, false)?;
108        file.set_len(0).map_err(new_std_io_error)?;
109        Ok((file, alignment))
110    }
111
112    fn open_raw(
113        path: &Path,
114        mut options: OpenOptions,
115        operation: ServiceOperation,
116    ) -> Result<File> {
117        options
118            .custom_flags(OFlags::DIRECT.bits() as i32)
119            .open(path)
120            .map_err(|err| match err.raw_os_error() {
121                Some(code)
122                    if code == Errno::INVAL.raw_os_error()
123                        || code == Errno::OPNOTSUPP.raw_os_error() =>
124                {
125                    Error::new(OpendalErrorKind::Unsupported, "O_DIRECT open failed")
126                        .with_context("service_operation", operation.0)
127                        .set_source(err)
128                }
129                _ => parse_error(ErrorContext::new(operation), err),
130            })
131    }
132
133    fn discover_or_cleanup(
134        path: &Path,
135        file: File,
136        operation: ServiceOperation,
137        cleanup_on_error: bool,
138    ) -> Result<(File, Self)> {
139        match Self::discover(&file) {
140            Ok(alignment) => Ok((file, alignment)),
141            Err(err) => {
142                let err = err.with_context("service_operation", operation.0);
143                if cleanup_on_error {
144                    drop(file);
145                    match std::fs::remove_file(path) {
146                        Ok(()) => {}
147                        Err(cleanup) if cleanup.kind() == ErrorKind::NotFound => {}
148                        Err(cleanup) => {
149                            return Err(err.with_context("file_cleanup", cleanup.to_string()));
150                        }
151                    }
152                }
153                Err(err)
154            }
155        }
156    }
157
158    fn discover(file: &File) -> Result<Self> {
159        match statx(file, "", AtFlags::EMPTY_PATH, StatxFlags::DIOALIGN) {
160            Ok(stat) if stat.stx_mask & StatxFlags::DIOALIGN.bits() != 0 => {
161                if stat.stx_dio_mem_align == 0 || stat.stx_dio_offset_align == 0 {
162                    return Err(Error::new(
163                        OpendalErrorKind::Unsupported,
164                        "file does not support direct I/O",
165                    ));
166                }
167                return Ok(Self {
168                    mem_align: stat.stx_dio_mem_align as usize,
169                    offset_align: stat.stx_dio_offset_align as usize,
170                });
171            }
172            Ok(_) | Err(Errno::NOSYS) => {}
173            Err(err) => return Err(new_std_io_error(err.into())),
174        }
175
176        // Some filesystems (including tmpfs) accept O_DIRECT while still using the
177        // page cache. Only fall back to legacy alignment on known filesystems.
178        let fs_type = fstatfs(file)
179            .map_err(|err| new_std_io_error(err.into()))?
180            .f_type as u32;
181        match fs_type {
182            EXT4_SUPER_MAGIC | XFS_SUPER_MAGIC => Ok(Self {
183                mem_align: 4096,
184                offset_align: 4096,
185            }),
186            _ => Err(Error::new(
187                OpendalErrorKind::Unsupported,
188                "filesystem does not report direct I/O alignment",
189            )
190            .with_context("filesystem_type", format!("{fs_type:#x}"))),
191        }
192    }

194    pub(super) fn read_at(self, file: &File, offset: u64, size: usize) -> Result<Buffer> {
195        let prefix = (offset % self.offset_align as u64) as usize;
196        let len = prefix
197            .checked_add(size)
198            .and_then(|size| {
199                size.div_ceil(self.offset_align)
200                    .checked_mul(self.offset_align)
201            })
202            .ok_or_else(|| Error::new(OpendalErrorKind::Unexpected, "direct read size overflow"))?;
203        let mut aligned = AlignedBuffer::new(len, self.mem_align)?;
204        let read_offset = offset - prefix as u64;
205        let read = loop {
206            match file.read_at(aligned.as_mut_slice(), read_offset) {
207                Err(err) if err.kind() == ErrorKind::Interrupted => continue,
208                result => break result.map_err(new_std_io_error)?,
209            }
210        };
211
212        let available = size.min(read.saturating_sub(prefix));
213        if available == 0 {
214            return Ok(Buffer::new());
215        }
216        let start = aligned.start + prefix;
217        Ok(Bytes::from(aligned.buf)
218            .slice(start..start + available)
219            .into())
220    }

222    pub(super) fn write_all_at(self, file: &File, mut offset: u64, mut buf: Buffer) -> Result<()> {
223        if buf.is_empty() {
224            return Ok(());
225        }
226        let blocks = (WRITE_BATCH_SIZE / self.offset_align).max(1);
227        let batch_size = blocks.min(buf.len().div_ceil(self.offset_align)) * self.offset_align;
228        let mut aligned = AlignedBuffer::new(batch_size, self.mem_align)?;
229        while !buf.is_empty() {
230            let len = buf.len().min(batch_size);
231            let aligned_len = len.div_ceil(self.offset_align) * self.offset_align;
232            let batch = &mut aligned.as_mut_slice()[..aligned_len];
233            buf.copy_to_slice(&mut batch[..len]);
234            batch[len..].fill(0);
235
236            let mut remaining = batch.len();
237            while remaining > 0 {
238                let written = match file.write_at(&batch[..remaining], offset) {
239                    Err(err) if err.kind() == ErrorKind::Interrupted => continue,
240                    result => result.map_err(new_std_io_error)?,
241                };
242                if written == 0 {
243                    return Err(new_std_io_error(io::Error::from(ErrorKind::WriteZero)));
244                }
245                if written % self.offset_align != 0 {
246                    return Err(Error::new(
247                        OpendalErrorKind::Unexpected,
248                        "unaligned short direct I/O write",
249                    ));
250                }
251                if written < remaining {
252                    // Keep the next syscall's buffer address aligned even when the
253                    // memory alignment exceeds the offset alignment.
254                    batch.copy_within(written..remaining, 0);
255                }
256                offset += written as u64;
257                remaining -= written;
258            }
259        }
260        Ok(())
261    }
262}
263
264struct AlignedBuffer {
265    buf: Vec<u8>,
266    start: usize,
267    len: usize,
268}
269
270impl AlignedBuffer {
271    fn new(len: usize, align: usize) -> Result<Self> {
272        let capacity = len.checked_add(align - 1).ok_or_else(|| {
273            Error::new(OpendalErrorKind::Unexpected, "aligned buffer size overflow")
274        })?;
275        let mut buf = Vec::new();
276        buf.try_reserve_exact(capacity).map_err(|err| {
277            Error::new(OpendalErrorKind::Unexpected, "allocate aligned buffer").set_source(err)
278        })?;
279        buf.resize(capacity, 0);
280        let misalignment = buf.as_ptr() as usize % align;
281        let start = if misalignment == 0 {
282            0
283        } else {
284            align - misalignment
285        };
286        Ok(Self { buf, start, len })
287    }
288
289    fn as_mut_slice(&mut self) -> &mut [u8] {
290        &mut self.buf[self.start..self.start + self.len]
291    }
