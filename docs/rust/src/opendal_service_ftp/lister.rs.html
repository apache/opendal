26/// Returns `true` for POSIX `LIST` summary lines such as `"total 0"` or `"total 8"`.
27///
28/// `suppaftp`'s `File::from_str` falls back to `ListParser::parse_mlst`, which
29/// treats arbitrary strings without `=` as filenames, turning `"total 8"` into a
30/// phantom 0-byte file entry.
31fn is_posix_total_line(line: &str) -> bool {
32    let trimmed = line.trim();
33    let Some(rest) = trimmed.strip_prefix("total") else {
34        return false;
35    };
36    let Some(first) = rest.chars().next() else {
37        return false;
38    };
39    if !first.is_ascii_whitespace() {
40        return false;
41    }
42    let count = rest.trim_start();
43    !count.is_empty() && count.bytes().all(|b| b.is_ascii_digit())
44}
45
46/// Parse a single line returned by FTP `LIST`, returning `Ok(None)` for blank
47/// lines, POSIX `"total N"` summary headers, and `"."` / `".."` directory entries.
48pub(crate) fn parse_list_line(line: &str) -> Result<Option<File>> {
49    if line.trim().is_empty() || is_posix_total_line(line) {
50        return Ok(None);
51    }
52
53    let file = File::from_str(line)
54        .map_err(|e| Error::new(ErrorKind::Unexpected, "parse file from response").set_source(e))?;
55
56    if matches!(file.name(), "" | "." | "..") {
57        return Ok(None);
58    }
59
60    Ok(Some(file))
61}

63pub struct FtpLister {
64    path: String,
65    file_iter: IntoIter<String>,
66    dir_entry: Option<oio::Entry>,
67}
68
69impl FtpLister {
70    pub fn new(path: &str, files: Vec<String>) -> Self {
71        Self {
72            path: path.to_string(),
73            file_iter: files.into_iter(),
74            dir_entry: None,
75        }
76    }
77
78    /// Return the listed directory itself before its children.
79    pub fn with_dir_entry(mut self) -> Self {
80        self.dir_entry = Some(oio::Entry::new(&self.path, MetadataBuilder::dir().build()));
81        self
82    }
83}
84
85impl oio::List for FtpLister {
86    async fn next(&mut self) -> Result<Option<oio::Entry>> {
87        if let Some(entry) = self.dir_entry.take() {
88            return Ok(Some(entry));
89        }
90
91        let de = loop {
92            let Some(file_str) = self.file_iter.next() else {
93                return Ok(None);
94            };
95            if let Some(de) = parse_list_line(file_str.as_str())? {
96                break de;
97            }
98        };
99
100        let path = self.path.to_string() + de.name();
101
102        let mut meta = if de.is_file() {
103            MetadataBuilder::file(de.size() as u64)
104        } else if de.is_directory() {
105            MetadataBuilder::dir()
106        } else {
107            MetadataBuilder::unknown()
108        };
109        meta.last_modified(Timestamp::try_from(de.modified())?);
110
111        let entry = if de.is_file() {
112            oio::Entry::new(&path, meta.build())
113        } else if de.is_directory() {
114            oio::Entry::new(&format!("{}/", path), meta.build())
115        } else {
116            oio::Entry::new(&path, meta.build())
117        };
118
119        Ok(Some(entry))
120    }
121}

123#[cfg(test)]
124mod tests {
125    use super::*;
126    use opendal_core::raw::oio::List as _;

128    #[tokio::test]
129    async fn test_lister_returns_dir_entry_first() {
130        let lines = vec!["-rw-r--r-- 1 user group 12 Nov  5  2018 file".to_string()];
131        let mut lister = FtpLister::new("dir/", lines).with_dir_entry();
132
133        let first = lister.next().await.unwrap().unwrap();
134        assert_eq!(first.path(), "dir/");
135        assert_eq!(first.metadata().mode(), EntryMode::DIR);
136        let second = lister.next().await.unwrap().unwrap();
137        assert_eq!(second.path(), "dir/file");
138        assert!(lister.next().await.unwrap().is_none());
139
140        // An empty directory still returns itself.
141        let mut lister = FtpLister::new("empty/", vec![]).with_dir_entry();
142        assert_eq!(lister.next().await.unwrap().unwrap().path(), "empty/");
143        assert!(lister.next().await.unwrap().is_none());
144    }

146    #[tokio::test]
147    async fn test_lister_filters_dots_total_header_and_blank_lines() {
148        let lines = vec![
149            "total 16".to_string(),
150            "   ".to_string(),
151            "drwxr-xr-x 2 user group 4096 Nov  5  2018 .".to_string(),
152            "drwxr-xr-x 4 user group 4096 Nov  5  2018 ..".to_string(),
153            "drwxr-xr-x 2 user group 4096 Nov  5  2018 subdir".to_string(),
154            "-rw-r--r-- 1 user group   12 Nov  5  2018 .hidden".to_string(),
155            "-rw-r--r-- 1 user group   42 Nov  5  2018 total 8".to_string(),
156            "10-19-20  03:19PM       <DIR>          .".to_string(),
157            "10-19-20  03:19PM       <DIR>          ..".to_string(),
158            "10-19-20  03:19PM       <DIR>          dosdir".to_string(),
159            "04-08-14  03:09PM                  128 dos.txt".to_string(),
160        ];
161
162        let mut lister = FtpLister::new("dir/", lines);
163        let mut entries = Vec::new();
164        while let Some(entry) = lister.next().await.unwrap() {
165            entries.push((
166                entry.path().to_string(),
167                entry.metadata().mode(),
168                entry.metadata().content_length(),
169            ));
170        }
171
172        assert_eq!(
173            entries,
174            vec![
175                ("dir/subdir/".to_string(), EntryMode::DIR, 0),
176                ("dir/.hidden".to_string(), EntryMode::FILE, 12),
177                ("dir/total 8".to_string(), EntryMode::FILE, 42),
178                ("dir/dosdir/".to_string(), EntryMode::DIR, 0),
179                ("dir/dos.txt".to_string(), EntryMode::FILE, 128),
180            ]
181        );
182    }

184    #[test]
185    fn test_is_posix_total_line() {
186        assert!(is_posix_total_line("total 0"));
187        assert!(is_posix_total_line("total 8"));
188        assert!(is_posix_total_line("  total   1234  "));
189        assert!(!is_posix_total_line("total"));
190        assert!(!is_posix_total_line("total abc"));
191        assert!(!is_posix_total_line("total8"));
192        assert!(!is_posix_total_line(
193            "-rw-r--r-- 1 user group 42 Nov  5  2018 total 8"
194        ));
195    }
