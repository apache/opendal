85impl GcsCore {
86    fn signer(&self, ctx: &OperationContext) -> Signer<Credential> {
87        self.signer.clone().with_context(
88            self.sign_ctx
89                .clone()
90                .with_http_send(ctx.http_transport().clone()),
91        )
92    }
93
94    pub async fn sign<T>(&self, ctx: &OperationContext, req: Request<T>) -> Result<Request<T>> {
95        if self.skip_signature {
96            return Ok(req);
97        }
98
99        let (mut parts, body) = req.into_parts();
100
101        self.signer(ctx)
102            .sign(&mut parts, None)
103            .await
104            .map_err(|err| new_request_sign_error(err.into()))?;
105
106        // Always remove host header, let users' client to set it based on
107        // HTTP version.
108        //
109        // As discussed in <https://github.com/seanmonstar/reqwest/issues/1809>,
110        // google server could send RST_STREAM of PROTOCOL_ERROR if our
111        // request contains host header.
112        parts.headers.remove(HOST);
113
114        Ok(Request::from_parts(parts, body))
115    }
116
117    pub async fn sign_query<T>(
118        &self,
119        ctx: &OperationContext,
120        req: Request<T>,
121        duration: Duration,
122    ) -> Result<Request<T>> {
123        if self.skip_signature {
124            return Ok(req);
125        }
126
127        let (mut parts, body) = req.into_parts();
128
129        self.signer(ctx)
130            .sign(&mut parts, Some(duration))
131            .await
132            .map_err(|err| new_request_sign_error(err.into()))?;
133
134        // Always remove host header, let users' client to set it based on
135        // HTTP version.
136        //
137        // As discussed in <https://github.com/seanmonstar/reqwest/issues/1809>,
138        // google server could send RST_STREAM of PROTOCOL_ERROR if our
139        // request contains host header.
140        parts.headers.remove(HOST);
141
142        Ok(Request::from_parts(parts, body))
143    }

151        ctx.http_transport().send(req).await
152    }

156    pub fn gcs_get_object_request(
157        &self,
158        path: &str,
159        range: BytesRange,
160        args: &OpRead,
161    ) -> Result<Request<Buffer>> {
162        let p = build_abs_path(&self.root, path);
163
164        let url = format!(
165            "{}/storage/v1/b/{}/o/{}?alt=media",
166            self.endpoint,
167            self.bucket,
168            gcs_percent_encode_path(&p)
169        );
170
171        let mut req = Request::get(&url);
172
173        if let Some(if_match) = args.if_match() {
174            req = req.header(IF_MATCH, if_match);
175        }
176        if let Some(if_none_match) = args.if_none_match() {
177            req = req.header(IF_NONE_MATCH, if_none_match);
178        }
179        if !range.is_full() {
180            req = req.header(http::header::RANGE, range.to_header());
181        }
182
183        let req = req
184            .extension(Operation::Read)
185            .extension(ServiceOperation("GetObject"));
186
187        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
188
189        Ok(req)
190    }

245    pub fn gcs_insert_object_request(
246        &self,
247        path: &str,
248        size: Option<u64>,
249        op: &OpWrite,
250        body: Buffer,
251    ) -> Result<Request<Buffer>> {
252        let p = build_abs_path(&self.root, path);
253
254        let request_metadata = InsertRequestMetadata {
255            storage_class: self.default_storage_class.as_deref(),
256            cache_control: op.cache_control(),
257            content_type: op.content_type(),
258            content_encoding: op.content_encoding(),
259            metadata: op.user_metadata(),
260        };
261
262        let mut url = format!(
263            "{}/upload/storage/v1/b/{}/o?uploadType={}&name={}",
264            self.endpoint,
265            self.bucket,
266            if request_metadata.is_empty() {
267                "media"
268            } else {
269                "multipart"
270            },
271            gcs_percent_encode_path(&p)
272        );
273
274        if let Some(acl) = &self.predefined_acl {
275            write!(&mut url, "&predefinedAcl={acl}").unwrap();
276        }
277
278        // Makes the operation conditional on whether the object's current generation
279        // matches the given value. Setting to 0 makes the operation succeed only if
280        // there are no live versions of the object.
281        if op.if_not_exists() {
282            write!(&mut url, "&ifGenerationMatch=0").unwrap();
283        }
284
285        let mut req = Request::post(&url);
286
287        req = req.header(CONTENT_LENGTH, size.unwrap_or_default());
288
289        if request_metadata.is_empty() {
290            let req = req
291                .extension(Operation::Write)
292                .extension(ServiceOperation("InsertObject"));
293            // If the metadata is empty, we do not set any `Content-Type` header,
294            // since if we had it in the `op.content_type()`, it would be already set in the
295            // `multipart` metadata body and this branch won't be executed.
296            let req = req.body(body).map_err(new_request_build_error)?;
297            Ok(req)
298        } else {
299            let mut multipart = Multipart::new();
300            let metadata_part = RelatedPart::new()
301                .header(
302                    CONTENT_TYPE,
303                    "application/json; charset=UTF-8".parse().unwrap(),
304                )
305                .content(
306                    serde_json::to_vec(&request_metadata)
307                        .expect("metadata serialization should succeed"),
308                );
309            multipart = multipart.part(metadata_part);
310
311            // Content-Type must be set, even if it is set in the metadata part
312            let content_type = op
313                .content_type()
314                .unwrap_or("application/octet-stream")
315                .parse()
316                .expect("Failed to parse content-type");
317            let media_part = RelatedPart::new()
318                .header(CONTENT_TYPE, content_type)
319                .content(body);
320            multipart = multipart.part(media_part);
321
322            let req = multipart.apply(
323                Request::post(url)
324                    .extension(Operation::Write)
325                    .extension(ServiceOperation("InsertObject")),
326            )?;
327
328            Ok(req)
329        }
330    }

332    // It's for presign operation. Gcs only supports query sign over XML API.
333    pub fn gcs_insert_object_xml_request(
334        &self,
335        path: &str,
336        args: &OpWrite,
337        body: Buffer,
338    ) -> Result<Request<Buffer>> {
339        let p = build_abs_path(&self.root, path);
340
341        let url = format!("{}/{}/{}", self.endpoint, self.bucket, p);
342
343        let mut req = Request::put(&url);
344
345        if let Some(user_metadata) = args.user_metadata() {
346            for (key, value) in user_metadata {
347                req = req.header(format!("{X_GOOG_META_PREFIX}{key}"), value)
348            }
349        }
350
351        if let Some(content_type) = args.content_type() {
352            req = req.header(CONTENT_TYPE, content_type);
353        }
354
355        if let Some(content_encoding) = args.content_encoding() {
356            req = req.header(CONTENT_ENCODING, content_encoding);
357        }
358
359        if let Some(acl) = &self.predefined_acl {
360            if let Some(predefined_acl_in_xml_spec) = predefined_acl_to_xml_header(acl) {
361                req = req.header(X_GOOG_ACL, predefined_acl_in_xml_spec);
362            } else {
363                log::warn!("Unrecognized predefined_acl. Ignoring");
364            }
365        }
366
367        if let Some(storage_class) = &self.default_storage_class {
368            req = req.header(X_GOOG_STORAGE_CLASS, storage_class);
369        }
370
371        let req = req
372            .extension(Operation::Write)
373            .extension(ServiceOperation("InsertObject"));
374
375        let req = req.body(body).map_err(new_request_build_error)?;
376
377        Ok(req)
378    }

479    pub async fn gcs_delete_objects(
480        &self,
481        ctx: &OperationContext,
482        paths: Vec<String>,
483    ) -> Result<Response<Buffer>> {
484        let uri = format!("{}/batch/storage/v1", self.endpoint);
485
486        let mut multipart = Multipart::new();
487
488        for (idx, path) in paths.iter().enumerate() {
489            let req = self.gcs_delete_object_request(path)?;
490
491            multipart = multipart.part(
492                MixedPart::from_request(req).part_header("content-id".parse().unwrap(), idx.into()),
493            );
494        }
495
496        let req = Request::post(uri)
497            .extension(Operation::Delete)
498            .extension(ServiceOperation("BatchDeleteObjects"));
499        let req = multipart.apply(req)?;
500
501        let req = self.sign(ctx, req).await?;
502        self.send(ctx, req).await
503    }

505    pub async fn gcs_rewrite_object(
506        &self,
507        ctx: &OperationContext,
508        from: &str,
509        to: &str,
510        args: &OpCopy,
511        max_bytes_rewritten_per_call: Option<usize>,
512        rewrite_token: Option<&str>,
513    ) -> Result<Response<Buffer>> {
514        let source = build_abs_path(&self.root, from);
515        let dest = build_abs_path(&self.root, to);
516
517        let url = format!(
518            "{}/storage/v1/b/{}/o/{}/rewriteTo/b/{}/o/{}",
519            self.endpoint,
520            self.bucket,
521            gcs_percent_encode_path(&source),
522            self.bucket,
523            gcs_percent_encode_path(&dest)
524        );
525
526        let mut url = QueryPairsWriter::new(&url);
527
528        if let Some(version) = args.source_version() {
529            url = url.push("sourceGeneration", &gcs_percent_encode_path(version));
530        }
531        if args.if_not_exists() {
532            url = url.push("ifGenerationMatch", "0");
533        }
534        if let Some(max_bytes) = max_bytes_rewritten_per_call {
535            url = url.push("maxBytesRewrittenPerCall", &max_bytes.to_string());
536        }
537        if let Some(token) = rewrite_token {
538            url = url.push("rewriteToken", &gcs_percent_encode_path(token));
539        }
540
541        let req = Request::post(url.finish())
542            .header(CONTENT_LENGTH, 0)
543            .extension(Operation::Copy)
544            .extension(ServiceOperation("RewriteObject"))
545            .body(Buffer::new())
546            .map_err(new_request_build_error)?;
547
548        let req = self.sign(ctx, req).await?;
549        self.send(ctx, req).await
550    }

552    pub async fn gcs_list_objects(
553        &self,
554        ctx: &OperationContext,
555        path: &str,
556        page_token: &str,
557        delimiter: &str,
558        limit: Option<usize>,
559        start_after: Option<String>,
560    ) -> Result<Response<Buffer>> {
561        let p = build_abs_path(&self.root, path);
562
563        let url = format!("{}/storage/v1/b/{}/o", self.endpoint, self.bucket,);
564
565        let mut url = QueryPairsWriter::new(&url);
566        url = url.push("prefix", &gcs_percent_encode_path(&p));
567
568        if !delimiter.is_empty() {
569            url = url.push("delimiter", delimiter);
570        }
571        if let Some(limit) = limit {
572            url = url.push("maxResults", &limit.to_string());
573        }
574        if let Some(start_after) = start_after {
575            let start_after = build_abs_path(&self.root, &start_after);
576            url = url.push("startOffset", &gcs_percent_encode_path(&start_after));
577        }
578
579        if !page_token.is_empty() {
580            // NOTE:
581            //
582            // GCS uses pageToken in request and nextPageToken in response
583            //
584            // Don't know how will those tokens be like so this part are copied
585            // directly from AWS S3 service.
586            url = url.push("pageToken", &gcs_percent_encode_path(page_token));
587        }
588
589        let req = Request::get(url.finish())
590            .extension(Operation::List)
591            .extension(ServiceOperation("ListObjects"))
592            .body(Buffer::new())
593            .map_err(new_request_build_error)?;
594
595        let req = self.sign(ctx, req).await?;
596
597        self.send(ctx, req).await
598    }

600    pub async fn gcs_initiate_multipart_upload(
601        &self,
602        ctx: &OperationContext,
603        path: &str,
604        op: &OpWrite,
605    ) -> Result<Response<Buffer>> {
606        let p = build_abs_path(&self.root, path);
607
608        let url = format!(
609            "{}/{}/{}?uploads",
610            self.endpoint,
611            self.bucket,
612            gcs_percent_encode_path(&p)
613        );
614
615        let mut builder = Request::post(&url)
616            .header(CONTENT_LENGTH, 0)
617            .extension(Operation::Write)
618            .extension(ServiceOperation("CreateMultipartUpload"));
619
620        if let Some(header_val) = op.content_disposition() {
621            builder = builder.header(CONTENT_DISPOSITION, header_val);
622        }
623
624        if let Some(header_val) = op.content_encoding() {
625            builder = builder.header(CONTENT_ENCODING, header_val);
626        }
627
628        if let Some(header_val) = op.content_type() {
629            builder = builder.header(CONTENT_TYPE, header_val);
630        }
631
632        if let Some(header_val) = op.cache_control() {
633            builder = builder.header(CACHE_CONTROL, header_val);
634        }
635
636        if let Some(metadata) = op.user_metadata() {
637            for (k, v) in metadata {
638                builder = builder.header(&format!("x-goog-meta-{k}"), v);
639            }
640        }
641
642        if let Some(acl) = self.predefined_acl.as_ref() {
643            if let Some(predefined_acl_in_xml_spec) = predefined_acl_to_xml_header(acl) {
644                builder = builder.header(X_GOOG_ACL, predefined_acl_in_xml_spec);
645            } else {
646                log::warn!("Unrecognized predefined_acl. Ignoring");
647            }
648        }
649
650        let req = builder
651            .body(Buffer::new())
652            .map_err(new_request_build_error)?;
653
654        let req = self.sign(ctx, req).await?;
655        self.send(ctx, req).await
656    }

705            gcs_percent_encode_path(&p),
706            gcs_percent_encode_path(upload_id)
707        );
708
709        let req = Request::post(&url);
710
711        let content = quick_xml::se::to_string(&CompleteMultipartUploadRequest { part: parts })
712            .map_err(new_xml_serialize_error)?;
713        // Make sure content length has been set to avoid post with chunked encoding.
714        let req = req.header(CONTENT_LENGTH, content.len());
715        // Set content-type to `application/xml` to avoid mixed with form post.
716        let req = req.header(CONTENT_TYPE, "application/xml");
717
718        let req = req
719            .extension(Operation::Write)
720            .extension(ServiceOperation("CompleteMultipartUpload"));
721
722        let req = req
723            .body(Buffer::from(Bytes::from(content)))
724            .map_err(new_request_build_error)?;
725
726        let req = self.sign(ctx, req).await?;
727        self.send(ctx, req).await
728    }

730    pub async fn gcs_abort_multipart_upload(
731        &self,
732        ctx: &OperationContext,
733        path: &str,
734        upload_id: &str,
735    ) -> Result<Response<Buffer>> {
736        let p = build_abs_path(&self.root, path);
737
738        let url = format!(
739            "{}/{}/{}?uploadId={}",
740            self.endpoint,
741            self.bucket,
742            gcs_percent_encode_path(&p),
743            gcs_percent_encode_path(upload_id)
744        );
745
746        let req = Request::delete(&url)
747            .extension(Operation::Write)
748            .extension(ServiceOperation("AbortMultipartUpload"))
749            .body(Buffer::new())
750            .map_err(new_request_build_error)?;
751        let req = self.sign(ctx, req).await?;
752        self.send(ctx, req).await
753    }

763impl GetObjectJsonResponse {
764    fn into_metadata(self, path: &str) -> Result<Metadata> {
765        let mut m = Metadata::new(EntryMode::from_path(path));
766
767        m.set_etag(&self.etag);
768        m.set_content_md5(&self.md5_hash);
769
770        let size = self
771            .size
772            .parse::<u64>()
773            .map_err(|e| Error::new(ErrorKind::Unexpected, "parse u64").set_source(e))?;
774        m.set_content_length(size);
775        if !self.content_type.is_empty() {
776            m.set_content_type(&self.content_type);
777        }
778
779        if !self.content_encoding.is_empty() {
780            m.set_content_encoding(&self.content_encoding);
781        }
782
783        if !self.cache_control.is_empty() {
784            m.set_cache_control(&self.cache_control);
785        }
786
787        if !self.content_disposition.is_empty() {
788            m.set_content_disposition(&self.content_disposition);
789        }
790
791        if !self.generation.is_empty() {
792            m.set_version(&self.generation);
793        }
794
795        if !self.updated.is_empty() {
796            m.set_last_modified(self.updated.parse::<Timestamp>()?);
797        }
798
799        if !self.metadata.is_empty() {
800            m = m.with_user_metadata(self.metadata);
801        }
802
803        Ok(m)
804    }

1220        code: usize,
1221        message: String,
1222        errors: Vec<GcsErrorDetail>,

1235    /// Parse error response into Error.
1236    pub(crate) fn parse_error(resp: Response<Buffer>) -> Error {
1237        let (parts, body) = resp.into_parts();
1238        let bs = body.to_bytes();
1239
1240        let (kind, retryable) = match parts.status {
1241            StatusCode::NOT_FOUND => (ErrorKind::NotFound, false),
1242            StatusCode::FORBIDDEN => (ErrorKind::PermissionDenied, false),
1243            StatusCode::PRECONDITION_FAILED | StatusCode::NOT_MODIFIED => {
1244                (ErrorKind::ConditionNotMatch, false)
1245            }
1246            StatusCode::TOO_MANY_REQUESTS => (ErrorKind::RateLimited, true),
1247            StatusCode::INTERNAL_SERVER_ERROR
1248            | StatusCode::BAD_GATEWAY
1249            | StatusCode::SERVICE_UNAVAILABLE
1250            | StatusCode::GATEWAY_TIMEOUT => (ErrorKind::Unexpected, true),
1251            _ => (ErrorKind::Unexpected, false),
1252        };
1253
1254        let message = match de::from_slice::<GcsErrorResponse>(&bs) {
1255            Ok(gcs_err) => format!("{gcs_err:?}"),
1256            Err(_) => String::from_utf8_lossy(&bs).into_owned(),
1257        };
1258
1259        let mut err = Error::new(kind, message);
1260
1261        err = with_error_response_context(err, parts);
1262
1263        if retryable {
1264            err = err.set_temporary();
1265        }
1266
1267        err
1268    }

1317    /// PATH_ENCODE_SET is the encode set for http url path.
1318    ///
1319    /// This set follows [encodeURIComponent](https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/encodeURIComponent) which will encode all non-ASCII characters except `A-Z a-z 0-9 - _ . ! ~ * ' ( )`
1320    ///
1321    /// Following characters is allowed in GCS, check "https://cloud.google.com/storage/docs/request-endpoints#encoding" for details
1322    static GCS_PATH_ENCODE_SET: AsciiSet = NON_ALPHANUMERIC
1323        .remove(b'-')
1324        .remove(b'_')
1325        .remove(b'.')
1326        .remove(b'*');
1327
1328    /// gcs_percent_encode_path will do percent encoding for http encode path.
1329    ///
1330    /// Follows [encodeURIComponent](https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/encodeURIComponent)
1331    /// which will encode all non-ASCII characters except `A-Z a-z 0-9 - _ . *`
1332    ///
1333    /// GCS does not allow '/'s in paths, this should also be dealt with
1334    pub(crate) fn gcs_percent_encode_path(path: &str) -> String {
1335        utf8_percent_encode(path, &GCS_PATH_ENCODE_SET).to_string()
1336    }
