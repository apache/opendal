66    /// Set the working directory root.

77
78    /// Set the gRPC endpoint.
79    pub fn endpoint(mut self, endpoint: &str) -> Self {
80        self.config.endpoint = (!endpoint.is_empty()).then(|| endpoint.to_string());
81        self
82    }
83
84    /// Set the Google OAuth 2.0 scope.
85    pub fn scope(mut self, scope: &str) -> Self {
86        self.config.scope = (!scope.is_empty()).then(|| scope.to_string());
87        self
88    }

90    /// Set the service account used by the GCE metadata server.
91    pub fn service_account(mut self, service_account: &str) -> Self {

96
97    /// Set a base64-encoded service account credential.
98    pub fn credential(mut self, credential: &str) -> Self {
99        self.config.credential = (!credential.is_empty()).then(|| credential.to_string());
100        self
101    }
102
103    /// Set the path to a service account credential file.
104    pub fn credential_path(mut self, path: &str) -> Self {
105        self.config.credential_path = (!path.is_empty()).then(|| path.to_string());
106        self
107    }
108
109    /// Set a custom Google credential provider.
110    ///
111    /// Implement the provider against [`opendal_core::reqsign_core`] and
112    /// [`crate::reqsign_google`] so that its types match the ones this service is
113    /// built with.
114    pub fn credential_provider(
115        mut self,
116        provider: impl ProvideCredential<Credential = Credential> + 'static,
117    ) -> Self {
118        self.credential_provider_chain = Some(
119            self.credential_provider_chain
120                .unwrap_or_default()
121                .push_front(provider),
122        );
123        self
124    }
125
126    /// Set a custom Google credential provider chain.
127    ///
128    /// Build the chain from [`opendal_core::reqsign_core`] and
129    /// [`crate::reqsign_google`] so that its types match the ones this service is
130    /// built with.
131    pub fn credential_provider_chain(mut self, chain: ProvideCredentialChain<Credential>) -> Self {
132        self.credential_provider_chain = Some(chain);
133        self
134    }
135
136    /// Set an OAuth 2.0 access token.
137    pub fn token(mut self, token: String) -> Self {
138        self.config.token = Some(token);
139        self
140    }
141
142    /// Disable the GCE metadata credential provider.
143    pub fn disable_vm_metadata(mut self) -> Self {
144        self.config.disable_vm_metadata = true;
145        self
146    }
147
148    /// Disable environment and well-known credential loading.
149    pub fn disable_config_load(mut self) -> Self {
150        self.config.disable_config_load = true;
151        self
152    }
153
154    /// Send requests without authentication.
155    pub fn skip_signature(mut self) -> Self {
156        self.config.skip_signature = true;
157        self
158    }

161impl GcsGrpcBuilder {
162    fn build_backend(self) -> Result<GcsGrpcBackend> {
163        debug!("backend build started: {self:?}");
164        let root = normalize_root(&self.config.root.unwrap_or_default());
165        if self.config.bucket.is_empty() {
166            return Err(
167                Error::new(ErrorKind::ConfigInvalid, "The bucket is misconfigured")
168                    .with_operation("Builder::build")
169                    .with_context("service", GCS_GRPC_SCHEME),
170            );
171        }
172
173        let endpoint = self
174            .config
175            .endpoint
176            .clone()
177            .unwrap_or_else(|| DEFAULT_GCS_GRPC_ENDPOINT.to_string());
178        let channel_endpoint = build_endpoint(&endpoint)?;
179        let scope = self
180            .config
181            .scope
182            .clone()
183            .unwrap_or_else(|| DEFAULT_GCS_SCOPE.to_string());
184
185        let ctx = Context::new().with_file_read(TokioFileRead).with_env(OsEnv);
186        #[cfg(not(target_arch = "wasm32"))]
187        let ctx = ctx.with_command_execute(TokioCommandExecute);
188
189        let mut default_credential = DefaultCredentialProvider::builder();
190        if self.config.disable_config_load {
191            default_credential = default_credential.no_env().no_well_known();
192        } else {
193            default_credential = default_credential
194                .env(EnvCredentialProvider::new().with_scope(&scope))
195                .well_known(WellKnownCredentialProvider::new().with_scope(&scope));
196        }
197        if self.config.disable_vm_metadata {
198            default_credential = default_credential.no_vm_metadata();
199        } else {
200            let mut vm_metadata = VmMetadataCredentialProvider::new().with_scope(&scope);
201            if let Some(service_account) = self.config.service_account.as_deref() {
202                vm_metadata = vm_metadata.with_service_account(service_account);
203            }
204            default_credential = default_credential.vm_metadata(vm_metadata);
205        }
206        let mut credential_chain = ProvideCredentialChain::new().push(default_credential.build());
207        if let Some(path) = self.config.credential_path.as_deref() {
208            credential_chain =
209                credential_chain.push_front(FileCredentialProvider::new(path).with_scope(&scope));
210        }
211        if let Some(content) = self.config.credential.as_deref() {
212            let provider = StaticCredentialProvider::from_base64(content).map_err(|err| {
213                Error::new(ErrorKind::ConfigInvalid, "invalid base64 credential")
214                    .with_operation("Builder::build")
215                    .with_context("service", GCS_GRPC_SCHEME)
216                    .set_source(err)
217            })?;
218            credential_chain = credential_chain.push_front(provider.with_scope(&scope));
219        }
220        if let Some(token) = self.config.token.as_deref() {
221            credential_chain = credential_chain.push_front(TokenCredentialProvider::new(token));
222        }
223        if let Some(custom) = self.credential_provider_chain {
224            credential_chain = credential_chain.push_front(custom);
225        }
226
227        let credential_provider = Arc::new(credential_chain);
228        let request_signer = Arc::new(RequestSigner::new("storage").with_scope(&scope));
229        let gcs_grpc_can_checksum = !self.config.skip_signature
230            && channel_endpoint.uri().scheme_str() == Some("https")
231            && channel_endpoint
232                .uri()
233                .host()
234                .is_some_and(|host| host.eq_ignore_ascii_case("storage.googleapis.com"))
235            && channel_endpoint.uri().port_u16().unwrap_or(443) == 443;
236        let capability = Capability {
237            stat: true,
238            stat_with_version: true,
239            read: true,
240            read_with_version: true,
241            read_with_suffix: true,
242            write: true,
243            write_can_empty: true,
244            write_can_multi: true,
245            write_with_content_type: true,
246            write_with_content_disposition: true,
247            write_with_content_encoding: true,
248            write_with_cache_control: true,
249            write_with_if_not_exists: true,
250            write_with_user_metadata: true,
251            write_with_checksum: gcs_grpc_can_checksum,
252            delete: true,
253            delete_with_version: true,
254            copy: true,
255            copy_with_if_not_exists: true,
256            copy_with_source_version: true,
257            list: true,
258            list_with_limit: true,
259            list_with_start_after: true,
260            list_with_recursive: true,
261            shared: true,
262            ..Default::default()
263        };
264        let bucket = self.config.bucket;
265        Ok(GcsGrpcBackend {
266            core: Arc::new(GcsGrpcCore {
267                info: ServiceInfo::new(GCS_GRPC_SCHEME, &root, &bucket),
268                capability,
269                endpoint,
270                bucket,
271                root,
272                channel_endpoint,
273                channel: Default::default(),
274                sign_ctx: ctx,
275                credential_provider,
276                request_signer,
277                signers: HttpTransportCache::new(),
278                skip_signature: self.config.skip_signature,
279            }),
280        })
281    }
282}

292fn build_endpoint(endpoint: &str) -> Result<Endpoint> {
293    let mut endpoint_builder = Endpoint::from_shared(endpoint.to_string()).map_err(|err| {
294        Error::new(ErrorKind::ConfigInvalid, "invalid GCS gRPC endpoint").set_source(err)
295    })?;
296    match endpoint_builder.uri().scheme() {
297        Some(scheme) if scheme == &http::uri::Scheme::HTTPS => {
298            endpoint_builder = endpoint_builder
299                .tls_config(ClientTlsConfig::new().with_webpki_roots())
300                .map_err(|err| {
301                    Error::new(
302                        ErrorKind::ConfigInvalid,
303                        "invalid GCS gRPC TLS configuration",
304                    )
305                    .set_source(err)
306                })?;
307        }
308        Some(scheme) if scheme == &http::uri::Scheme::HTTP => {}
309        _ => {
310            return Err(Error::new(
311                ErrorKind::ConfigInvalid,
312                "GCS gRPC endpoint must use http or https",
313            ));
314        }
315    }
316    Ok(endpoint_builder)
317}

347        Err(Error::new(

349            "operation is not supported",
350        ))

352
353    async fn stat(&self, ctx: &OperationContext, path: &str, args: OpStat) -> Result<RpStat> {
354        if path == "/" {
355            return Ok(RpStat::new(MetadataBuilder::dir().build()));
356        }
357        let object = self.core.object_name(path);
358        let request = GetObjectRequest {
359            bucket: self.core.bucket_resource(),
360            object,
361            generation: parse_generation(args.version())?,
362        };
363        let request = self
364            .core
365            .request(ctx, request, &[("bucket", &self.core.bucket_resource())])
366            .await?;
367        let response = self
368            .core
369            .client()
370            .get_object(request)
371            .await
372            .map_err(|status| {
373                parse_status(ErrorContext::new(ServiceOperation("GetObject")), status)
374            })?;
375        Ok(RpStat::new(parse_object(response.get_ref())))
376    }

501            let uri = req.uri().to_string();
502            let body = String::from_utf8_lossy(req.body()).into_owned();

505            let response_body = if uri.contains("/computeMetadata/v1/") {

513            } else {
514                panic!("unexpected request uri: {uri}");
515            };
516
517            Ok(http::Response::builder()
518                .status(http::StatusCode::OK)
519                .body(Bytes::from_static(response_body.as_bytes()))
520                .unwrap())
521        }

729    #[cfg(unix)]
730    struct TempScript {
731        dir: std::path::PathBuf,
732        script_path: std::path::PathBuf,
733        marker_path: std::path::PathBuf,
734    }
735
736    #[cfg(unix)]
737    impl TempScript {
738        fn new(body: &str) -> Self {
739            use std::os::unix::fs::PermissionsExt;
740            use std::sync::atomic::{AtomicU64, Ordering};
741
742            static COUNTER: AtomicU64 = AtomicU64::new(0);
743            let dir = std::env::temp_dir().join(format!(
744                "opendal-gcs-grpc-cmd-{}-{}",
745                std::process::id(),
746                COUNTER.fetch_add(1, Ordering::Relaxed)
747            ));
748            std::fs::create_dir_all(&dir).unwrap();
749            let marker_path = dir.join("executed");
750            let script_path = dir.join("cred.sh");
751            let script = format!("#!/bin/sh\ntouch \"{}\"\n{}\n", marker_path.display(), body);
752            std::fs::write(&script_path, script).unwrap();
753            std::fs::set_permissions(&script_path, std::fs::Permissions::from_mode(0o755)).unwrap();
754            Self {
755                dir,
756                script_path,
757                marker_path,
758            }
759        }
760    }
761
762    #[cfg(unix)]
763    impl Drop for TempScript {
764        fn drop(&mut self) {
765            let _ = std::fs::remove_dir_all(&self.dir);
766        }
767    }
