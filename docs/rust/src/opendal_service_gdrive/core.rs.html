104    pub async fn trash_path_if_exists(&self, ctx: &OperationContext, path: &str) -> Result<()> {
105        let mut target_id = match self.resolve_path(ctx, path).await? {
106            Some(id) => Some(id),
107            None => self.resolve_path_after_refresh(ctx, path).await?,
108        };
109
110        if let Some(id) = target_id.take() {
111            let mut resp = self.gdrive_trash(ctx, &id).await?;
112            if resp.status() == StatusCode::NOT_FOUND {
113                self.refresh_path(path).await;
114                target_id = self.resolve_path(ctx, path).await?;
115                if let Some(id) = target_id {
116                    resp = self.gdrive_trash(ctx, &id).await?;
117                } else {
118                    return Ok(());
119                }
120            }
121
122            if resp.status() == StatusCode::NOT_FOUND {
123                return Ok(());
124            }
125            if resp.status() != StatusCode::OK {
126                return Err(parse_error(resp));
127            }
128
129            self.invalidate_file_id(path).await;
130            self.invalidate_dir_id(path).await;
131        }
132
133        Ok(())
134    }

136    pub async fn record_recent_upsert(&self, path: &str, metadata: Metadata) {
137        let mut recent_entries = self.recent_entries.lock().await;
138        prune_recent_entries(&mut recent_entries);
139
140        let mode = metadata.mode();
141        let path = canonical_recent_path(path, mode);
142        let expires_at = Timestamp::now() + Duration::from_secs(15);
143        insert_recent_entry(
144            &mut recent_entries.entries,
145            &path,
146            mode,
147            GdriveRecentEntry {
148                metadata: Some(metadata),
149                expires_at,
150            },
151        );
152        revive_recent_parent_dirs(&mut recent_entries, &path, expires_at);
153    }

175    pub async fn recent_entry_for_path(&self, path: &str) -> GdriveRecentPathState {
176        let mut recent_entries = self.recent_entries.lock().await;
177        prune_recent_entries(&mut recent_entries);
178
179        let recent_entry = lookup_recent_entry(&recent_entries.entries, path);
180        let tombstone = lookup_recent_tombstone(&recent_entries.tombstones, path);
181
182        match (recent_entry, tombstone) {
183            (Some(entry), Some(tombstone_expires_at))
184                if tombstone_expires_at > entry.expires_at =>
185            {
186                GdriveRecentPathState::Deleted
187            }
188            (Some(entry), _) => match entry.metadata.clone() {
189                Some(metadata) => GdriveRecentPathState::Present(Box::new(metadata)),
190                None => GdriveRecentPathState::Deleted,
191            },
192            (None, Some(_)) => GdriveRecentPathState::Deleted,
193            (None, None) => GdriveRecentPathState::Missing,
194        }
195    }

197    pub async fn recent_entries_for_list(
198        &self,
199        scope_path: &str,
200        recursive: bool,
201    ) -> Vec<(String, Metadata)> {
202        let mut recent_entries = self.recent_entries.lock().await;
203        prune_recent_entries(&mut recent_entries);
204
205        let mut entries = recent_entries
206            .entries
207            .iter()
208            .filter_map(|(path, entry)| {
209                let metadata = entry.metadata.clone()?;
210                if metadata.mode().is_dir() && path != &normalize_dir_path(path) {
211                    return None;
212                }
213                if let Some(tombstone_expires_at) =
214                    lookup_recent_tombstone(&recent_entries.tombstones, path)
215                    && tombstone_expires_at > entry.expires_at
216                {
217                    return None;
218                }
219                if let Some(latest_entry) = lookup_recent_entry(&recent_entries.entries, path)
220                    && latest_entry.expires_at > entry.expires_at
221                {
222                    return None;
223                }
224                if recent_entry_in_scope(scope_path, path, metadata.mode(), recursive) {
225                    Some((path.clone(), metadata))
226                } else {
227                    None
228                }
229            })
230            .collect::<Vec<_>>();
231
232        entries.sort_by(|left, right| left.0.cmp(&right.0));
233        entries
234    }

255    pub async fn gdrive_get(
256        &self,
257        ctx: &OperationContext,
258        path: &str,
259        range: BytesRange,
260    ) -> Result<Response<HttpBody>> {
261        let path = build_abs_path(&self.root, path);
262        match self.recent_entry_for_path(&path).await {
263            GdriveRecentPathState::Deleted => {
264                return Err(Error::new(
265                    ErrorKind::NotFound,
266                    format!("path not found: {path}"),
267                ));
268            }
269            GdriveRecentPathState::Present(_) | GdriveRecentPathState::Missing => {}
270        }
271        let path_id = match self.resolve_path(ctx, &path).await? {
272            Some(id) => id,
273            None => match self.resolve_path_after_refresh(ctx, &path).await? {
274                Some(id) => id,
275                None => {
276                    return Err(Error::new(
277                        ErrorKind::NotFound,
278                        format!("path not found: {path}"),
279                    ));
280                }
281            },
282        };
283
284        let url: String = format!("https://www.googleapis.com/drive/v3/files/{path_id}?alt=media");
285
286        let mut req = Request::get(&url)
287            .extension(Operation::Read)
288            .extension(ServiceOperation("DownloadFile"))
289            .header(header::RANGE, range.to_header())
290            .body(Buffer::new())
291            .map_err(new_request_build_error)?;
292        self.sign(ctx, &mut req).await?;
293
294        ctx.http_transport().fetch(req).await
295    }

297    pub async fn gdrive_list(
298        &self,
299        ctx: &OperationContext,
300        file_id: &str,
301        page_size: i32,
302        next_page_token: &str,
303    ) -> Result<Response<Buffer>> {
304        let q = format!("'{file_id}' in parents and trashed = false");
305        let url = "https://www.googleapis.com/drive/v3/files";
306        let mut url = QueryPairsWriter::new(url);
307        url = url.push("pageSize", &page_size.to_string());
308        url = url.push("q", &percent_encode_path(&q));
309        url = url.push(
310            "fields",
311            "nextPageToken,files(id,name,mimeType,size,modifiedTime,parents)",
312        );
313        if !next_page_token.is_empty() {
314            url = url.push("pageToken", next_page_token);
315        };
316
317        let mut req = Request::get(url.finish())
318            .extension(Operation::List)
319            .extension(ServiceOperation("ListFiles"))
320            .body(Buffer::new())
321            .map_err(new_request_build_error)?;
322        self.sign(ctx, &mut req).await?;
323
324        ctx.http_transport().send(req).await
325    }

327    /// List multiple directories in a single API call using OR query.
328    /// This is much more efficient than listing directories one by one.
329    /// Google Drive API supports up to ~50 parent IDs in a single query.
330    pub async fn gdrive_list_batch(
331        &self,
332        ctx: &OperationContext,
333        file_ids: &[String],
334        page_size: i32,
335        next_page_token: &str,
336    ) -> Result<Response<Buffer>> {
337        if file_ids.is_empty() {
338            return Err(Error::new(
339                ErrorKind::Unexpected,
340                "file_ids cannot be empty",
341            ));
342        }
343
344        // Build OR query: ('id1' in parents or 'id2' in parents or ...)
345        let parents_query = file_ids
346            .iter()
347            .map(|id| format!("'{}' in parents", id))
348            .collect::<Vec<_>>()
349            .join(" or ");
350        let q = format!("({}) and trashed = false", parents_query);
351
352        let url = "https://www.googleapis.com/drive/v3/files";
353        let mut url = QueryPairsWriter::new(url);
354        url = url.push("pageSize", &page_size.to_string());
355        url = url.push("q", &percent_encode_path(&q));
356        // Include 'parents' field to know which directory each file belongs to
357        url = url.push(
358            "fields",
359            "nextPageToken,files(id,name,mimeType,size,modifiedTime,parents)",
360        );
361        if !next_page_token.is_empty() {
362            url = url.push("pageToken", next_page_token);
363        };
364
365        let mut req = Request::get(url.finish())
366            .extension(Operation::List)
367            .extension(ServiceOperation("ListFiles"))
368            .body(Buffer::new())
369            .map_err(new_request_build_error)?;
370        self.sign(ctx, &mut req).await?;
371
372        ctx.http_transport().send(req).await
373    }

375    // Update with content and metadata
376    pub async fn gdrive_patch_metadata_request(
377        &self,
378        ctx: &OperationContext,
379        source: &str,
380        target: &str,
381    ) -> Result<Response<Buffer>> {
382        let source_file_id = match self.resolve_path(ctx, source).await? {
383            Some(id) => id,
384            None => self
385                .resolve_path_after_refresh(ctx, source)
386                .await?
387                .ok_or(Error::new(
388                    ErrorKind::NotFound,
389                    format!("source path not found: {source}"),
390                ))?,
391        };
392        let source_parent = get_parent(source);
393        let source_parent_id = match self.resolve_path(ctx, source_parent).await? {
394            Some(id) => id,
395            None => self
396                .resolve_path_after_refresh(ctx, source_parent)
397                .await?
398                .expect("old parent must exist"),
399        };
400
401        let target_parent_id = self.ensure_dir(ctx, get_parent(target)).await?;
402        let target_file_name = get_basename(target);
403
404        let metadata = &json!({
405            "name": target_file_name,
406            "removeParents": [source_parent_id],
407            "addParents": [target_parent_id],
408        });
409
410        let url = format!("https://www.googleapis.com/drive/v3/files/{source_file_id}");
411        let mut req = Request::patch(url)
412            .extension(Operation::Rename)
413            .extension(ServiceOperation("MoveFile"))
414            .body(Buffer::from(Bytes::from(metadata.to_string())))
415            .map_err(new_request_build_error)?;
416
417        self.sign(ctx, &mut req).await?;
418
419        ctx.http_transport().send(req).await
420    }

445    /// Create a file with the content.
446    pub async fn gdrive_upload_simple_request(
447        &self,
448        ctx: &OperationContext,
449        path: &str,
450        size: u64,
451        body: Buffer,
452    ) -> Result<Response<Buffer>> {
453        let parent = self.ensure_dir(ctx, get_parent(path)).await?;
454
455        let url = "https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart";
456
457        let file_name = get_basename(path);
458
459        let metadata = serde_json::to_vec(&json!({
460            "name": file_name,
461            "parents": [parent],
462        }))
463        .map_err(new_json_serialize_error)?;
464
465        let req = Request::post(url)
466            .header("X-Upload-Content-Length", size)
467            .extension(Operation::Write)
468            .extension(ServiceOperation("UploadFile"));
469
470        let multipart = Multipart::new()
471            .part(
472                FormDataPart::new("metadata")
473                    .header(
474                        header::CONTENT_TYPE,
475                        "application/json; charset=UTF-8".parse().unwrap(),
476                    )
477                    .content(metadata),
478            )
479            .part(
480                FormDataPart::new("file")
481                    .header(
482                        header::CONTENT_TYPE,
483                        "application/octet-stream".parse().unwrap(),
484                    )
485                    .content(body),
486            );
487
488        let mut req = multipart.apply(req)?;
489
490        self.sign(ctx, &mut req).await?;
491
492        ctx.http_transport().send(req).await
493    }

529    pub async fn gdrive_copy(
530        &self,
531        ctx: &OperationContext,
532        from: &str,
533        to: &str,
534    ) -> Result<Response<Buffer>> {
535        let from = build_abs_path(&self.root, from);
536
537        let from_file_id = match self.resolve_path(ctx, &from).await? {
538            Some(id) => id,
539            None => match self.resolve_path_after_refresh(ctx, &from).await? {
540                Some(id) => id,
541                None => {
542                    return Err(Error::new(
543                        ErrorKind::NotFound,
544                        "the file to copy does not exist",
545                    ));
546                }
547            },
548        };
549
550        let to_name = get_basename(to);
551        let to_path = build_abs_path(&self.root, to);
552        let to_parent_id = self.ensure_dir(ctx, get_parent(&to_path)).await?;
553
554        self.trash_path_if_exists(ctx, &to_path).await?;
555
556        let url = format!("https://www.googleapis.com/drive/v3/files/{from_file_id}/copy");
557
558        let request_body = &json!({
559            "name": to_name,
560            "parents": [to_parent_id],
561        });
562        let body = Buffer::from(Bytes::from(request_body.to_string()));
563
564        let mut req = Request::post(&url)
565            .extension(Operation::Copy)
566            .extension(ServiceOperation("CopyFile"))
567            .body(body)
568            .map_err(new_request_build_error)?;
569        self.sign(ctx, &mut req).await?;
570
571        ctx.http_transport().send(req).await
572    }

575#[derive(Clone)]
576pub struct GdriveRecentEntry {
577    metadata: Option<Metadata>,
578    expires_at: Timestamp,
579}
580
581#[derive(Default)]
582pub struct GdriveRecentState {
583    entries: HashMap<String, GdriveRecentEntry>,
584    tombstones: HashMap<String, Timestamp>,
585}
586
587#[derive(Clone, Debug)]
588pub enum GdriveRecentPathState {
589    Missing,
590    Deleted,
591    Present(Box<Metadata>),
592}
593
594pub(crate) fn normalize_dir_path(path: &str) -> String {
595    if path.is_empty() || path.ends_with('/') {
596        path.to_string()
597    } else {
598        format!("{path}/")
599    }
600}
601
602fn canonical_recent_path(path: &str, mode: EntryMode) -> String {
603    if mode.is_dir() {
604        normalize_dir_path(path)
605    } else {
606        path.to_string()
607    }
608}
609
610fn prune_recent_entries(entries: &mut GdriveRecentState) {
611    let now = Timestamp::now();
612    entries.entries.retain(|_, entry| entry.expires_at > now);
613    entries.tombstones.retain(|_, expires_at| *expires_at > now);
614}
615
616fn lookup_recent_entry<'a>(
617    entries: &'a HashMap<String, GdriveRecentEntry>,
618    path: &str,
619) -> Option<&'a GdriveRecentEntry> {
620    entries.get(path)
621}
622
623fn lookup_recent_tombstone(
624    tombstones: &HashMap<String, Timestamp>,
625    path: &str,
626) -> Option<Timestamp> {
627    let candidates = recent_path_candidates(path);
628    tombstones
629        .iter()
630        .filter(|(tombstone, _)| {
631            candidates
632                .iter()
633                .any(|candidate| path_under_tombstone(candidate, tombstone))
634        })
635        .map(|(_, expires_at)| *expires_at)
636        .max()
637}
638
639fn revive_recent_parent_dirs(entries: &mut GdriveRecentState, path: &str, expires_at: Timestamp) {
640    let mut parent = get_parent(path).to_string();
641
642    while !parent.is_empty() && parent != "/" {
643        if lookup_recent_tombstone(&entries.tombstones, &parent).is_some() {
644            insert_recent_entry(
645                &mut entries.entries,
646                &parent,
647                EntryMode::DIR,
648                GdriveRecentEntry {
649                    metadata: Some(Metadata::new(EntryMode::DIR)),
650                    expires_at,
651                },
652            );
653        }
654
655        parent = get_parent(&parent).to_string();
656    }
657}
658
659fn insert_recent_entry(
660    entries: &mut HashMap<String, GdriveRecentEntry>,
661    path: &str,
662    mode: EntryMode,
663    entry: GdriveRecentEntry,
664) {
665    entries.insert(path.to_string(), entry.clone());
666
667    if mode.is_dir() {
668        let alias = path.trim_end_matches('/');
669        if !alias.is_empty() {
670            entries.insert(alias.to_string(), entry);
671        }
672    }
673}
674
675fn recent_path_candidates(path: &str) -> Vec<String> {
676    let mut candidates = vec![path.to_string()];
677    let dir_path = normalize_dir_path(path);
678    if dir_path != path {
679        candidates.push(dir_path);
680    }
681    candidates
682}
683
684fn path_under_tombstone(path: &str, tombstone: &str) -> bool {
685    tombstone.is_empty() || path == tombstone || path.starts_with(tombstone)
686}
687
688fn recent_entry_in_scope(scope_path: &str, path: &str, mode: EntryMode, recursive: bool) -> bool {
689    if recursive {
690        if scope_path.is_empty() {
691            return true;
692        }
693
694        if scope_path.ends_with('/') {
695            return path.starts_with(scope_path) && path != scope_path;
696        }
697
698        return path == canonical_recent_path(scope_path, mode) || path.starts_with(scope_path);
699    }
700
701    let parent = get_parent(path);
702    if scope_path.is_empty() {
703        parent == "/"
704    } else {
705        parent == scope_path
706    }
707}

709#[derive(Clone)]
710pub struct GdriveSigner {
711    pub client_id: String,
712    pub client_secret: String,
713    pub refresh_token: String,
714
715    pub access_token: String,
716    pub expires_in: Timestamp,
717}
718
719impl GdriveSigner {
720    /// Create a new signer.
721    pub fn new() -> Self {
722        GdriveSigner {
723            client_id: "".to_string(),
724            client_secret: "".to_string(),
725            refresh_token: "".to_string(),
726            access_token: "".to_string(),
727            expires_in: Timestamp::MIN,
728        }
729    }
730
731    /// Sign a request.
732    pub async fn sign<T>(&mut self, ctx: &OperationContext, req: &mut Request<T>) -> Result<()> {
733        if !self.access_token.is_empty() && self.expires_in > Timestamp::now() {
734            let value = format!("Bearer {}", self.access_token)
735                .parse()
736                .expect("access token must be valid header value");
737
738            req.headers_mut().insert(header::AUTHORIZATION, value);
739            return Ok(());
740        }
741
742        let url = format!(
743            "https://oauth2.googleapis.com/token?refresh_token={}&client_id={}&client_secret={}&grant_type=refresh_token",
744            self.refresh_token, self.client_id, self.client_secret
745        );
746
747        {
748            let req = Request::post(url)
749                .header(header::CONTENT_LENGTH, 0)
750                .body(Buffer::new())
751                .map_err(new_request_build_error)?;
752
753            let resp = ctx.http_transport().send(req).await?;
754            let status = resp.status();
755
756            match status {
757                StatusCode::OK => {
758                    let resp_body = resp.into_body();
759                    let token: GdriveTokenResponse = serde_json::from_reader(resp_body.reader())
760                        .map_err(new_json_deserialize_error)?;
761                    self.access_token.clone_from(&token.access_token);
762                    self.expires_in = Timestamp::now() + Duration::from_secs(token.expires_in)
763                        - Duration::from_secs(120);
764                }
765                _ => {
766                    return Err(parse_error(resp));
767                }
768            }
769        }
770
771        let auth_header_content = format!("Bearer {}", self.access_token);
772        req.headers_mut()
773            .insert(header::AUTHORIZATION, auth_header_content.parse().unwrap());
774
775        Ok(())
776    }
777}

789impl crate::path_index::GdrivePathQueryer for GdrivePathQuery {
790    async fn root(&self, _: &OperationContext) -> Result<String> {
791        Ok("root".to_string())
792    }
793
794    async fn query(
795        &self,
796        ctx: &OperationContext,
797        parent_id: &str,
798        name: &str,
799    ) -> Result<Option<String>> {
800        let mut queries = vec![
801            // Make sure name has been replaced with escaped name.
802            //
803            // ref: <https://developers.google.com/drive/api/guides/ref-search-terms>
804            format!(
805                "name = '{}'",
806                name.replace('\'', "\\'").trim_end_matches('/')
807            ),
808            format!("'{}' in parents", parent_id),
809            "trashed = false".to_string(),
810        ];
811        if name.ends_with('/') {
812            queries.push("mimeType = 'application/vnd.google-apps.folder'".to_string());
813        }
814        let query = queries.join(" and ");
815        let order_by = "modifiedTime desc,createdTime desc";
816
817        let url = format!(
818            "https://www.googleapis.com/drive/v3/files?q={}&orderBy={}&pageSize=1",
819            percent_encode_path(query.as_str()),
820            percent_encode_path(order_by)
821        );
822
823        let mut req = Request::get(&url)
824            .extension(Operation::Stat)
825            .extension(ServiceOperation("ListFiles"))
826            .body(Buffer::new())
827            .map_err(new_request_build_error)?;
828
829        self.signer.lock().await.sign(ctx, &mut req).await?;
830
831        let resp = ctx.http_transport().send(req).await?;
832        let status = resp.status();
833
834        match status {
835            StatusCode::OK => {
836                let body = resp.into_body();
837                let meta: GdriveFileList =
838                    serde_json::from_reader(body.reader()).map_err(new_json_deserialize_error)?;
839
840                if let Some(f) = meta.files.first() {
841                    Ok(Some(f.id.clone()))
842                } else {
843                    Ok(None)
844                }
845            }
846            _ => Err(parse_error(resp)),
847        }
848    }
849
850    async fn create_dir(
851        &self,
852        ctx: &OperationContext,
853        parent_id: &str,
854        name: &str,
855    ) -> Result<String> {
856        let url = "https://www.googleapis.com/drive/v3/files";
857
858        let content = serde_json::to_vec(&json!({
859            "name": name.trim_end_matches('/'),
860            "mimeType": "application/vnd.google-apps.folder",
861            // If the parent is not provided, the folder will be created in the root folder.
862            "parents": [parent_id],
863        }))
864        .map_err(new_json_serialize_error)?;
865
866        let mut req = Request::post(url)
867            .extension(Operation::CreateDir)
868            .extension(ServiceOperation("CreateFolder"))
869            .header(header::CONTENT_TYPE, "application/json")
870            .body(Buffer::from(Bytes::from(content)))
871            .map_err(new_request_build_error)?;
872
873        self.signer.lock().await.sign(ctx, &mut req).await?;
874
875        let resp = ctx.http_transport().send(req).await?;
876        if !resp.status().is_success() {
877            return Err(parse_error(resp));
878        }
879
880        let body = resp.into_body();
881        let file: GdriveFile =
882            serde_json::from_reader(body.reader()).map_err(new_json_deserialize_error)?;
883        Ok(file.id)
884    }
885}

1080mod error {
1081    use http::Response;
1082    use http::StatusCode;
1083    use serde::Deserialize;
1084
1085    use opendal_core::raw::*;
1086    use opendal_core::*;
1087
1088    #[derive(Default, Debug, Deserialize)]
1089    struct GdriveError {
1090        error: GdriveInnerError,
1091    }
1092
1093    #[derive(Default, Debug, Deserialize)]
1094    struct GdriveInnerError {
1095        message: String,
1096    }
1097
1098    /// Parse error response into Error.
1099    pub(crate) fn parse_error(resp: Response<Buffer>) -> Error {
1100        let (parts, body) = resp.into_parts();
1101        let bs = body.to_bytes();
1102
1103        let (mut kind, mut retryable) = match parts.status {
1104        StatusCode::NOT_FOUND => (ErrorKind::NotFound, false),
1105        StatusCode::FORBIDDEN => (ErrorKind::PermissionDenied, false),
1106        StatusCode::INTERNAL_SERVER_ERROR
1107        | StatusCode::BAD_GATEWAY
1108        | StatusCode::SERVICE_UNAVAILABLE
1109        | StatusCode::GATEWAY_TIMEOUT
1110        // Gdrive sometimes return METHOD_NOT_ALLOWED for our requests for abuse detection.
1111        | StatusCode::METHOD_NOT_ALLOWED => (ErrorKind::Unexpected, true),
1112        _ => (ErrorKind::Unexpected, false),
1113    };
1114
1115        let (message, gdrive_err) = serde_json::from_slice::<GdriveError>(bs.as_ref())
1116            .map(|gdrive_err| (format!("{gdrive_err:?}"), Some(gdrive_err)))
1117            .unwrap_or_else(|_| (String::from_utf8_lossy(&bs).into_owned(), None));
1118
1119        if let Some(gdrive_err) = gdrive_err {
1120            (kind, retryable) = parse_gdrive_error_code(gdrive_err.error.message.as_str())
1121                .unwrap_or((kind, retryable));
1122        }
1123
1124        let mut err = Error::new(kind, message);
1125
1126        err = with_error_response_context(err, parts);
1127
1128        if retryable {
1129            err = err.set_temporary();
1130        }
1131
1132        err
1133    }
1134
1135    pub fn parse_gdrive_error_code(message: &str) -> Option<(ErrorKind, bool)> {
1136        match message {
1137            // > Please reduce your request rate.
1138            //
1139            // It's Ok to retry since later on the request rate may get reduced.
1140            "User rate limit exceeded." => Some((ErrorKind::RateLimited, true)),
1141            _ => None,
1142        }
1143    }
1144}
