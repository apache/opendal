31pub struct GhacWriter(pub TwoWays<GhacWriterV1, GhacWriterV2>);
32
33impl GhacWriter {
34    /// TODO: maybe we can move the signed url logic to azblob service instead.
35    pub fn new(
36        core: Arc<GhacCore>,
37        ctx: OperationContext,
38        executor: Executor,
39        write_path: String,
40        url: String,
41    ) -> Result<Self> {
42        match core.service_version {
43            GhacVersion::V1 => Ok(GhacWriter(TwoWays::One(GhacWriterV1 {
44                core,
45                ctx,
46                path: write_path,
47                url,
48                size: 0,
49            }))),
50            GhacVersion::V2 => {
51                let uri = http::Uri::from_str(&url)
52                    .map_err(new_http_uri_invalid_error)?
53                    .into_parts();
54                let (Some(scheme), Some(authority), Some(pq)) =
55                    (uri.scheme, uri.authority, uri.path_and_query)
56                else {
57                    return Err(Error::new(
58                        ErrorKind::Unexpected,
59                        "ghac returns invalid signed url",
60                    )
61                    .with_context("url", HttpUri::new(url).redacted_uri()));
62                };
63                let endpoint = format!("{scheme}://{authority}");
64                let Some((container, path)) = pq.path().trim_matches('/').split_once("/") else {
65                    return Err(Error::new(
66                        ErrorKind::Unexpected,
67                        "ghac returns invalid signed url that bucket or path is missing",
68                    )
69                    .with_context("url", HttpUri::new(url).redacted_uri()));
70                };
71                let Some(query) = pq.query() else {
72                    return Err(Error::new(
73                        ErrorKind::Unexpected,
74                        "ghac returns invalid signed url that sas is missing",
75                    )
76                    .with_context("url", HttpUri::new(url).redacted_uri()));
77                };
78                let azure_core = Arc::new(AzblobCore {
79                    info: ServiceInfo::new("azblob", "/", container),
80                    capability: Capability {
81                        stat: true,
82                        stat_with_if_match: true,
83                        stat_with_if_none_match: true,
84
85                        read: true,
86
87                        read_with_if_match: true,
88                        read_with_if_none_match: true,
89                        read_with_override_content_disposition: true,
90                        read_with_if_modified_since: true,
91                        read_with_if_unmodified_since: true,
92
93                        write: true,
94                        write_can_append: true,
95                        write_can_empty: true,
96                        write_can_multi: true,
97                        write_with_cache_control: true,
98                        write_with_content_type: true,
99                        write_with_if_not_exists: true,
100                        write_with_if_none_match: true,
101                        write_with_user_metadata: true,
102
103                        copy: true,
104
105                        list: true,
106                        list_with_recursive: true,
107
108                        shared: true,
109
110                        ..Default::default()
111                    },
112                    container: container.to_string(),
113                    root: "/".to_string(),
114                    endpoint,
115                    encryption_key: None,
116                    encryption_key_sha256: None,
117                    encryption_algorithm: None,
118                    skip_signature: false,
119                    sign_ctx: Context::new(),
120                    credential_provider: Arc::new(StaticCredentialProvider::new_sas_token(query)),
121                    request_signer: Arc::new(RequestSigner::new()),
122                    signers: HttpTransportCache::new(),
123                });
124                let w = AzblobWriter::new(
125                    azure_core,
126                    ctx.clone(),
127                    OpWrite::default(),
128                    percent_decode_path(path),
129                );
130                let writer = oio::BlockWriter::new(executor, w, 4);
131                Ok(GhacWriter(TwoWays::Two(GhacWriterV2 {
132                    core,
133                    ctx,
134                    writer,
135                    path: write_path,
136                    url,
137                    size: 0,
138                })))
139            }
140        }
141    }
142}

144pub struct GhacLazyWriter {
145    core: Arc<GhacCore>,
146    ctx: OperationContext,
147    executor: Executor,
148    path: String,
149    inner: Option<GhacWriter>,
150}
151
152impl GhacLazyWriter {
153    pub fn new(
154        core: Arc<GhacCore>,
155        ctx: OperationContext,
156        executor: Executor,
157        path: String,
158    ) -> Self {
159        Self {
160            core,
161            ctx,
162            executor,
163            path,
164            inner: None,
165        }
166    }
167
168    async fn inner(&mut self) -> Result<&mut GhacWriter> {
169        if self.inner.is_none() {
170            let url = self.core.ghac_get_upload_url(&self.ctx, &self.path).await?;
171            self.inner = Some(GhacWriter::new(
172                self.core.clone(),
173                self.ctx.clone(),
174                self.executor.clone(),
175                self.path.clone(),
176                url,
177            )?);
178        }
179
180        Ok(self.inner.as_mut().expect("writer must be initialized"))
181    }
182}
183
184impl oio::Write for GhacLazyWriter {
185    async fn write(&mut self, bs: Buffer) -> Result<()> {
186        self.inner().await?.write(bs).await
187    }
188
189    async fn abort(&mut self) -> Result<()> {
190        match &mut self.inner {
191            Some(w) => w.abort().await,
192            None => Ok(()),
193        }
194    }
195
196    async fn close(&mut self) -> Result<Metadata> {
197        self.inner().await?.close().await
198    }
199}

215pub struct GhacWriterV1 {
216    core: Arc<GhacCore>,
217    ctx: OperationContext,
218
219    path: String,
220    url: String,
221    size: u64,
222}
223
224impl oio::Write for GhacWriterV1 {
225    async fn write(&mut self, bs: Buffer) -> Result<()> {
226        if bs.is_empty() {
227            return Ok(());
228        }
229
230        let size = bs.len() as u64;
231        let offset = self.size;
232
233        let resp = self
234            .core
235            .ghac_v1_write(&self.ctx, &self.url, size, offset, bs)
236            .await?;
237        if !resp.status().is_success() {
238            return Err(
239                parse_error(ErrorContext::new(ServiceOperation("UploadChunk")), resp)
240                    .with_operation("Backend::ghac_upload"),
241            );
242        }
243        self.size += size;
244        Ok(())
245    }
246
247    async fn abort(&mut self) -> Result<()> {
248        Ok(())
249    }
250
251    async fn close(&mut self) -> Result<Metadata> {
252        self.core
253            .ghac_finalize_upload(&self.ctx, &self.path, &self.url, self.size)
254            .await?;
255        Ok({
256            let mut metadata = MetadataBuilder::unknown();
257            metadata.set_file(self.size);
258            metadata.build()
259        })
260    }
261}

263pub struct GhacWriterV2 {
264    core: Arc<GhacCore>,
265    ctx: OperationContext,
266    writer: oio::BlockWriter<AzblobWriter>,
267
268    path: String,
269    url: String,
270    size: u64,
271}
272
273impl oio::Write for GhacWriterV2 {
274    async fn write(&mut self, bs: Buffer) -> Result<()> {
275        let size = bs.len() as u64;
276
277        self.writer.write(bs).await?;
278        self.size += size;
279        Ok(())
280    }
281
282    async fn close(&mut self) -> Result<Metadata> {
283        self.writer.close().await?;
284        self.core
285            .ghac_finalize_upload(&self.ctx, &self.path, &self.url, self.size)
286            .await?;
287        Ok({
288            let mut metadata = MetadataBuilder::unknown();
289            metadata.set_file(self.size);
290            metadata.build()
291        })
292    }
293
294    async fn abort(&mut self) -> Result<()> {
295        self.writer.abort().await
296    }

345    #[derive(Debug)]
346    struct RecordedRequest {
347        method: Method,
348        uri: String,
349        content_range: Option<String>,
350        content_length: Option<String>,
351        body: Bytes,
352    }
353
354    type Responder = Box<dyn Fn(&Request<Buffer>) -> (StatusCode, Bytes) + Send + Sync>;
355
356    struct RecordingTransport {
357        requests: Arc<Mutex<Vec<RecordedRequest>>>,
358        responder: Responder,
359    }
360
361    impl HttpTransport for RecordingTransport {
362        async fn fetch(&self, req: Request<Buffer>) -> Result<Response<HttpBody>> {
363            let (status, resp_body) = (self.responder)(&req);
364            let method = req.method().clone();
365            let uri = req.uri().to_string();
366            let content_range = req
367                .headers()
368                .get(CONTENT_RANGE)
369                .and_then(|v| v.to_str().ok())
370                .map(ToOwned::to_owned);
371            let content_length = req
372                .headers()
373                .get(CONTENT_LENGTH)
374                .and_then(|v| v.to_str().ok())
375                .map(ToOwned::to_owned);
376            let body = req.into_body().to_bytes();
377
378            self.requests.lock().unwrap().push(RecordedRequest {
379                method,
380                uri,
381                content_range,
382                content_length,
383                body,
384            });
385
386            let buf = Buffer::from(resp_body);
387            let len = buf.len() as u64;
388            Response::builder()
389                .status(status)
390                .body(HttpBody::new(futures::stream::iter([Ok(buf)]), Some(len)))
391                .map_err(new_request_build_error)
392        }
393    }

397        tokio::time::timeout(Duration::from_secs(10), async {
398            let requests = Arc::new(Mutex::new(Vec::new()));
399            let transport = RecordingTransport {
400                requests: requests.clone(),
401                responder: Box::new(|_| (StatusCode::NO_CONTENT, Bytes::new())),
402            };
403            let ctx = OperationContext::from_parts(
404                HttpTransporter::new(transport),
405                Executor::with(TokioExecutor),
406            );
407            let core = test_core(GhacVersion::V1);
408
409            let mut writer = GhacWriter::new(
410                core.clone(),
411                ctx.clone(),
412                ctx.executor().clone(),
413                "cache-key".to_string(),
414                "https://results.actions.githubusercontent.com/_apis/artifactcache/caches/7"
415                    .to_string(),
416            )
417            .unwrap();
418
419            // Empty write at offset 0 must not panic or send any HTTP request.
420            writer.write(Buffer::new()).await.unwrap();
421            assert!(requests.lock().unwrap().is_empty());
422
423            writer.write(Buffer::from("abc")).await.unwrap();
424            // Empty write at offset > 0 must not send an inverted Content-Range.
425            writer.write(Buffer::new()).await.unwrap();
426            writer.write(Buffer::from("de")).await.unwrap();
427
428            let meta = writer.close().await.unwrap();
429            assert_eq!(meta.content_length(), 5);
430
431            let recorded = requests.lock().unwrap();
432            assert_eq!(recorded.len(), 3);
433            assert_eq!(recorded[0].method, Method::PATCH);
434            assert_eq!(recorded[0].content_range.as_deref(), Some("bytes 0-2/*"));
435            assert_eq!(recorded[0].content_length.as_deref(), Some("3"));
436            assert_eq!(recorded[0].body, Bytes::from_static(b"abc"));

438            assert_eq!(recorded[1].method, Method::PATCH);

440            assert_eq!(recorded[1].content_length.as_deref(), Some("2"));

442
443            assert_eq!(recorded[2].method, Method::POST);
444            assert_eq!(recorded[2].body, Bytes::from_static(br#"{"size":5}"#));
445        })
446        .await
447        .expect("test timed out");
448    }

515    struct HangingBlockTransport {
516        started: Mutex<Option<oneshot::Sender<()>>>,
517        cancelled: Mutex<Option<oneshot::Sender<()>>>,
518    }
519
520    struct DropSignal(Option<oneshot::Sender<()>>);
521
522    impl Drop for DropSignal {
523        fn drop(&mut self) {
524            if let Some(tx) = self.0.take() {
525                let _ = tx.send(());
526            }
527        }
528    }
529
530    impl HttpTransport for HangingBlockTransport {
531        async fn fetch(&self, _req: Request<Buffer>) -> Result<Response<HttpBody>> {
532            if let Some(tx) = self.started.lock().unwrap().take() {
533                let _ = tx.send(());
534            }
535            let _guard = DropSignal(self.cancelled.lock().unwrap().take());
536            std::future::pending::<Result<Response<HttpBody>>>().await
537        }
538    }

541    async fn test_ghac_v2_writer_abort_cancels_inflight_block_upload() {
542        tokio::time::timeout(Duration::from_secs(10), async {
543            let (started_tx, started_rx) = oneshot::channel();
544            let (cancelled_tx, cancelled_rx) = oneshot::channel();
545            let transport = HangingBlockTransport {
546                started: Mutex::new(Some(started_tx)),
547                cancelled: Mutex::new(Some(cancelled_tx)),
548            };
549            let ctx = OperationContext::from_parts(
550                HttpTransporter::new(transport),
551                Executor::with(TokioExecutor),
552            );
553            let mut writer = GhacWriter::new(
554                test_core(GhacVersion::V2),
555                ctx.clone(),
556                ctx.executor().clone(),
557                "cache-key".to_string(),
558                "https://account.blob.core.windows.net/cache-container/archive.tzst?sv=2023-01-03&sig=test-sig"
559                    .to_string(),
560            )
561            .unwrap();
562
563            // First write fills BlockWriter's single-block cache; second write schedules
564            // the first block onto the background executor.
565            writer.write(Buffer::from("chunk-1")).await.unwrap();
566            writer.write(Buffer::from("chunk-2")).await.unwrap();
567
568            started_rx
569                .await
570                .expect("background block upload should start");
571
572            writer.abort().await.unwrap();
573
574            // Keep `writer` alive across this wait so cancellation proves `abort()` itself
575            // tore down the in-flight BlockWriter task rather than `Drop`.
576            cancelled_rx
577                .await
578                .expect("abort() must cancel in-flight BlockWriter upload tasks");
579            drop(writer);
580        })
581        .await
582        .expect("test timed out");
583    }
