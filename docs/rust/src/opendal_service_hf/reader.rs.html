28pub enum HfReadStream {
29    Http(HttpBody),
30    Xet(XetDownloadStream),
31}
32
33/// Converts an opendal byte range into the `Option<Range<u64>>` the `xet`
34/// crate expects: `None` for a full read, `Some(start..end)` otherwise, with
35/// an open-ended range (`size` unknown) mapped to `start..u64::MAX`.
36fn xet_range(range: BytesRange) -> Option<Range<u64>> {
37    if range.is_full() {
38        None
39    } else {
40        let start = range.offset();
41        let end = range.size().map(|s| start + s).unwrap_or(u64::MAX);
42        Some(start..end)
43    }
44}
45
46impl HfReadStream {
47    /// Build the stream directly from an already-known XET hash + size,
48    /// whether that came from a resolve response just parsed in [`HfReader::open`]
49    /// or was reused from its cache.
50    async fn new_xet(
51        group: &XetDownloadStreamGroup,
52        hash: &str,
53        size: u64,
54        range: BytesRange,
55    ) -> Result<(RpRead, Self)> {
56        let metadata = {
57            let metadata = MetadataBuilder::file(size);
58            metadata.build()
59        };
60        let xet_range = xet_range(range);
61
62        let mut stream = group
63            .download_stream(XetFileInfo::new(hash.to_string(), size), xet_range)
64            .await
65            .map_err(|err| {
66                Error::new(
67                    ErrorKind::Unexpected,
68                    "failed to create xet download stream",
69                )
70                .set_source(err)
71            })?;
72        stream.start();
73        Ok((RpRead::new(metadata), Self::Xet(stream)))
74    }
75
76    /// Build the stream from a plain (non-XET) resolve response: it already
77    /// carries the range-correct bytes, so there is nothing left to fetch.
78    fn new_http(path: &str, resp: Response<HttpBody>) -> Result<(RpRead, Self)> {
79        let metadata = parse_into_metadata(path, resp.headers())?;
80        Ok((RpRead::new(metadata), Self::Http(resp.into_body())))
81    }
82
83    /// Retrieve a resolve response's XET metadata, or hand the response
84    /// back untouched if it doesn't carry any -- its body is then the
85    /// actual range content, still there for [`Self::new_http`].
86    async fn maybe_xet(
87        resp: Response<HttpBody>,
88    ) -> Result<std::result::Result<XetFileResponse, Response<HttpBody>>> {
89        if !resp.headers().contains_key("x-xet-hash") {
90            return Ok(Err(resp));
91        }
92        let (_, mut body) = resp.into_parts();
93        let buf = body.to_buffer().await?;
94        let info: XetFileResponse =
95            serde_json::from_reader(buf.reader()).map_err(new_json_deserialize_error)?;
96        Ok(Ok(info))
97    }
98}
99
100fn map_session_error(e: SessionError) -> Error {
101    Error::new(ErrorKind::Unexpected, "xet read error").set_source(e)
102}

104impl oio::ReadStream for HfReadStream {
105    async fn read(&mut self) -> Result<Buffer> {
106        match self {
107            Self::Http(body) => body.read().await,
108            Self::Xet(stream) => match stream.next().await {
109                Ok(Some(bytes)) => Ok(Buffer::from(bytes)),
110                Ok(None) => Ok(Buffer::new()),
111                Err(e) => Err(map_session_error(e)),
112            },
113        }
114    }
115}

117/// Reader returned by this backend.
118pub struct HfReader {
119    backend: HfBackend,
120    ctx: OperationContext,
121    path: String,
122    // `Some((hash, size))`/`None` once this path is known to be (or not be)
123    // XET-backed; only consulted in `HfDownloadMode::Xet`. Single-flighted via
124    // `get_or_try_init`, so concurrent cold opens share one classifying
125    // resolve and every later range reuses it. If not XET-backed, that first
126    // resolve's body goes unused and every open falls back to its own fresh
127    // resolve -- one wasted request per reader, not per range.
128    //
129    // Never re-validated for this reader's life, so a file changed mid-read
130    // (e.g. a new commit on a floating `main`) can serve stale content --
131    // scoped to one reader, not the `Operator`, to bound that window.
132    resolved: OnceCell<Option<(String, u64)>>,
133    // Built once per reader and reused for every later XET range, seeded
134    // from `HfCore`'s cached CAS token. Reuse across concurrent ranges
135    // relies on the `xet` crate's documented (not type-enforced) support
136    // for many streams per group.
137    xet_group: OnceCell<XetDownloadStreamGroup>,
138}
139
140impl HfReader {
141    pub(super) fn new(backend: HfBackend, ctx: OperationContext, path: &str, _: OpRead) -> Self {
142        Self {
143            backend,
144            ctx,
145            path: path.to_string(),
146            resolved: OnceCell::new(),
147            xet_group: OnceCell::new(),
148        }
149    }
150
151    /// Build a XET stream via this reader's cached group, creating it on
152    /// first use.
153    async fn xet_stream(
154        &self,
155        core: &HfCore,
156        hash: &str,
157        size: u64,
158        range: BytesRange,
159    ) -> Result<(RpRead, HfReadStream)> {
160        let group = self
161            .xet_group
162            .get_or_try_init(|| core.xet_download_group(&self.ctx))
163            .await?;
164        HfReadStream::new_xet(group, hash, size, range).await
165    }
166
167    /// Classify an already-fetched resolve response and dispatch to the
168    /// matching stream. Used by [`Self::open`]'s Http-mode branch, which
169    /// classifies the same response it fetches bytes from. The Xet-mode
170    /// fallback (re-resolving in `HfDownloadMode::Http` after classification
171    /// came back non-XET) does not call this -- it never re-checks for
172    /// XET metadata on that second resolve, so it goes straight to
173    /// `new_http` instead.
174    async fn dispatch(
175        &self,
176        core: &HfCore,
177        path: &str,
178        range: BytesRange,
179        resp: Response<HttpBody>,
180    ) -> Result<(RpRead, HfReadStream)> {
181        match HfReadStream::maybe_xet(resp).await? {
182            Ok(info) => self.xet_stream(core, &info.hash, info.size, range).await,
183            Err(resp) => HfReadStream::new_http(path, resp),
184        }
185    }
186}
187
188impl oio::StreamRead for HfReader {
189    async fn open(&self, range: BytesRange) -> Result<(RpRead, Box<dyn oio::ReadStreamDyn>)> {
190        let core = &self.backend.core;
191        let path = self.path.as_str();
192
193        if core.download_mode != HfDownloadMode::Xet {
194            // Http mode: resolve() is itself the byte-fetching request, done
195            // fresh per range -- there's no separate metadata step to cache.
196            let resp = core
197                .resolve(&self.ctx, path, range, core.download_mode)
198                .await?;
199            let (rp, stream) = self.dispatch(core, path, range, resp).await?;
200            return Ok((rp, Box::new(stream) as Box<dyn oio::ReadStreamDyn>));
201        }
202
203        // Xet mode: classification is single-flighted, see `resolved`'s
204        // doc comment for why that's free for the XET outcome and cheap
205        // (one extra request, once per reader) for the NotXet one.
206        let classification = self
207            .resolved
208            .get_or_try_init(|| async {
209                let resp = core
210                    .resolve(&self.ctx, path, range, HfDownloadMode::Xet)
211                    .await?;
212                match HfReadStream::maybe_xet(resp).await? {
213                    Ok(info) => Ok(Some((info.hash, info.size))),
214                    Err(_) => Ok(None),
215                }
216            })
217            .await?;
218
219        let (rp, stream) = match classification {
220            Some((hash, size)) => self.xet_stream(core, hash, *size, range).await?,
221            None => {
222                let resp = core
223                    .resolve(&self.ctx, path, range, HfDownloadMode::Http)
224                    .await?;
225                HfReadStream::new_http(path, resp)?
226            }
227        };
228
229        Ok((rp, Box::new(stream) as Box<dyn oio::ReadStreamDyn>))
230    }
231}

404    /// Documents a known sharp edge in the single-flighted classification:
405    /// the `Range` header on the one shared classifying resolve comes from
406    /// whichever concurrent `open()` call happens to win `get_or_try_init`'s
407    /// race, not from the caller that ends up consuming the result. All
408    /// racing callers get that winner's classification regardless of their
409    /// own requested range. See `resolved`'s field doc and the review notes
410    /// on this PR -- not yet confirmed against the real HF service, so no
411    /// fix has been applied; this test only pins today's behavior so a
412    /// future change here is deliberate.
413    #[tokio::test]
414    async fn test_concurrent_cold_opens_classify_with_one_racing_callers_range() -> Result<()> {
415        let (core, ctx, mock_client) = create_test_core(
416            HfRepoType::Model,
417            "test-user/test-repo",
418            "main",
419            "https://huggingface.co",
420        );
421        mock_client.set_xet_backed(&"00".repeat(32), 4);
422        mock_client.set_xet_token_expires_at(u64::MAX);
423        let reader = hf_reader(core, ctx, "xet-file.bin");
424
425        let ranges = [
426            BytesRange::new(0, Some(1)),
427            BytesRange::new(10, Some(1)),
428            BytesRange::new(20, Some(1)),
429        ];
430        let (r1, r2, r3) = futures::join!(
431            reader.open(ranges[0]),
432            reader.open(ranges[1]),
433            reader.open(ranges[2]),
434        );
435        r1?;
436        r2?;
437        r3?;
438
439        // Exactly one classifying resolve fires (see
440        // `test_concurrent_cold_opens_on_xet_file_share_one_group`), and its
441        // `Range` header must be exactly one of the three racing callers'
442        // own ranges -- not absent, and not some range not asked for by any
443        // of them.
444        let captured = mock_client
445            .get_captured_classify_range_header()
446            .expect("the shared classifying resolve must carry a Range header");

448            ranges.iter().any(|r| r.to_header() == captured),

453        Ok(())
454    }
