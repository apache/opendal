33/// Access DeepSeek's 3FS (Fire-Flyer File System) through USRBIO for data and
34/// the mounted filesystem for metadata.
35///
36/// Requires Linux, a running 3FS FUSE client, and its compatible USRBIO shared
37/// library. The root must already exist. Reads retain shared memory while space
38/// permits; writes copy into shared memory. See the crate documentation for
39/// memory limits and failure semantics.
40#[derive(Debug, Default)]
41pub struct Hf3fsBuilder {
42    pub(crate) config: Hf3fsConfig,
43}
44
45impl Hf3fsBuilder {
46    /// Set the existing absolute directory used as the storage root.
47    pub fn root(mut self, root: &str) -> Self {
48        self.config.root = Some(root.into());
49        self
50    }
51
52    /// Set the path to the compatible `libhf3fs_api_shared.so` library.
53    /// Defaults to the system dynamic loader's search path.
54    pub fn library_path(mut self, path: &str) -> Self {
55        self.config.library_path = Some(path.into());
56        self
57    }
58
59    /// Set the positive maximum number of in-flight requests per ring.
60    /// Defaults to 32. Each operator owns one read ring and one write ring.
61    pub fn queue_depth(mut self, depth: usize) -> Self {
62        self.config.queue_depth = Some(depth);
63        self
64    }
65
66    /// Set the positive maximum request size in bytes. Defaults to 1 MiB.
67    /// Shared memory usage is approximately `3 * queue_depth * block_size`.
68    pub fn block_size(mut self, size: usize) -> Self {
69        self.config.block_size = Some(size);
70        self
71    }
72}
73
74impl Builder for Hf3fsBuilder {
75    type Config = Hf3fsConfig;
76
77    fn build(self) -> Result<impl Service> {
78        let depth = self.config.queue_depth.unwrap_or(32);
79        let block_size = self.config.block_size.unwrap_or(1024 * 1024);
80        if depth == 0
81            || depth >= i32::MAX as usize
82            || block_size == 0
83            || depth
84                .checked_mul(3)
85                .and_then(|n| n.checked_mul(block_size))
86                .is_none_or(|n| n > isize::MAX as usize)
87        {
88            return Err(Error::new(
89                ErrorKind::ConfigInvalid,
90                "invalid USRBIO queue depth or block size",
91            ));
92        }
93        let root = self
94            .config
95            .root
96            .as_deref()
97            .filter(|p| Path::new(p).is_absolute())
98            .ok_or_else(|| {
99                Error::new(ErrorKind::ConfigInvalid, "an absolute 3FS root is required")
100            })?;
101        let root = Path::new(root).canonicalize().map_err(new_std_io_error)?;
102        if !root.is_dir() {
103            return Err(Error::new(
104                ErrorKind::ConfigInvalid,
105                "3FS root must be a directory",
106            ));
107        }
108        let root_str = root
109            .to_str()
110            .ok_or_else(|| Error::new(ErrorKind::ConfigInvalid, "3FS root must be valid UTF-8"))?;
111        let api = Api::load(Path::new(
112            self.config
113                .library_path
114                .as_deref()
115                .unwrap_or("libhf3fs_api_shared.so"),
116        ))?;
117        let c_root = CString::new(root.as_os_str().as_bytes())
118            .map_err(|e| Error::new(ErrorKind::ConfigInvalid, "root contains NUL").set_source(e))?;
119        let mount = api.mount_point(&c_root)?;
120        let read = IoDriver::start(api.clone(), mount.clone(), depth, block_size, true)?;
121        let write = IoDriver::start(api.clone(), mount, depth, block_size, false)?;
122        let metadata = Arc::new(opendal_service_fs::Fs::default().root(root_str).build()?);
123        Ok(Hf3fsBackend {
124            core: Arc::new(Hf3fsCore {
125                root,
126                api,
127                read,
128                write,
129            }),
130            metadata,
131        })
132    }
133}

135pub(crate) struct Hf3fsCore {
136    pub root: PathBuf,
137    pub api: Arc<Api>,
138    pub read: IoDriver,
139    pub write: IoDriver,
140}
141
142impl Hf3fsCore {
143    pub fn path(&self, path: &str) -> Result<PathBuf> {
144        confined_join(&self.root, path)
145    }
146}
147
148struct Hf3fsBackend {
149    core: Arc<Hf3fsCore>,
150    metadata: Servicer,
151}
152
153impl std::fmt::Debug for Hf3fsBackend {
154    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
155        f.debug_struct("Hf3fsBackend")
156            .field("root", &self.core.root)
157            .finish_non_exhaustive()
158    }
159}
160
161impl Service for Hf3fsBackend {
162    type Reader = oio::PositionReader<Hf3fsReader>;
163    type Writer = Hf3fsWriter;
164    type Lister = oio::Lister;
165    type Deleter = oio::Deleter;
166    type Copier = ();
167    type Composer = ();
168
169    fn info(&self) -> ServiceInfo {
170        ServiceInfo::new(HF3FS_SCHEME, self.core.root.to_string_lossy(), "")
171    }
172
173    fn capability(&self) -> Capability {
174        Capability {
175            stat: true,
176            read: true,
177            write: true,
178            write_can_empty: true,
179            write_can_multi: true,
180            create_dir: true,
181            list: true,
182            delete: true,
183            delete_with_recursive: true,
184            rename: true,
185            shared: true,
186            ..Default::default()
187        }
188    }
189
190    fn read(&self, _: &OperationContext, path: &str, _: OpRead) -> Result<Self::Reader> {
191        Ok(oio::PositionReader::new(Hf3fsReader {
192            core: self.core.clone(),
193            path: self.core.path(path)?,
194        })
195        .with_max_buf_size(self.core.read.block_size))
196    }
197
198    fn write(&self, ctx: &OperationContext, path: &str, args: OpWrite) -> Result<Self::Writer> {
199        Ok(Hf3fsWriter::new(
200            self.core.clone(),
201            self.core.path(path)?,
202            ctx.executor().clone(),
203            args,
204        ))
205    }
206
207    async fn stat(&self, ctx: &OperationContext, path: &str, args: OpStat) -> Result<RpStat> {
208        self.metadata.stat(ctx, path, args).await
209    }
210
211    async fn create_dir(
212        &self,
213        ctx: &OperationContext,
214        path: &str,
215        args: OpCreateDir,
216    ) -> Result<RpCreateDir> {
217        self.metadata.create_dir(ctx, path, args).await
218    }
219
220    fn list(&self, ctx: &OperationContext, path: &str, args: OpList) -> Result<Self::Lister> {
221        self.metadata.list(ctx, path, args)
222    }
223
224    fn delete(&self, ctx: &OperationContext) -> Result<Self::Deleter> {
225        self.metadata.delete(ctx)
226    }
227
228    fn copy(&self, _: &OperationContext, _: &str, _: &str, _: OpCopy) -> Result<Self::Copier> {
229        Err(Error::new(
230            ErrorKind::Unsupported,
231            "3FS native copy is unsupported",
232        ))
233    }
234
235    async fn presign(&self, _: &OperationContext, _: &str, _: OpPresign) -> Result<RpPresign> {
236        Err(Error::new(
237            ErrorKind::Unsupported,
238            "3FS presign is unsupported",
239        ))
240    }
241
242    async fn rename(
243        &self,
244        ctx: &OperationContext,
245        from: &str,
246        to: &str,
247        args: OpRename,
248    ) -> Result<RpRename> {
249        self.metadata.rename(ctx, from, to, args).await
250    }
251}

261    #[test]
262    fn builder_rejects_invalid_config_values() {
263        let dir = tempfile::tempdir().unwrap();
264        let root = dir.path().to_str().unwrap();
265
266        let err = Hf3fsBuilder::default().build().err().unwrap();
267        assert_eq!(err.kind(), ErrorKind::ConfigInvalid);
268
269        let err = Hf3fsBuilder::default()
270            .root("relative/path")
271            .build()
272            .err()
273            .unwrap();
274        assert_eq!(err.kind(), ErrorKind::ConfigInvalid);
275
276        let file = dir.path().join("not-a-dir");
277        std::fs::write(&file, b"data").unwrap();
278        let err = Hf3fsBuilder::default()
279            .root(file.to_str().unwrap())
280            .build()
281            .err()
282            .unwrap();
283        assert_eq!(err.kind(), ErrorKind::ConfigInvalid);
284
285        for builder in [
286            Hf3fsBuilder::default().root(root).queue_depth(0),
287            Hf3fsBuilder::default()
288                .root(root)
289                .queue_depth(i32::MAX as usize),
290            Hf3fsBuilder::default().root(root).block_size(0),
291            Hf3fsBuilder::default()
292                .root(root)
293                .queue_depth(4)
294                .block_size(isize::MAX as usize / 2),
295        ] {
296            let err = builder.build().err().unwrap();
297            assert_eq!(err.kind(), ErrorKind::ConfigInvalid);
298        }
299    }

317    #[test]
318    fn config_from_uri_parses_root_and_options() {
319        let uri = OperatorUri::new(
320            "hf3fs:///mnt/3fs/data",
321            [
322                ("queue_depth".to_string(), "16".to_string()),
323                ("block_size".to_string(), "65536".to_string()),
324                (
325                    "library_path".to_string(),
326                    "/opt/3fs/libhf3fs_api_shared.so".to_string(),
327                ),
328            ],
329        )
330        .unwrap();
331        let cfg = Hf3fsConfig::from_uri(&uri).unwrap();
332        assert_eq!(cfg.root.as_deref(), Some("/mnt/3fs/data"));
333        assert_eq!(cfg.queue_depth, Some(16));
334        assert_eq!(cfg.block_size, Some(65536));
335        assert_eq!(
336            cfg.library_path.as_deref(),
337            Some("/opt/3fs/libhf3fs_api_shared.so")
338        );
339    }

341    #[test]
342    fn core_path_confines_paths_to_root() {
343        let dir = tempfile::tempdir().unwrap();
344        let api = Api::noop();
345        let (read, _read_rx, _) = IoDriver::for_test(api.clone(), 2, 4096, true);
346        let (write, _write_rx, _) = IoDriver::for_test(api.clone(), 2, 4096, false);
347        let core = Hf3fsCore {
348            root: dir.path().to_path_buf(),
349            api,
350            read,
351            write,
352        };
353
354        assert_eq!(core.path("").unwrap(), dir.path());
355        assert_eq!(core.path("/").unwrap(), dir.path());
356        assert_eq!(core.path("a/b/c").unwrap(), dir.path().join("a/b/c"));
357        assert_eq!(core.path("a/b/").unwrap(), dir.path().join("a/b"));
358
359        for invalid in ["../escape", "a/../../escape", "/absolute"] {
360            let err = core.path(invalid).unwrap_err();
361            assert_eq!(err.kind(), ErrorKind::NotFound, "path: {invalid}");
362        }
363    }
