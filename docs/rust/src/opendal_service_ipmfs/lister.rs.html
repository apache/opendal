49impl oio::PageList for IpmfsLister {
50    async fn next_page(&self, ctx: &mut oio::PageContext) -> Result<()> {
51        let resp = self.core.ipmfs_ls(&self.ctx, &self.path).await?;
52
53        if resp.status() != StatusCode::OK {
54            let err = parse_error(ErrorContext::new(ServiceOperation("FilesLs")), resp);
55            if matches!(err.kind(), ErrorKind::NotFound) {
56                // treat as empty listing
57                ctx.done = true;
58                return Ok(());
59            }
60            return Err(err);
61        }
62
63        let bs = resp.into_body();
64        let entries_body: IpfsLsResponse =
65            serde_json::from_reader(bs.reader()).map_err(new_json_deserialize_error)?;
66
67        let trimmed = self.path.trim_matches('/');
68        let prefix = if trimmed.is_empty() {
69            String::new()
70        } else {
71            format!("{trimmed}/")
72        };
73
74        // Add current directory entry when processing the first page
75        if ctx.token.is_empty() && !ctx.done {
76            let path = if prefix.is_empty() { "/" } else { &prefix };
77            ctx.entries
78                .push_back(oio::Entry::new(path, MetadataBuilder::dir().build()));
79        }
80
81        // Mark dir stream has been consumed.
82        ctx.done = true;
83
84        for object in entries_body.entries.unwrap_or_default() {
85            let (path, meta) = match object.mode() {
86                EntryMode::FILE => (
87                    format!("{prefix}{}", object.name),
88                    MetadataBuilder::file(object.size).build(),
89                ),
90                EntryMode::DIR => (
91                    format!("{prefix}{}/", object.name),
92                    MetadataBuilder::dir().build(),
93                ),
94                EntryMode::Unknown => (
95                    format!("{prefix}{}", object.name),
96                    MetadataBuilder::unknown().build(),
97                ),
98            };
99
100            ctx.entries.push_back(oio::Entry::new(&path, meta));
101        }
102
103        Ok(())
104    }
105}

107#[derive(Deserialize, Default, Debug)]
108#[serde(default)]
109struct IpfsLsResponseEntry {
110    #[serde(rename = "Name")]
111    name: String,
112    #[serde(rename = "Type")]
113    file_type: i64,
114    #[serde(rename = "Size")]
115    size: u64,
116}
117
118impl IpfsLsResponseEntry {
119    /// ref: <https://github.com/ipfs/specs/blob/main/UNIXFS.md#data-format>
120    ///
121    /// ```protobuf
122    /// enum DataType {
123    ///     Raw = 0;
124    ///     Directory = 1;
125    ///     File = 2;
126    ///     Metadata = 3;
127    ///     Symlink = 4;
128    ///     HAMTShard = 5;
129    /// }
130    /// ```
131    fn mode(&self) -> EntryMode {
132        match &self.file_type {
133            1 => EntryMode::DIR,
134            0 | 2 => EntryMode::FILE,
135            _ => EntryMode::Unknown,
136        }
137    }
138}
139
140#[derive(Deserialize, Default, Debug)]
141#[serde(default)]
142struct IpfsLsResponse {
143    #[serde(rename = "Entries")]
144    entries: Option<Vec<IpfsLsResponseEntry>>,
145}

161    struct StaticLsTransport {
162        status: StatusCode,
163        body: String,
164    }
165
166    impl HttpTransport for StaticLsTransport {
167        async fn fetch(&self, _req: Request<Buffer>) -> Result<Response<HttpBody>> {
168            let body = Buffer::from(self.body.clone());
169            let len = body.len() as u64;
170            Ok(Response::builder()
171                .status(self.status)
172                .body(HttpBody::new(stream::iter([Ok(body)]), Some(len)))
173                .unwrap())
174        }
175    }

177    fn make_lister(root: &str, path: &str, status: StatusCode, body: &str) -> IpmfsLister {
178        let core = Arc::new(IpmfsCore {
179            info: ServiceInfo::new(IPMFS_SCHEME, root, ""),
180            capability: Capability {
181                list: true,
182                ..Default::default()
183            },
184            root: root.to_string(),
185            endpoint: "http://127.0.0.1:5001".to_string(),
186        });
187        let ctx =
188            OperationContext::new().with_http_transport(HttpTransporter::new(StaticLsTransport {
189                status,
190                body: body.to_string(),
191            }));
192        IpmfsLister::new(core, ctx, path)
193    }

195    #[tokio::test]
196    async fn test_list_entry_modes_and_metadata() -> Result<()> {
197        tokio::time::timeout(Duration::from_secs(5), async {
198            let body = r#"{
199                "Entries": [
200                    {"Name": "raw_file", "Type": 0, "Size": 12},
201                    {"Name": "subdir", "Type": 1, "Size": 0},
202                    {"Name": "unixfs_file", "Type": 2, "Size": 34},
203                    {"Name": "meta_node", "Type": 3, "Size": 5},
204                    {"Name": "symlink_entry", "Type": 4, "Size": 7},
205                    {"Name": "hamt_dir", "Type": 5, "Size": 0},
206                    {"Name": "future_type", "Type": 99, "Size": 9}
207                ]
208            }"#;
209
210            let op =
211                Operator::new(IpmfsBuilder::default().endpoint("http://127.0.0.1:5001"))?
212                    .with_context(OperationContext::new().with_http_transport(
213                        HttpTransporter::new(StaticLsTransport {
214                            status: StatusCode::OK,
215                            body: body.to_string(),
216                        }),
217                    ));
218
219            let entries = op.list("/").await?;
220            let got: Vec<(&str, EntryMode, u64)> = entries
221                .iter()
222                .map(|e| (e.path(), e.metadata().mode(), e.metadata().content_length()))
223                .collect();
224
225            assert_eq!(
226                got,
227                vec![
228                    ("/", EntryMode::DIR, 0),
229                    ("raw_file", EntryMode::FILE, 12),
230                    ("subdir/", EntryMode::DIR, 0),
231                    ("unixfs_file", EntryMode::FILE, 34),
232                    ("meta_node", EntryMode::Unknown, 0),
233                    ("symlink_entry", EntryMode::Unknown, 0),
234                    ("hamt_dir", EntryMode::Unknown, 0),
235                    ("future_type", EntryMode::Unknown, 0),
236                ]
237            );
238            Ok(())
239        })
240        .await
241        .expect("test timed out")
242    }

244    #[tokio::test]
245    async fn test_list_non_trailing_slash_path() -> Result<()> {
246        tokio::time::timeout(Duration::from_secs(5), async {
247            let body = r#"{
248                "Entries": [
249                    {"Name": "a", "Type": 0, "Size": 5},
250                    {"Name": "nested", "Type": 1, "Size": 0}
251                ]
252            }"#;
253
254            for root in ["/", "/mfs/"] {
255                for input_path in ["sub", "sub/"] {
256                    let lister = make_lister(root, input_path, StatusCode::OK, body);
257                    let mut page_ctx = oio::PageContext {
258                        done: false,
259                        token: String::new(),
260                        entries: VecDeque::new(),
261                    };
262
263                    lister.next_page(&mut page_ctx).await?;
264                    assert!(page_ctx.done);
265
266                    let got: Vec<(String, EntryMode, u64)> = page_ctx
267                        .entries
268                        .iter()
269                        .map(|e| {
270                            (
271                                e.path().to_string(),
272                                e.mode(),
273                                e.metadata().content_length(),
274                            )
275                        })
276                        .collect();
277
278                    assert_eq!(
279                        got,
280                        vec![
281                            ("sub/".to_string(), EntryMode::DIR, 0),
282                            ("sub/a".to_string(), EntryMode::FILE, 5),
283                            ("sub/nested/".to_string(), EntryMode::DIR, 0),
284                        ],
285                        "root={root} input_path={input_path}"
286                    );
287                }
288            }
289
290            Ok(())
291        })
292        .await
293        .expect("test timed out")
294    }
