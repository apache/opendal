39#[derive(Debug)]
40pub enum SocketStream {
41    Tcp(TcpStream),
42    Tls(Box<TlsStream<TcpStream>>),
43    #[cfg(unix)]
44    Unix(UnixStream),
45}
46
47impl SocketStream {
48    pub async fn connect_tcp(addr_str: &str) -> io::Result<Self> {
49        let stream = TcpStream::connect(addr_str).await?;
50        Ok(SocketStream::Tcp(stream))
51    }
52
53    #[cfg(unix)]
54    pub async fn connect_unix(path: &str) -> io::Result<Self> {
55        let stream = UnixStream::connect(path).await?;
56        Ok(SocketStream::Unix(stream))
57    }
58}

72    }

108/// Endpoint for memcached connection.
109#[derive(Clone, Debug)]
110pub enum Endpoint {
111    Tcp(String), // host:port
112    Tls {
113        address: String,
114        server_name: ServerName<'static>,
115        config: Arc<ClientConfig>,
116    },
117    #[cfg(unix)]
118    Unix(String), // socket path
119}
120
121impl Endpoint {
122    pub(super) fn tls(address: String, host: &str) -> Result<Self> {
123        let server_name =
124            ServerName::try_from(host.trim_matches(['[', ']']).to_owned()).map_err(|err| {
125                Error::new(ErrorKind::ConfigInvalid, "invalid TLS server name").set_source(err)
126            })?;
127        let native = rustls_native_certs::load_native_certs();
128        let mut roots = RootCertStore::empty();
129        roots.add_parsable_certificates(native.certs);
130        if roots.is_empty() {
131            let mut err = Error::new(
132                ErrorKind::ConfigInvalid,
133                "no usable native root certificates were loaded",
134            );
135            if let Some(source) = native.errors.into_iter().next() {
136                err = err.set_source(source);
137            }
138            return Err(err);
139        }
140        let config =
141            ClientConfig::builder_with_provider(Arc::new(rustls::crypto::ring::default_provider()))
142                .with_safe_default_protocol_versions()
143                .map_err(|err| {
144                    Error::new(ErrorKind::ConfigInvalid, "TLS protocols are unavailable")
145                        .set_source(err)
146                })?
147                .with_root_certificates(roots)
148                .with_no_client_auth();
149        Ok(Self::Tls {
150            address,
151            server_name,
152            config: Arc::new(config),
153        })
154    }
155}

157/// A connection manager for `memcache_async::ascii::Protocol`.
158#[derive(Clone)]
159struct MemcacheConnectionManager {
160    endpoint: Endpoint,
161    username: Option<String>,
162    password: Option<String>,
163}
164
165impl MemcacheConnectionManager {
166    fn new(endpoint: Endpoint, username: Option<String>, password: Option<String>) -> Self {
167        Self {
168            endpoint,
169            username,
170            password,
171        }
172    }
173}
174
175impl ManageObject for MemcacheConnectionManager {
176    type Object = binary::Connection;
177    type Error = Error;
178
179    async fn create(&self) -> Result<Self::Object, Self::Error> {
180        let conn = match &self.endpoint {
181            Endpoint::Tcp(addr) => SocketStream::connect_tcp(addr)
182                .await
183                .map_err(new_std_io_error)?,
184            Endpoint::Tls {
185                address,
186                server_name,
187                config,
188            } => {
189                let stream = TcpStream::connect(address)
190                    .await
191                    .map_err(new_std_io_error)?;
192                let stream = TlsConnector::from(Arc::clone(config))
193                    .connect(server_name.clone(), stream)
194                    .await
195                    .map_err(new_std_io_error)?;
196                SocketStream::Tls(Box::new(stream))
197            }
198            #[cfg(unix)]
199            Endpoint::Unix(path) => SocketStream::connect_unix(path)
200                .await
201                .map_err(new_std_io_error)?,
202        };
203
204        let mut conn = binary::Connection::new(conn);
205
206        if let (Some(username), Some(password)) = (self.username.as_ref(), self.password.as_ref()) {
207            conn.auth(username, password).await?;
208        }
209        Ok(conn)
210    }
211
212    async fn is_recyclable(
213        &self,
214        o: &mut Self::Object,
215        _: &ObjectStatus,
216    ) -> Result<(), Self::Error> {
217        if o.is_poisoned() {
218            return Err(Error::new(
219                ErrorKind::Unexpected,
220                "memcached connection is poisoned",
221            ));
222        }
223        o.version().await.map(|_| ())
224    }
225}

228pub struct MemcachedCore {
229    default_ttl: Option<Duration>,
230    conn: Arc<bounded::Pool<MemcacheConnectionManager>>,
231}
232
233/// memcached reads an expiration larger than 30 days as an absolute Unix
234/// timestamp rather than as an offset from now, so a longer TTL sent verbatim
235/// lands in 1970 and the server stores the item as already expired.
236///
237/// See the `Expiration times` section of memcached's protocol.txt.
238pub(super) const MAX_RELATIVE_EXPIRATION_SECS: u64 = 60 * 60 * 24 * 30;
239
240impl MemcachedCore {
241    pub fn new(
242        endpoint: Endpoint,
243        username: Option<String>,
244        password: Option<String>,
245        default_ttl: Option<Duration>,
246        connection_pool_max_size: Option<usize>,
247    ) -> Self {
248        let conn = bounded::Pool::new(
249            bounded::PoolConfig::new(connection_pool_max_size.unwrap_or(10)),
250            MemcacheConnectionManager::new(endpoint, username, password),
251        );
252
253        Self { default_ttl, conn }
254    }
255
256    async fn conn(&self) -> Result<bounded::Object<MemcacheConnectionManager>> {
257        let fut = self.conn.get();
258
259        tokio::select! {
260            _ = tokio::time::sleep(Duration::from_secs(10)) => {
261                Err(Error::new(ErrorKind::Unexpected, "connection request: timeout").set_temporary())
262            }
263            result = fut => match result {
264                Ok(conn) => Ok(conn),
265                Err(err) => Err(err),
266            }
267        }
268    }
269
270    pub async fn get(&self, key: &str) -> Result<Option<Buffer>> {
271        let mut conn = self.conn().await?;
272        let result = conn.get(&percent_encode_path(key)).await?;
273        Ok(result.map(Buffer::from))
274    }
275
276    pub async fn set(&self, key: &str, value: Buffer, expires: Option<Duration>) -> Result<()> {
277        let expiration = resolve_expiration(expires, self.default_ttl)?;
278        let mut conn = self.conn().await?;
279        conn.set(&percent_encode_path(key), &value.to_vec(), expiration)
280            .await
281    }
282
283    pub async fn delete(&self, key: &str) -> Result<()> {
284        let mut conn = self.conn().await?;
285
286        conn.delete(&percent_encode_path(key)).await
287    }
288}
289
290fn resolve_expiration(expires: Option<Duration>, default_ttl: Option<Duration>) -> Result<u32> {
291    if let Some(expires) = expires {
292        return encode_expiration(expires);
293    }
294
295    match default_ttl {
296        Some(ttl) if !ttl.is_zero() => encode_expiration(ttl),
297        _ => Ok(0),
298    }
299}
300
301fn encode_expiration(expires: Duration) -> Result<u32> {
302    if expires.is_zero() {
303        return Err(Error::new(
304            ErrorKind::ConfigInvalid,
305            "expires must be positive",
306        ));
307    }
308    let seconds = expires.as_nanos().div_ceil(1_000_000_000);
309    let expiration = if seconds <= u128::from(MAX_RELATIVE_EXPIRATION_SECS) {
310        seconds
311    } else {
312        let now = SystemTime::now()
313            .duration_since(UNIX_EPOCH)
314            .map_err(|err| {
315                Error::new(
316                    ErrorKind::Unexpected,
317                    "system clock is before the Unix epoch",
318                )
319                .set_source(err)
320            })?;
321        u128::from(now.as_secs()) + seconds
322    };
323    u32::try_from(expiration).map_err(|err| {
324        Error::new(
325            ErrorKind::ConfigInvalid,
326            "expires is too large for Memcached",
327        )
328        .set_source(err)
329    })
330}

338    fn resolve_memcached_expiration() {
339        assert_eq!(resolve_expiration(None, None).unwrap(), 0);
340        assert_eq!(resolve_expiration(None, Some(Duration::ZERO)).unwrap(), 0);
341
342        for ttl in [
343            Duration::from_nanos(1),
344            Duration::from_millis(1),
345            Duration::from_millis(999),
346        ] {
347            assert_eq!(resolve_expiration(None, Some(ttl)).unwrap(), 1);
348        }
349
350        assert_eq!(
351            resolve_expiration(None, Some(Duration::from_secs(1))).unwrap(),
352            1
353        );
354        assert_eq!(
355            resolve_expiration(None, Some(Duration::from_secs(60))).unwrap(),
356            60
357        );
358        assert_eq!(
359            resolve_expiration(None, Some(Duration::from_millis(1500))).unwrap(),
360            2
361        );
362        assert_eq!(
363            resolve_expiration(Some(Duration::from_secs(60)), Some(Duration::from_nanos(1)))
364                .unwrap(),
365            60
366        );
367
368        let err =
369            resolve_expiration(Some(Duration::ZERO), Some(Duration::from_secs(60))).unwrap_err();
370        assert_eq!(err.kind(), ErrorKind::ConfigInvalid);

447    async fn read_binary_request(
448        stream: &mut TcpStream,
449    ) -> std::io::Result<(binary::PacketHeader, Vec<u8>)> {
450        use tokio::io::AsyncReadExt;
451
452        let mut header_buf = [0u8; 24];
453        stream.read_exact(&mut header_buf).await?;
454        let header = binary::PacketHeader::from_bytes(header_buf);
455        let mut body = vec![0u8; header.total_body_length as usize];
456        stream.read_exact(&mut body).await?;
457        Ok((header, body))
458    }

460    async fn write_binary_response(
461        stream: &mut TcpStream,
462        opcode: u8,
463        status: u16,
464        opaque: u32,
465        extras: &[u8],
466        value: &[u8],
467    ) -> std::io::Result<()> {
468        use tokio::io::AsyncWriteExt;
469
470        let header = binary::PacketHeader {
471            magic: binary::Magic::Response as u8,
472            opcode,
473            extras_length: extras.len() as u8,
474            vbucket_id_or_status: status,
475            total_body_length: (extras.len() + value.len()) as u32,
476            opaque,
477            ..Default::default()
478        };
479        stream.write_all(&header.to_bytes()).await?;
480        stream.write_all(extras).await?;
481        stream.write_all(value).await?;
482        stream.flush().await
483    }

485    #[tokio::test]
486    async fn cancelled_pooled_request_discards_socket_and_prevents_cross_key_leak() {
487        use std::sync::atomic::{AtomicUsize, Ordering};
488        use tokio::sync::oneshot;
489
490        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
491        let addr = listener.local_addr().unwrap().to_string();
492        let accepted_conns = Arc::new(AtomicUsize::new(0));
493        let accepted_conns_server = Arc::clone(&accepted_conns);
494
495        let (req1_received_tx, req1_received_rx) = oneshot::channel::<()>();
496        let (release_reply1_tx, release_reply1_rx) = oneshot::channel::<()>();
497        let (reply1_written_tx, reply1_written_rx) = oneshot::channel::<()>();
498
499        let server_task = tokio::spawn(async move {
500            // Connection #1: receives GET /foo, waits until cancelled, then writes stale reply.
501            let (mut conn1, _) = listener.accept().await.unwrap();
502            accepted_conns_server.fetch_add(1, Ordering::SeqCst);
503            let (req1, body1) = read_binary_request(&mut conn1).await.unwrap();
504            assert_eq!(req1.opcode, binary::Opcode::Get as u8);
505            assert_eq!(body1, b"/foo");
506            let _ = req1_received_tx.send(());
507            let _ = release_reply1_rx.await;
508            // Best-effort write: the client already dropped the future, so the stale reply
509            // lands in the kernel socket buffer while conn1 sits idle in the pool.
510            let _ = write_binary_response(
511                &mut conn1,
512                binary::Opcode::Get as u8,
513                binary::constants::OK_STATUS,
514                req1.opaque,
515                &[0, 0, 0, 0],
516                b"secret-foo-value",
517            )
518            .await;
519            let _ = reply1_written_tx.send(());
520
521            // Connection #2: pool must discard poisoned conn1 and open a fresh connection.
522            let (mut conn2, _) = listener.accept().await.unwrap();
523            accepted_conns_server.fetch_add(1, Ordering::SeqCst);
524            let (req2, body2) = read_binary_request(&mut conn2).await.unwrap();
525            assert_eq!(req2.opcode, binary::Opcode::Get as u8);
526            assert_eq!(body2, b"/bar");
527            write_binary_response(
528                &mut conn2,
529                binary::Opcode::Get as u8,
530                binary::constants::OK_STATUS,
531                req2.opaque,
532                &[0, 0, 0, 0],
533                b"value-of-bar",
534            )
535            .await
536            .unwrap();
537        });
538
539        let core = MemcachedCore::new(Endpoint::Tcp(addr), None, None, None, Some(1));
540
541        // Start GET /foo and deterministically cancel it mid-flight once the request is on the wire.
542        let mut req1_received_rx = req1_received_rx;
543        tokio::select! {
544            _ = core.get("/foo") => panic!("first request should have been cancelled before replying"),
545            res = &mut req1_received_rx => res.unwrap(),
546        }
547
548        // Let the server write the delayed /foo reply into conn1's socket buffer.
549        release_reply1_tx.send(()).unwrap();
550        reply1_written_rx.await.unwrap();
551
552        // Next caller must never observe VERSION bytes or stale /foo data.
553        let got = core
554            .get("/bar")
555            .await
556            .unwrap()
557            .expect("expected /bar value");
558        assert_eq!(got.to_bytes().as_ref(), b"value-of-bar");
559        assert_eq!(accepted_conns.load(Ordering::SeqCst), 2);
560
561        server_task.await.unwrap();
562    }

564    #[tokio::test]
565    async fn pooled_connection_recovers_after_server_error_with_body() {
566        use std::sync::atomic::{AtomicUsize, Ordering};
567
568        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
569        let addr = listener.local_addr().unwrap().to_string();
570        let accepted_conns = Arc::new(AtomicUsize::new(0));
571        let accepted_conns_server = Arc::clone(&accepted_conns);
572
573        let server_task = tokio::spawn(async move {
574            let (mut conn, _) = listener.accept().await.unwrap();
575            accepted_conns_server.fetch_add(1, Ordering::SeqCst);
576
577            // 1. SET /big -> 0x0003 Value too large with ASCII error body.
578            let (set_req, _) = read_binary_request(&mut conn).await.unwrap();
579            assert_eq!(set_req.opcode, binary::Opcode::Set as u8);
580            write_binary_response(
581                &mut conn,
582                binary::Opcode::Set as u8,
583                0x0003,
584                set_req.opaque,
585                &[],
586                b"Object too large for cache",
587            )
588            .await
589            .unwrap();
590
591            // 2. Pool recycles the same connection via VERSION health check.
592            let (ver_req, _) = read_binary_request(&mut conn).await.unwrap();
593            assert_eq!(ver_req.opcode, binary::Opcode::Version as u8);
594            write_binary_response(
595                &mut conn,
596                binary::Opcode::Version as u8,
597                binary::constants::OK_STATUS,
598                ver_req.opaque,
599                &[],
600                b"1.6.29",
601            )
602            .await
603            .unwrap();
604
605            // 3. Subsequent GET /ok on the same connection returns clean value.
606            let (get_req, get_body) = read_binary_request(&mut conn).await.unwrap();
607            assert_eq!(get_req.opcode, binary::Opcode::Get as u8);
608            assert_eq!(get_body, b"/ok");
609            write_binary_response(
610                &mut conn,
611                binary::Opcode::Get as u8,
612                binary::constants::OK_STATUS,
613                get_req.opaque,
614                &[0, 0, 0, 0],
615                b"ok-value",
616            )
617            .await
618            .unwrap();
619        });
620
621        let core = MemcachedCore::new(Endpoint::Tcp(addr), None, None, None, Some(1));
622
623        let err = core
624            .set("/big", Buffer::from("oversized"), None)
625            .await
626            .unwrap_err();
627        assert_eq!(err.kind(), ErrorKind::Unexpected);
628
629        let got = core.get("/ok").await.unwrap().expect("expected /ok value");
630        assert_eq!(got.to_bytes().as_ref(), b"ok-value");
631        assert_eq!(accepted_conns.load(Ordering::SeqCst), 1);
632
633        server_task.await.unwrap();
634    }
