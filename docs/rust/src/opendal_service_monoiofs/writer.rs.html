31enum WriterRequest {
32    Write {
33        pos: u64,
34        buf: Bytes,
35        tx: oneshot::Sender<Result<()>>,
36    },
37    Close {
38        tx: oneshot::Sender<Result<monoio::fs::Metadata>>,
39    },
40}
41
42pub struct MonoiofsWriter {
43    core: Arc<MonoiofsCore>,
44    tx: Option<mpsc::UnboundedSender<WriterRequest>>,
45    pos: u64,
46}
47
48impl MonoiofsWriter {
49    pub async fn new(core: Arc<MonoiofsCore>, path: PathBuf, append: bool) -> Result<Self> {
50        let (open_result_tx, open_result_rx) = oneshot::channel();
51        let (tx, rx) = mpsc::unbounded();
52        core.spawn(move || Self::worker_entrypoint(path, append, rx, open_result_tx))
53            .await;
54        let pos = core.unwrap(open_result_rx.await)?;
55        Ok(Self {
56            core,
57            tx: Some(tx),
58            pos,
59        })
60    }
61
62    /// entrypoint of worker task that runs in context of monoio
63    async fn worker_entrypoint(
64        path: PathBuf,
65        append: bool,
66        mut rx: mpsc::UnboundedReceiver<WriterRequest>,
67        open_result_tx: oneshot::Sender<Result<u64>>,
68    ) {
69        let result = OpenOptions::new()
70            .write(true)
71            .create(true)
72            .append(append)
73            .truncate(!append)
74            .open(path)
75            .await;
76        // [`monoio::fs::File`] is non-Send, hence it is kept within
77        // worker thread
78        let file = match result {
79            Ok(file) => file,
80            Err(e) => {
81                // discard the result if send failed due to MonoiofsWriter::new
82                // cancelled since we are going to exit anyway
83                let _ = open_result_tx.send(Err(new_std_io_error(e)));
84                return;
85            }
86        };
87        // Linux applies O_APPEND to positional writes. Elsewhere use EOF as the offset.
88        let initial_pos = if append && !cfg!(target_os = "linux") {
89            match file.metadata().await {
90                Ok(meta) => meta.len(),
91                Err(e) => {
92                    let _ = file.close().await;
93                    let _ = open_result_tx.send(Err(new_std_io_error(e)));
94                    return;
95                }
96            }
97        } else {
98            0
99        };
100        if open_result_tx.send(Ok(initial_pos)).is_err() {
101            // MonoiofsWriter::new is cancelled, exit worker task
102            let _ = file.close().await;
103            return;
104        }
105
106        while let Ok(req) = rx.recv().await {
107            match req {
108                WriterRequest::Write { pos, buf, tx } => {
109                    let (result, _) = file.write_all_at(buf, pos).await;
110                    let _ = tx.send(result.map_err(new_std_io_error));
111                }
112                WriterRequest::Close { tx } => {
113                    let result = match file.sync_all().await {
114                        Ok(()) => file.metadata().await,
115                        Err(err) => Err(err),
116                    };
117                    let _ = file.close().await;
118                    let _ = tx.send(result.map_err(new_std_io_error));
119                    return;
120                }
121            }
122        }
123        let _ = file.close().await;
124    }
125}

127impl oio::Write for MonoiofsWriter {
128    /// Send write request to worker thread and wait for result. Actual
129    /// write happens in [`MonoiofsWriter::worker_entrypoint`] running
130    /// on worker thread.
131    async fn write(&mut self, mut bs: Buffer) -> Result<()> {
132        let Some(tx) = self.tx.as_ref() else {
133            return Err(Error::new(ErrorKind::Unexpected, "writer has been closed"));
134        };
135        while bs.has_remaining() {
136            let buf = bs.current();
137            let n = buf.len();
138            let (resp_tx, resp_rx) = oneshot::channel();
139            self.core.unwrap(tx.send(WriterRequest::Write {
140                pos: self.pos,
141                buf,
142                tx: resp_tx,
143            }));
144            self.core.unwrap(resp_rx.await)?;
145            self.pos += n as u64;
146            bs.advance(n);
147        }
148        Ok(())
149    }
150
151    /// Send close request to worker thread and wait for result. Actual
152    /// close happens in [`MonoiofsWriter::worker_entrypoint`] running
153    /// on worker thread.
154    async fn close(&mut self) -> Result<Metadata> {
155        let Some(tx) = self.tx.take() else {
156            return Err(Error::new(ErrorKind::Unexpected, "writer has been closed"));
157        };
158        let (resp_tx, resp_rx) = oneshot::channel();
159        self.core
160            .unwrap(tx.send(WriterRequest::Close { tx: resp_tx }));
161        let file_meta = self.core.unwrap(resp_rx.await)?;
162
163        let mode = if file_meta.is_dir() {
164            EntryMode::DIR
165        } else if file_meta.is_file() {
166            EntryMode::FILE
167        } else {
168            EntryMode::Unknown
169        };
170        let mut meta = match mode {
171            EntryMode::FILE => MetadataBuilder::file(file_meta.len()),
172            EntryMode::DIR => MetadataBuilder::dir(),
173            EntryMode::Unknown => MetadataBuilder::unknown(),
174        };
175        meta.last_modified(Timestamp::try_from(
176            file_meta.modified().map_err(new_std_io_error)?,
177        )?);
178        Ok(meta.build())
179    }
180
181    async fn abort(&mut self) -> Result<()> {
182        Err(Error::new(
183            ErrorKind::Unsupported,
184            "Monoiofs doesn't support abort",
185        ))
186    }
187}

196    #[tokio::test]
197    async fn test_writer_use_after_close_returns_unexpected_error() {
198        let temp_dir = tempfile::tempdir().unwrap();
199        let backend = Monoiofs::default()
200            .root(temp_dir.path().to_str().unwrap())
201            .build()
202            .unwrap();
203        let ctx = OperationContext::default();
204
205        // write + close + close must return Unexpected instead of panicking.
206        let mut writer = backend
207            .write(&ctx, "close_twice.txt", OpWrite::default())
208            .unwrap();
209        writer.write(Buffer::from("hello")).await.unwrap();
210        let meta = writer.close().await.unwrap();
211        assert_eq!(meta.content_length(), 5);
212
213        let err = writer.close().await.unwrap_err();
214        assert_eq!(err.kind(), ErrorKind::Unexpected);
215
216        // close + write must return Unexpected instead of panicking.
217        let mut writer = backend
218            .write(&ctx, "write_after_close.txt", OpWrite::default())
219            .unwrap();
220        writer.close().await.unwrap();
221
222        let err = writer.write(Buffer::from("late")).await.unwrap_err();
223        assert_eq!(err.kind(), ErrorKind::Unexpected);
224
225        // Worker thread must still be alive and serve subsequent operations.
226        let mut writer = backend
227            .write(&ctx, "after_errors.txt", OpWrite::default())
228            .unwrap();
229        writer.write(Buffer::from("ok")).await.unwrap();
230        let meta = writer.close().await.unwrap();
231        assert_eq!(meta.content_length(), 2);
232    }

234    #[tokio::test]
235    async fn test_writer_appends_without_overwriting() {
236        let temp_dir = tempfile::tempdir().unwrap();
237        let root = temp_dir.path();
238        let op = Operator::new(Monoiofs::default().root(root.to_str().unwrap())).unwrap();
239
240        let core = Arc::new(MonoiofsCore::new(root.to_path_buf(), 1, 1024));
241
242        // Appending to a new file starts at pos 0.
243        let mut new_writer = MonoiofsWriter::new(core.clone(), root.join("new"), true)
244            .await
245            .unwrap();
246        new_writer.write(Buffer::from("init")).await.unwrap();
247        new_writer.close().await.unwrap();
248        assert_eq!(op.read("new").await.unwrap().to_bytes(), b"init".as_slice());
249
250        op.write("f", "hello").await.unwrap();
251
252        let mut writer = MonoiofsWriter::new(core, root.join("f"), true)
253            .await
254            .unwrap();
255
256        writer
257            .write(Buffer::from(vec![
258                Bytes::from_static(b"wo"),
259                Bytes::from_static(b"rld"),
260            ]))
261            .await
262            .unwrap();
263        let meta = writer.close().await.unwrap();
264        assert_eq!(meta.content_length(), 10);
265
266        assert_eq!(
267            op.read("f").await.unwrap().to_bytes(),
268            b"helloworld".as_slice()
269        );
270
271        op.write_with("f", "!").append(true).await.unwrap();
272        assert_eq!(
273            op.read("f").await.unwrap().to_bytes(),
274            b"helloworld!".as_slice()
275        );
276    }
