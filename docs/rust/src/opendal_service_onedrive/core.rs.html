50// OneDrive returns 400 when try to access a dir with the POSIX special directory entries
51const SPECIAL_POSIX_ENTRIES: [&str; 3] = [".", "/", ""];
52
53// organizes a few core module functions
54impl OneDriveCore {
55    // OneDrive personal's base URL. `me` is an alias that represents the user's "Drive".
56    pub(crate) const DRIVE_ROOT_URL: &'static str =
57        "https://graph.microsoft.com/v1.0/me/drive/root";
58
59    /// Get a URL to an OneDrive item
60    pub(crate) fn onedrive_item_url(&self, path: &str, build_absolute_path: bool) -> String {
61        // OneDrive requires the root to be the same as `DRIVE_ROOT_URL`.
62        // For files under the root, the URL pattern becomes `https://graph.microsoft.com/v1.0/me/drive/root:<path>:`
63        if self.root == "/" && SPECIAL_POSIX_ENTRIES.contains(&path) {
64            Self::DRIVE_ROOT_URL.to_string()
65        } else {
66            // OneDrive returns 400 when try to access a folder with a ending slash
67            let absolute_path = if build_absolute_path {
68                let rooted_path = build_rooted_abs_path(&self.root, path);
69                rooted_path
70                    .strip_suffix('/')
71                    .unwrap_or(rooted_path.as_str())
72                    .to_string()
73            } else {
74                path.to_string()
75            };
76            format!(
77                "{}:{}",
78                Self::DRIVE_ROOT_URL,
79                percent_encode_path(&absolute_path),
80            )
81        }
82    }

110    /// Create a directory at path if not exist, return the metadata about the folder
111    ///
112    /// When the folder exist, this function works exactly the same as [`onedrive_get_stat_plain()`].
113    ///
114    /// * `path` - a relative folder path
115    pub(crate) async fn ensure_directory(
116        &self,
117        ctx: &OperationContext,
118        path: &str,
119    ) -> Result<OneDriveItem> {
120        let response = self.onedrive_get_stat_plain(ctx, path).await?;
121        let item: OneDriveItem = match response.status() {
122            StatusCode::OK => {
123                let bytes = response.into_body();
124                serde_json::from_reader(bytes.reader()).map_err(new_json_deserialize_error)?
125            }
126            StatusCode::NOT_FOUND => {
127                // We must create directory for the destination
128                let response = self.onedrive_create_dir(ctx, path).await?;
129                match response.status() {
130                    StatusCode::CREATED | StatusCode::OK => {
131                        let bytes = response.into_body();
132                        serde_json::from_reader(bytes.reader())
133                            .map_err(new_json_deserialize_error)?
134                    }
135                    _ => return Err(parse_error(response)),
136                }
137            }
138            _ => return Err(parse_error(response)),
139        };
140
141        Ok(item)
142    }

160//
161// - ID
162// - file path

167    /// Send a stat request about a particular path, including:
168    ///
169    /// - Get stat object only if ETag not matches
170    /// - whether to get the object version
171    ///
172    /// See also [`onedrive_get_stat_plain()`].
173    pub(crate) async fn onedrive_stat(
174        &self,
175        ctx: &OperationContext,
176        path: &str,
177        args: OpStat,
178    ) -> Result<Metadata> {
179        let mut url: String = self.onedrive_item_url(path, true);
180        if args.version().is_some() {
181            url += "?$expand=versions(";
182            url += VERSION_SELECT_PARAM;
183            url += ")";
184        }
185
186        let mut request = Request::get(&url);
187        if let Some(etag) = args.if_none_match() {
188            request = request.header(header::IF_NONE_MATCH, etag);
189        }
190
191        let mut request = request
192            .extension(Operation::Stat)
193            .extension(ServiceOperation("GetItem"))
194            .body(Buffer::new())
195            .map_err(new_request_build_error)?;
196
197        self.sign(ctx, &mut request).await?;
198
199        let response = ctx.http_transport().send(request).await?;
200        if !response.status().is_success() {
201            return Err(parse_error(response));
202        }
203
204        let bytes = response.into_body();
205        let decoded_response: OneDriveItem =
206            serde_json::from_reader(bytes.reader()).map_err(new_json_deserialize_error)?;
207
208        let entry_mode: EntryMode = match decoded_response.item_type {
209            ItemType::Folder { .. } => EntryMode::DIR,
210            ItemType::File { .. } => EntryMode::FILE,
211        };
212
213        let mut meta = Metadata::new(entry_mode)
214            .with_etag(decoded_response.e_tag)
215            .with_content_length(decoded_response.size.max(0) as u64);
216
217        if let Some(version) = args.version() {
218            for item_version in decoded_response.versions.as_deref().unwrap_or_default() {
219                if item_version.id == version {
220                    meta.set_version(version);
221                    break; // early exit
222                }
223            }
224
225            if meta.version().is_none() {
226                return Err(Error::new(
227                    ErrorKind::NotFound,
228                    "cannot find this version of the item",
229                ));
230            }
231        }
232
233        let last_modified = decoded_response.last_modified_date_time;
234        let date_utc_last_modified = last_modified.parse::<Timestamp>()?;
235        meta.set_last_modified(date_utc_last_modified);
236
237        Ok(meta)
238    }

288        Request::get(&url)

290            .extension(ServiceOperation("ListChildren"))
291            .body(Buffer::new())
292            .map_err(new_request_build_error)
293    }

324    /// Download a file
325    ///
326    /// OneDrive handles a download in 2 steps:
327    /// 1. Returns a 302 with a presigned URL. If `If-None-Match` succeed, returns 304.
328    /// 2. With the presigned URL, we can send a GET:
329    ///   1. When getting an item succeed with a `Range` header, we get a 206 Partial Content response.
330    ///   2. When succeed, we get a 200 response.
331    ///
332    /// Read more at https://learn.microsoft.com/en-us/graph/api/driveitem-get-content
333    pub(crate) async fn onedrive_get_content(
334        &self,
335        ctx: &OperationContext,
336        path: &str,
337        range: BytesRange,
338        args: &OpRead,
339    ) -> Result<Response<HttpBody>> {
340        // We can't "select" the OneDrive API response fields when reading because "select" shadows not found error
341        let url: String = format!("{}:/content", self.onedrive_item_url(path, true));
342
343        let mut request = Request::get(&url).header(header::RANGE, range.to_header());
344        if let Some(etag) = args.if_none_match() {
345            request = request.header(header::IF_NONE_MATCH, etag);
346        }
347
348        let mut request = request
349            .extension(Operation::Read)
350            .extension(ServiceOperation("DownloadContent"))
351            .body(Buffer::new())
352            .map_err(new_request_build_error)?;
353
354        self.sign(ctx, &mut request).await?;
355
356        ctx.http_transport().fetch(request).await
357    }

359    /// Upload a file
360    ///
361    /// When creating a file,
362    ///
363    /// * OneDrive returns 201 if the file is new.
364    /// * OneDrive returns 200 if successfully overwrote the file successfully.
365    ///
366    /// Read more at https://learn.microsoft.com/en-us/onedrive/developer/rest-api/api/driveitem_put_content
367    ///
368    /// This function is different than uploading a file with chunks.
369    /// See also [`create_upload_session()`] and [`OneDriveWriter::write_chunked`].
370    pub async fn onedrive_upload_simple(
371        &self,
372        ctx: &OperationContext,
373        path: &str,
374        args: &OpWrite,
375        body: Buffer,
376    ) -> Result<Response<Buffer>> {
377        let url = format!(
378            "{}:/content?@microsoft.graph.conflictBehavior={}&{}",
379            self.onedrive_item_url(path, true),
380            REPLACE_EXISTING_ITEM_WHEN_CONFLICT,
381            GENERAL_SELECT_PARAM
382        );
383
384        // OneDrive upload API documentation requires "text/plain" as the content type.
385        // In practice, OneDrive ignores the content type,
386        // but decides the type (when stating) based on the extension name.
387        // Also, when the extension name is unknown to OneDrive,
388        // OneDrive sets the content type as "application/octet-stream".
389        // We keep the content type according to the documentation.
390        let mut request = Request::put(&url)
391            .header(header::CONTENT_LENGTH, body.len())
392            .header(header::CONTENT_TYPE, "text/plain");
393
394        // when creating a new file, `IF-Match` has no effect.
395        // when updating a file with the `If-Match`, and if the ETag mismatched,
396        // OneDrive will return 412 Precondition Failed
397        if let Some(if_match) = args.if_match() {
398            request = request.header(header::IF_MATCH, if_match);
399        }
400
401        let mut request = request
402            .extension(Operation::Write)
403            .extension(ServiceOperation("UploadContent"))
404            .body(body)
405            .map_err(new_request_build_error)?;
406
407        self.sign(ctx, &mut request).await?;
408
409        ctx.http_transport().send(request).await
410    }

412    #[allow(clippy::too_many_arguments)]
413    pub(crate) async fn onedrive_chunked_upload(
414        &self,
415        ctx: &OperationContext,
416        url: &str,
417        args: &OpWrite,
418        offset: usize,
419        chunk_end: usize,
420        total_len: usize,
421        body: Buffer,
422    ) -> Result<Response<Buffer>> {
423        let mut request = Request::put(url);
424
425        let range = format!("bytes {offset}-{chunk_end}/{total_len}");
426        request = request.header(header::CONTENT_RANGE, range);
427
428        let size = chunk_end - offset + 1;
429        request = request.header(header::CONTENT_LENGTH, size);
430
431        if let Some(mime) = args.content_type() {
432            request = request.header(header::CONTENT_TYPE, mime)
433        }
434
435        let request = request
436            .extension(Operation::Write)
437            .extension(ServiceOperation("UploadFragment"))
438            .body(body)
439            .map_err(new_request_build_error)?;
440        // OneDrive documentation requires not sending the `Authorization` header
441
442        ctx.http_transport().send(request).await
443    }

540    /// Initialize a copy
541    ///
542    /// * `source` - the path to the source folder without the root
543    /// * `destination` - the path to the destination folder without the root
544    ///
545    /// See also: [`wait_until_complete()`]
546    pub(crate) async fn initialize_copy(
547        &self,
548        ctx: &OperationContext,
549        source: &str,
550        destination: &str,
551    ) -> Result<String> {
552        // we must validate if source exist
553        let response = self.onedrive_get_stat_plain(ctx, source).await?;
554        if !response.status().is_success() {
555            return Err(parse_error(response));
556        }
557
558        // We need to stat the destination parent folder to get a parent reference
559        let destination_parent = get_parent(destination).to_string();
560        let basename = get_basename(destination);
561
562        let item = self.ensure_directory(ctx, &destination_parent).await?;
563        let body = OneDrivePatchRequestBody {
564            parent_reference: ParentReference {
565                path: "".to_string(), // irrelevant for copy
566                drive_id: item.parent_reference.drive_id,
567                id: item.id,
568            },
569            name: basename.to_string(),
570        };
571
572        // ensure the destination file or folder doesn't exist
573        let response = self.onedrive_get_stat_plain(ctx, destination).await?;
574        match response.status() {
575            // We must remove the file or folder because
576            // OneDrive doesn't support `conflictBehavior` for the consumer OneDrive.
577            // `conflictBehavior` seems to work for the consumer OneDrive sometimes could be a coincidence.
578            // Read more at https://learn.microsoft.com/en-us/graph/api/driveitem-copy
579            StatusCode::OK => {
580                let response = self.onedrive_delete(ctx, destination).await?;
581                match response.status() {
582                    StatusCode::NO_CONTENT | StatusCode::NOT_FOUND => {} // expected, intentionally empty
583                    _ => return Err(parse_error(response)),
584                }
585            }
586            StatusCode::NOT_FOUND => {} // expected, intentionally empty
587            _ => return Err(parse_error(response)),
588        }
589
590        let url: String = format!("{}:/copy", self.onedrive_item_url(source, true));
591
592        let body_bytes = serde_json::to_vec(&body).map_err(new_json_serialize_error)?;
593        let buffer = Buffer::from(Bytes::from(body_bytes));
594        let mut request = Request::post(&url)
595            .header(header::CONTENT_TYPE, "application/json")
596            .extension(Operation::Copy)
597            .extension(ServiceOperation("CopyItem"))
598            .body(buffer)
599            .map_err(new_request_build_error)?;
600
601        self.sign(ctx, &mut request).await?;
602
603        let response = ctx.http_transport().send(request).await?;
604        match response.status() {
605            StatusCode::ACCEPTED => parse_location(response.headers())?
606                .ok_or_else(|| {
607                    Error::new(
608                        ErrorKind::Unexpected,
609                        "OneDrive didn't return a location URL",
610                    )
611                })
612                .map(String::from),
613            _ => Err(parse_error(response)),
614        }
615    }

617    pub(crate) async fn wait_until_complete(
618        &self,
619        ctx: &OperationContext,
620        monitor_url: String,
621    ) -> Result<()> {
622        for _attempt in 0..MAX_MONITOR_ATTEMPT {
623            let mut request = Request::get(monitor_url.to_string())
624                .header(header::CONTENT_TYPE, "application/json")
625                .extension(Operation::Copy)
626                .extension(ServiceOperation("MonitorCopy"))
627                .body(Buffer::new())
628                .map_err(new_request_build_error)?;
629
630            self.sign(ctx, &mut request).await?;
631
632            let response = ctx.http_transport().send(request).await?;
633            let status: OneDriveMonitorStatus =
634                serde_json::from_reader(response.into_body().reader())
635                    .map_err(new_json_deserialize_error)?;
636            if status.status == "completed" {
637                return Ok(());
638            }
639
640            tokio::time::sleep(Duration::from_secs(MONITOR_WAIT_SECOND)).await;
641        }
642
643        Err(Error::new(
644            ErrorKind::Unexpected,
645            "Exceed monitoring timeout",
646        ))
647    }

649    pub(crate) async fn onedrive_move(
650        &self,
651        ctx: &OperationContext,
652        source: &str,
653        destination: &str,
654    ) -> Result<()> {
655        // We must validate if the source folder exists.
656        let response = self.onedrive_get_stat_plain(ctx, source).await?;
657        if !response.status().is_success() {
658            return Err(Error::new(ErrorKind::NotFound, "source not found"));
659        }
660
661        // We want a parent reference about the destination's parent, or the destination folder itself.
662        let destination_parent = get_parent(destination).to_string();
663        let basename = get_basename(destination);
664
665        let item = self.ensure_directory(ctx, &destination_parent).await?;
666        let body = OneDrivePatchRequestBody {
667            parent_reference: ParentReference {
668                path: "".to_string(), // irrelevant for update
669                // reusing `ParentReference` for convenience. The API requires this value to be correct.
670                drive_id: item.parent_reference.drive_id,
671                id: item.id,
672            },
673            name: basename.to_string(),
674        };
675        let body_bytes = serde_json::to_vec(&body).map_err(new_json_serialize_error)?;
676        let buffer = Buffer::from(Bytes::from(body_bytes));
677        let url: String = format!(
678            "{}?@microsoft.graph.conflictBehavior={}&$select=id",
679            self.onedrive_item_url(source, true),
680            REPLACE_EXISTING_ITEM_WHEN_CONFLICT
681        );
682        let mut request = Request::patch(&url)
683            .header(header::CONTENT_TYPE, "application/json")
684            .extension(Operation::Rename)
685            .extension(ServiceOperation("MoveItem"))
686            .body(buffer)
687            .map_err(new_request_build_error)?;
688
689        self.sign(ctx, &mut request).await?;
690
691        let response = ctx.http_transport().send(request).await?;
692        match response.status() {
693            // can get etag, metadata, etc...
694            StatusCode::OK => Ok(()),
695            _ => Err(parse_error(response)),
696        }
697    }
698}

700// keeps track of OAuth 2.0 tokens and refreshes the access token.
701pub struct OneDriveSigner {
702    pub client_id: String,
703    pub client_secret: String,
704    pub refresh_token: String,
705
706    pub access_token: String,
707    pub expires_in: Timestamp,
708}
709
710// OneDrive is part of Graph API hence shares the same authentication and authorization processes.
711// `common` applies to account types:
712//
713// - consumers
714// - work and school account
715//
716// set to `common` for simplicity
717const ONEDRIVE_REFRESH_TOKEN: &str = "https://login.microsoftonline.com/common/oauth2/v2.0/token";
718
719impl OneDriveSigner {
720    pub fn new() -> Self {
721        OneDriveSigner {
722            client_id: "".to_string(),
723            client_secret: "".to_string(),
724            refresh_token: "".to_string(),
725            access_token: "".to_string(),
726            expires_in: Timestamp::MIN,
727        }
728    }
729
730    async fn refresh_tokens(&mut self, ctx: &OperationContext) -> Result<()> {
731        // OneDrive users must provide at least this required permission scope
732        let encoded_payload = format!(
733            "client_id={}&client_secret={}&scope=offline_access%20Files.ReadWrite&refresh_token={}&grant_type=refresh_token",
734            percent_encode_path(self.client_id.as_str()),
735            percent_encode_path(self.client_secret.as_str()),
736            percent_encode_path(self.refresh_token.as_str())
737        );
738        let request = Request::post(ONEDRIVE_REFRESH_TOKEN)
739            .header(header::CONTENT_TYPE, "application/x-www-form-urlencoded")
740            .body(Buffer::from(encoded_payload))
741            .map_err(new_request_build_error)?;
742
743        let response = ctx.http_transport().send(request).await?;
744        match response.status() {
745            StatusCode::OK => {
746                let resp_body = response.into_body();
747                let data: GraphOAuthRefreshTokenResponseBody =
748                    serde_json::from_reader(resp_body.reader())
749                        .map_err(new_json_deserialize_error)?;
750                self.access_token = data.access_token;
751                self.refresh_token = data.refresh_token;
752                self.expires_in = Timestamp::now() + Duration::from_secs(data.expires_in)
753                    - Duration::from_secs(120); // assumes 2 mins graceful transmission for implementation simplicity
754                Ok(())
755            }
756            _ => Err(parse_error(response)),
757        }
758    }
759
760    /// Sign a request.
761    pub async fn sign<T>(
762        &mut self,
763        ctx: &OperationContext,
764        request: &mut Request<T>,
765    ) -> Result<()> {
766        if !self.access_token.is_empty() && self.expires_in > Timestamp::now() {
767            let value = format!("Bearer {}", self.access_token)
768                .parse()
769                .expect("access_token must be valid header value");
770
771            request.headers_mut().insert(header::AUTHORIZATION, value);
772            return Ok(());
773        }
774
775        self.refresh_tokens(ctx).await?;
776
777        let auth_header_content = format!("Bearer {}", self.access_token)
778            .parse()
779            .expect("Fetched access_token is invalid as a header value");
780
781        request
782            .headers_mut()
783            .insert(header::AUTHORIZATION, auth_header_content);
784
785        Ok(())
786    }
787}

920mod error {
921    use http::Response;
922    use http::StatusCode;
923
924    use opendal_core::raw::*;
925    use opendal_core::*;
926
927    /// Parse error response into Error.
928    pub(crate) fn parse_error(response: Response<Buffer>) -> Error {
929        let (parts, body) = response.into_parts();
930        let bs = body.to_bytes();
931
932        let (kind, retryable) = match parts.status {
933            StatusCode::NOT_FOUND => (ErrorKind::NotFound, false),
934            // The OneDrive service replaces resources.
935            // However, the Onedrive doesn't have Strong Read-After-Write properties,
936            // the concurrent requests to create directories might result in errors.
937            //
938            // Running behavior tests can yield HTTP 409 Conflict because of the consistency guarantee.
939            //
940            // Read more about `REPLACE_EXISTING_ITEM_WHEN_CONFLICT` in `graph_model.rs`.
941            StatusCode::CONFLICT => (ErrorKind::AlreadyExists, true),
942            StatusCode::FORBIDDEN => (ErrorKind::PermissionDenied, false),
943            StatusCode::INTERNAL_SERVER_ERROR
944            | StatusCode::BAD_GATEWAY
945            | StatusCode::SERVICE_UNAVAILABLE
946            | StatusCode::GATEWAY_TIMEOUT => (ErrorKind::Unexpected, true),
947            StatusCode::NOT_MODIFIED | StatusCode::PRECONDITION_FAILED => {
948                (ErrorKind::ConditionNotMatch, false)
949            }
950            _ => (ErrorKind::Unexpected, false),
951        };
952
953        let message = String::from_utf8_lossy(&bs);
954
955        let mut err = Error::new(kind, message);
956
957        err = with_error_response_context(err, parts);
958
959        if retryable {
960            err = err.set_temporary();
961        }
962
963        err
964    }
965}
