18use std::collections::{HashSet, VecDeque};
19use std::sync::Arc;
20
21use opendal_core::raw::*;
22use opendal_core::*;
23
24use super::core::{RedisCore, format_redis_error};
25
26/// Scan keys for a standalone Redis connection.
27pub struct RedisLister {
28    core: Arc<RedisCore>,
29    root: String,
30    prefix: String,
31    pattern: String,
32    count: Option<usize>,
33    recursive: bool,
34    // Redis-compatible servers can return opaque, non-numeric cursors.
35    cursor: Vec<u8>,
36    done: bool,
37    entries: VecDeque<oio::Entry>,
38    seen: HashSet<String>,
39}
40
41impl RedisLister {
42    pub fn new(core: Arc<RedisCore>, root: &str, path: &str, args: OpList) -> Self {
43        let prefix = build_absolute_path(root, path);
44        // Some Redis-compatible servers only optimize a literal prefix followed by '*'.
45        // Keep the server filter broad and check the full literal prefix locally.
46        let pattern = match prefix.find(['*', '?', '[', ']', '\\', '\0']) {
47            Some(index) => format!("{}*", &prefix[..index]),
48            None => format!("{prefix}*"),
49        };
50        Self {
51            core,
52            root: build_absolute_path(root, "/"),
53            prefix,
54            pattern,
55            count: args.limit().map(|limit| limit.clamp(1, i32::MAX as usize)),
56            recursive: args.recursive(),
57            cursor: b"0".to_vec(),
58            done: false,
59            entries: VecDeque::new(),
60            seen: HashSet::new(),
61        }
62    }
63
64    async fn next_page(&mut self) -> Result<()> {
65        let mut conn = self.core.conn().await?;
66        let mut scan = redis::cmd("SCAN");
67        scan.arg(&self.cursor[..]).arg("MATCH").arg(&self.pattern);
68        if let Some(count) = self.count {
69            scan.arg("COUNT").arg(count);
70        }
71        let (next_cursor, keys): (Vec<u8>, Vec<Vec<u8>>) = scan
72            .query_async(&mut *conn)
73            .await
74            .map_err(format_redis_error)?;
75
76        let mut lengths = redis::pipe();
77        let mut paths = Vec::new();
78        let relative_prefix_len = self.prefix.len() - self.root.len();
79        for key in &keys {
80            let Ok(key) = std::str::from_utf8(key) else {
81                continue;
82            };
83            let Some(suffix) = key.strip_prefix(&self.prefix) else {
84                continue;
85            };
86            let path = &key[self.root.len()..];
87            // Ignore external keys that OpenDAL cannot address by their exact path.
88            if !path.is_empty() && (path == "/" || normalize_path(path) != path) {
89                continue;
90            }
91            let path = if path.is_empty() { "/" } else { path };
92            let path = if !self.recursive
93                && let Some(slash) = suffix.find('/')
94            {
95                &path[..relative_prefix_len + slash + 1]
96            } else {
97                path
98            };
99            if self.seen.contains(path) {
100                continue;
101            }
102            // Compact metadata uses a 16-bit bitmap and a 16-bit offset for the path.
103            if path.len() > u16::MAX as usize - 2 * size_of::<u16>() {
104                return Err(Error::new(
105                    ErrorKind::Unexpected,
106                    "Redis key is too long to list",
107                ));
108            }
109            lengths.cmd("STRLEN").arg(key).cmd("TYPE").arg(key);
110            paths.push(path);
111        }
112
113        if !paths.is_empty() {
114            // STRLEN returns zero for a missing key and errors for other data types.
115            // TYPE distinguishes empty strings from keys that cannot be read as files.
116            let metadata: Vec<(redis::RedisResult<u64>, String)> = lengths
117                .ignore_errors()
118                .query_async(&mut *conn)
119                .await
120                .map_err(format_redis_error)?;
121            let mut entries = Vec::new();
122            for (path, (length, key_type)) in paths.into_iter().zip(metadata) {
123                if key_type != "string" {
124                    continue;
125                }
126                let length = length.map_err(format_redis_error)?;
127                let metadata = if path.ends_with('/') {
128                    MetadataBuilder::dir().build()
129                } else {
130                    MetadataBuilder::file(length).build()
131                };
132                entries.push(oio::Entry::new(path, metadata));
133            }
134            for entry in entries {
135                if self.seen.insert(entry.path().to_owned()) {
136                    self.entries.push_back(entry);
137                }
138            }
139        }
140
141        self.done = next_cursor == b"0";
142        self.cursor = next_cursor;
143        Ok(())
144    }
145}
146
147impl oio::List for RedisLister {
148    async fn next(&mut self) -> Result<Option<oio::Entry>> {
149        while self.entries.is_empty() && !self.done {
150            self.next_page().await?;
151        }
152        Ok(self.entries.pop_front())
153    }
154}
