112/// Re-export of the [`reqsign_aws_v4`] crate that this service signs requests
113/// with.
114///
115/// [`S3::credential_provider_chain`] takes a chain of providers for
116/// [`reqsign_aws_v4::Credential`]. Implement providers against this re-export and
117/// [`opendal_core::reqsign_core`] rather than depending on the `reqsign` crates
118/// directly, so that the types always match the ones this service is built with.
119pub use reqsign_aws_v4;
120
121/// URI scheme used for service registration and scheme-driven construction.
122pub const S3_SCHEME: &str = "s3";
123/// URI scheme for the Cloudflare R2 preset.
124pub const R2_SCHEME: &str = "r2";
125/// URI scheme for the MinIO preset.
126pub const MINIO_SCHEME: &str = "minio";
127
128/// Register the Amazon S3 URI scheme with an operator registry.
129///
130/// Registration enables scheme-driven construction through
131/// [`opendal_core::Operator::from_uri`] and
132/// [`opendal_core::Operator::via_iter`]. Direct construction through
133/// [`opendal_core::Operator::new`] does not require registration.
134pub fn register_s3_service(registry: &opendal_core::OperatorRegistry) {
135    registry.register::<S3>(S3_SCHEME);
136}
137
138/// Register the Cloudflare R2 URI scheme with an operator registry.
139///
140/// Registration enables scheme-driven construction through
141/// [`opendal_core::Operator::from_uri`] and
142/// [`opendal_core::Operator::via_iter`]. Direct construction through
143/// [`opendal_core::Operator::new`] does not require registration.
144pub fn register_r2_service(registry: &opendal_core::OperatorRegistry) {
145    registry.register::<R2>(R2_SCHEME);
146}
147
148/// Register the MinIO URI scheme with an operator registry.
149///
150/// Registration enables scheme-driven construction through
151/// [`opendal_core::Operator::from_uri`] and
152/// [`opendal_core::Operator::via_iter`]. Direct construction through
153/// [`opendal_core::Operator::new`] does not require registration.
154pub fn register_minio_service(registry: &opendal_core::OperatorRegistry) {
155    registry.register::<Minio>(MINIO_SCHEME);
156}

158#[cfg(test)]
159mod tests {
160    use super::*;
161
162    #[test]
163    fn registers_s3_and_provider_schemes() {
164        let registry = opendal_core::OperatorRegistry::get();
165        register_s3_service(registry);
166        register_r2_service(registry);
167        register_minio_service(registry);
168
169        let schemes = registry.schemes();
170        assert!(schemes.contains(S3_SCHEME));
171        assert!(schemes.contains(R2_SCHEME));
172        assert!(schemes.contains(MINIO_SCHEME));
173
174        let r2 = registry
175            .load(("r2://bucket/root", [("account_id", "example-account")]))
176            .unwrap();
177        assert_eq!(r2.info().scheme(), R2_SCHEME);
178
179        let minio = registry
180            .load((
181                "minio://bucket/root",
182                [("endpoint", "http://127.0.0.1:9000")],
183            ))
184            .unwrap();
185        assert_eq!(minio.info().scheme(), MINIO_SCHEME);
186    }
