64    pub async fn get_client(&self) -> Result<&SqlitePool> {
65        self.pool
66            .get_or_try_init(|| async {
67                SqlitePool::connect_with(self.config.clone())
68                    .await
69                    .map_err(parse_sqlite_error)
70            })
71            .await
72    }
73
74    pub async fn get(&self, path: &str) -> Result<Option<Buffer>> {
75        let pool = self.get_client().await?;
76        let table = &self.table;
77        let key_field = &self.key_field;
78        let value_field = &self.value_field;
79
80        // SQLite prefers using standard `"` for identifier quotes instead of
81        // MySQL-compatible backtick ` or [] quotes.
82        // Identifier quotes are case-sensitive and allow special characters.
83        // Read more https://www.sqlite.org/quirks.html#double_quoted_string_literals_are_accepted
84        //
85        // We use:
86        // - pre-quoted, escaped identifiers for trusted table and field configuration
87        // - bind parameters for values to avoid malformed SQL and SQL injection
88        // to ensure correctness. `AssertSqlSafe` records this audit for sqlx.
89        let value: Option<Vec<u8>> = sqlx::query_scalar(sqlx::AssertSqlSafe(format!(
90            "SELECT {value_field} FROM {table} WHERE {key_field} = $1 LIMIT 1",
91        )))
92        .bind(path)
93        .fetch_optional(pool)
94        .await
95        .map_err(parse_sqlite_error)?;
96
97        Ok(value.map(Buffer::from))
98    }

100    pub async fn get_length(&self, path: &str) -> Result<Option<u64>> {
101        let pool = self.get_client().await?;
102        let table = &self.table;
103        let key_field = &self.key_field;
104        let value_field = &self.value_field;
105
106        let value: Option<i64> = sqlx::query_scalar(sqlx::AssertSqlSafe(format!(
107            "SELECT LENGTH(CAST({value_field} AS BLOB)) FROM {table} WHERE {key_field} = $1 LIMIT 1",
108        )))
109        .bind(path)
110        .fetch_optional(pool)
111        .await
112        .map_err(parse_sqlite_error)?;
113
114        value
115            .map(|v| {
116                u64::try_from(v).map_err(|err| {
117                    Error::new(ErrorKind::Unexpected, "sqlite value length is invalid")
118                        .set_source(err)
119                })
120            })
121            .transpose()
122    }

134        .await
135        .map_err(parse_sqlite_error)
136    }

138    pub async fn get_range(
139        &self,
140        path: &str,
141        offset: u64,
142        limit: Option<u64>,
143    ) -> Result<Option<(Buffer, u64)>> {
144        // SQLite BLOB/TEXT values are at most `i32::MAX` bytes and `SUBSTR`
145        // converts its start and length arguments to signed 32-bit integers.
146        // An offset >= `i32::MAX` is therefore always at or past EOF.
147        let Some(start) = offset.checked_add(1).and_then(|v| i32::try_from(v).ok()) else {
148            return Ok(self
149                .get_length(path)
150                .await?
151                .map(|length| (Buffer::new(), length)));
152        };
153
154        let pool = self.get_client().await?;
155        let table = &self.table;
156        let key_field = &self.key_field;
157        let value_field = &self.value_field;
158        // Any requested length above `i32::MAX` exceeds SQLite's maximum value
159        // size, so omit the length argument to read through EOF without 32-bit
160        // truncation in `SUBSTR`.
161        let value = match limit.and_then(|v| i32::try_from(v).ok()) {
162            Some(limit) => {
163                sqlx::query_as(sqlx::AssertSqlSafe(format!(
164                    "SELECT SUBSTR(CAST({value_field} AS BLOB), $1, $2), LENGTH(CAST({value_field} AS BLOB)) FROM {table} WHERE {key_field} = $3 LIMIT 1",
165                )))
166                .bind(start)
167                .bind(limit)
168                .bind(path)
169                .fetch_optional(pool)
170                .await
171            }
172            None => {
173                sqlx::query_as(sqlx::AssertSqlSafe(format!(
174                    "SELECT SUBSTR(CAST({value_field} AS BLOB), $1), LENGTH(CAST({value_field} AS BLOB)) FROM {table} WHERE {key_field} = $2 LIMIT 1",
175                )))
176                .bind(start)
177                .bind(path)
178                .fetch_optional(pool)
179                .await
180            }
181        };
182        let value: Option<(Vec<u8>, i64)> = value.map_err(parse_sqlite_error)?;
183
184        Ok(value.map(|(bs, size)| (Buffer::from(bs), size as u64)))
185    }

187    pub async fn get_suffix(&self, path: &str, size: u64) -> Result<Option<(Buffer, u64)>> {
188        if size == 0 {
189            return Ok(self
190                .get_length(path)
191                .await?
192                .map(|length| (Buffer::new(), length)));
193        }
194        // SQLite BLOB/TEXT values are at most `i32::MAX` bytes, so any suffix
195        // of `i32::MAX` or more bytes returns the full value.
196        if size >= i32::MAX as u64 {
197            return Ok(self.get(path).await?.map(|bs| {
198                let length = bs.len() as u64;
199                (bs, length)
200            }));
201        }
202
203        let pool = self.get_client().await?;
204        let table = &self.table;
205        let key_field = &self.key_field;
206        let value_field = &self.value_field;
207        let size = size as i64;
208        // Compute a 1-based start index in `1..=i32::MAX` using 64-bit SQL
209        // arithmetic so `SUBSTR` never receives a negative or out-of-range
210        // 32-bit index.
211        let value: Option<(Vec<u8>, i64)> = sqlx::query_as(sqlx::AssertSqlSafe(format!(
212            "SELECT SUBSTR(CAST({value_field} AS BLOB), MAX(1, LENGTH(CAST({value_field} AS BLOB)) - $1 + 1)), LENGTH(CAST({value_field} AS BLOB)) FROM {table} WHERE {key_field} = $2 LIMIT 1",
213        )))
214        .bind(size)
215        .bind(path)
216        .fetch_optional(pool)
217        .await
218        .map_err(parse_sqlite_error)?;
219
220        Ok(value.map(|(bs, total_size)| (Buffer::from(bs), total_size as u64)))
221    }

223    pub async fn set(&self, path: &str, value: Buffer) -> Result<()> {
224        let pool = self.get_client().await?;
225        let table = &self.table;
226        let key_field = &self.key_field;
227        let value_field = &self.value_field;
228
229        // `INSERT OR REPLACE` deletes the conflicting row, which resets
230        // unrelated columns and can delete a different row when another unique
231        // constraint conflicts. Update only the value column on a key conflict.
232        sqlx::query(sqlx::AssertSqlSafe(format!(
233            "INSERT INTO {table} ({key_field}, {value_field}) VALUES ($1, $2)
234            ON CONFLICT ({key_field}) DO UPDATE SET {value_field} = excluded.{value_field}",
235        )))
236        .bind(path)
237        .bind(value.to_vec())
238        .execute(pool)
239        .await
240        .map_err(parse_sqlite_error)?;
241
242        Ok(())
243    }

245    pub async fn rename(&self, from: &str, to: &str, if_not_exists: bool) -> Result<()> {
246        let pool = self.get_client().await?;
247        // Serialize competing writers before checking the destination.
248        let mut tx = pool
249            .begin_with("BEGIN IMMEDIATE")
250            .await
251            .map_err(parse_sqlite_error)?;
252        let table = &self.table;
253        let key = &self.key_field;
254
255        let (source_exists, destination_exists): (bool, bool) =
256            sqlx::query_as(sqlx::AssertSqlSafe(format!(
257                "SELECT EXISTS (SELECT 1 FROM {table} WHERE {key} = $1),
258                        EXISTS (SELECT 1 FROM {table} WHERE {key} = $2)",
259            )))
260            .bind(from)
261            .bind(to)
262            .fetch_one(&mut *tx)
263            .await
264            .map_err(parse_sqlite_error)?;
265
266        if !source_exists {
267            return Err(Error::new(ErrorKind::NotFound, "key not found in sqlite"));
268        }
269        if destination_exists {
270            if if_not_exists {
271                return Err(Error::new(
272                    ErrorKind::ConditionNotMatch,
273                    "destination key already exists in sqlite",
274                ));
275            }
276
277            // The key collation can make two different paths match the source row.
278            sqlx::query(sqlx::AssertSqlSafe(format!(
279                "DELETE FROM {table} WHERE {key} = $2 AND {key} <> $1",
280            )))
281            .bind(from)
282            .bind(to)
283            .execute(&mut *tx)
284            .await
285            .map_err(parse_sqlite_error)?;
286        }
287
288        // ABORT preserves rows that conflict through other constraints, even if
289        // the schema configures a different default conflict policy.
290        let result = sqlx::query(sqlx::AssertSqlSafe(format!(
291            "UPDATE OR ABORT {table} SET {key} = $2 WHERE {key} = $1",
292        )))
293        .bind(from)
294        .bind(to)
295        .execute(&mut *tx)
296        .await
297        .map_err(|err| {
298            if err
299                .as_database_error()
300                .is_some_and(|err| err.is_unique_violation())
301            {
302                Error::new(
303                    ErrorKind::Conflict,
304                    "sqlite rename violates a unique constraint",
305                )
306                .set_source(err)
307            } else {
308                parse_sqlite_error(err)
309            }
310        })?;
311        if result.rows_affected() == 0 {
312            return Err(Error::new(
313                ErrorKind::Unexpected,
314                "sqlite rename did not update the source key",
315            ));
316        }
317
318        tx.commit().await.map_err(parse_sqlite_error)
319    }

321    pub async fn delete(&self, path: &str) -> Result<()> {
322        let pool = self.get_client().await?;
323        let table = &self.table;
324        let key_field = &self.key_field;
325
326        if path.is_empty() || path.ends_with('/') {
327            // Serialize competing writers so a child cannot be inserted between
328            // the emptiness check and deleting the directory marker.
329            let mut tx = pool
330                .begin_with("BEGIN IMMEDIATE")
331                .await
332                .map_err(parse_sqlite_error)?;
333
334            let has_children: bool = sqlx::query_scalar(sqlx::AssertSqlSafe(format!(
335                "SELECT EXISTS (SELECT 1 FROM {table} WHERE instr({key_field}, $1) = 1 AND {key_field} <> $1)",
336            )))
337            .bind(path)
338            .fetch_one(&mut *tx)
339            .await
340            .map_err(parse_sqlite_error)?;
341
342            if has_children {
343                tx.rollback().await.map_err(parse_sqlite_error)?;
344                return Err(Error::new(ErrorKind::Unexpected, "directory not empty"));
345            }
346
347            sqlx::query(sqlx::AssertSqlSafe(format!(
348                "DELETE FROM {table} WHERE {key_field} = $1",
349            )))
350            .bind(path)
351            .execute(&mut *tx)
352            .await
353            .map_err(parse_sqlite_error)?;
354
355            return tx.commit().await.map_err(parse_sqlite_error);
356        }
357
358        sqlx::query(sqlx::AssertSqlSafe(format!(
359            "DELETE FROM {table} WHERE {key_field} = $1",
360        )))
361        .bind(path)
362        .execute(pool)
363        .await
364        .map_err(parse_sqlite_error)?;
365
366        Ok(())
367    }

370// SQLite escapes double quotes in identifiers by doubling them. Queries
371// interpolate only these quoted identifiers and bind paths and values.
372fn quote_identifier(identifier: &str) -> String {
373    format!("\"{}\"", identifier.replace('"', "\"\""))
