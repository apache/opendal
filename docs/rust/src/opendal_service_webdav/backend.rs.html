37/// [WebDAV](https://datatracker.ietf.org/doc/html/rfc4918) backend support.

40pub struct WebdavBuilder {

42}

44impl Debug for WebdavBuilder {

46        f.debug_struct("WebdavBuilder")
47            .field("config", &self.config)
48            .finish_non_exhaustive()

52impl WebdavBuilder {
53    /// Set endpoint for http backend.
54    ///
55    /// For example: `https://example.com`
56    pub fn endpoint(mut self, endpoint: &str) -> Self {
57        self.config.endpoint = if endpoint.is_empty() {
58            None
59        } else {
60            Some(endpoint.to_string())
61        };
62
63        self
64    }
65
66    /// set the username for Webdav
67    ///
68    /// default: no username
69    pub fn username(mut self, username: &str) -> Self {
70        if !username.is_empty() {
71            self.config.username = Some(username.to_owned());
72        }
73        self
74    }
75
76    /// set the password for Webdav
77    ///
78    /// default: no password
79    pub fn password(mut self, password: &str) -> Self {
80        if !password.is_empty() {
81            self.config.password = Some(password.to_owned());
82        }
83        self
84    }
85
86    /// set the bearer token for Webdav
87    ///
88    /// default: no access token
89    pub fn token(mut self, token: &str) -> Self {
90        if !token.is_empty() {
91            self.config.token = Some(token.to_string());
92        }
93        self
94    }
95
96    /// Set root path of http backend.
97    pub fn root(mut self, root: &str) -> Self {
98        self.config.root = if root.is_empty() {
99            None
100        } else {
101            Some(root.to_string())
102        };
103
104        self
105    }
106
107    /// Disable automatic parent directory creation before writing files.
108    ///
109    /// By default, OpenDAL checks for parent directories using `PROPFIND` and creates
110    /// missing directories using `MKCOL` before writing files. Enable this option to
111    /// skip those checks and `MKCOL` calls for servers such as `bazel-remote`.
112    ///
113    /// Explicit `create_dir` calls are unaffected.
114    ///
115    /// Default: `false`
116    pub fn disable_create_dir(mut self, disable: bool) -> Self {
117        self.config.disable_create_dir = disable;
118        self
119    }
120
121    /// Disable native recursive listing with WebDAV's `Depth: infinity`.
122    ///
123    /// Set this to `true` for servers that reject or ignore infinity depth.
124    /// With capability simulation enabled, OpenDAL walks the directory tree
125    /// using `Depth: 1` requests instead.
126    ///
127    /// The default is `false`.
128    pub fn disable_depth_infinity(mut self, disable: bool) -> Self {
129        self.config.disable_depth_infinity = disable;
130        self
131    }

133    /// Deprecated: WebDAV user metadata capability is enabled by default.
134    #[deprecated(
135        since = "0.57.0",
136        note = "WebDAV user metadata capability is enabled by default. Use CapabilityOverrideLayer to override write_with_user_metadata for endpoints without PROPPATCH support."
137    )]
138    pub fn enable_user_metadata(self, _enable: bool) -> Self {
139        self
140    }

144    /// This prefix is used in PROPPATCH/PROPFIND XML requests.

147    /// Default: "opendal"

178impl Builder for WebdavBuilder {
179    type Config = WebdavConfig;
180
181    fn build(self) -> Result<impl Service> {
182        debug!("backend build started: {:?}", self);
183
184        let endpoint = match &self.config.endpoint {
185            Some(v) => v,
186            None => {
187                return Err(Error::new(ErrorKind::ConfigInvalid, "endpoint is empty")
188                    .with_context("service", WEBDAV_SCHEME));
189            }
190        };
191        // Some services might return the path with suffix `/remote.php/webdav/`, we need to trim them.
192        let server_path = http::Uri::from_str(endpoint)
193            .map_err(|err| {
194                Error::new(ErrorKind::ConfigInvalid, "endpoint is invalid")
195                    .with_context("service", WEBDAV_SCHEME)
196                    .set_source(err)
197            })?
198            .path()
199            .trim_end_matches('/')
200            .to_string();
201
202        let root = normalize_root(&self.config.root.clone().unwrap_or_default());
203        debug!("backend use root {root}");
204
205        let mut authorization = None;
206        if let Some(username) = &self.config.username {
207            authorization = Some(format_authorization_by_basic(
208                username,
209                self.config.password.as_deref().unwrap_or_default(),
210            )?);
211        }
212        if let Some(token) = &self.config.token {
213            authorization = Some(format_authorization_by_bearer(token)?)
214        }
215
216        let core = Arc::new(WebdavCore {
217            info: ServiceInfo::new(WEBDAV_SCHEME, &root, ""),
218            capability: Capability {
219                stat: true,
220
221                read: true,
222                read_with_suffix: true,
223                read_with_if_match: true,
224                read_with_if_none_match: true,
225                read_with_if_modified_since: true,
226                read_with_if_unmodified_since: true,
227
228                write: true,
229                write_can_empty: true,
230                write_with_if_match: true,
231                write_with_if_none_match: true,
232                write_with_if_not_exists: true,
233                write_with_user_metadata: true,
234
235                create_dir: true,
236                delete: true,
237                delete_with_recursive: true,
238                delete_with_if_match: true,
239
240                copy: true,
241
242                rename: true,
243                rename_with_if_not_exists: true,
244
245                list: true,
246                list_with_recursive: !self.config.disable_depth_infinity,
247
248                shared: true,
249
250                ..Default::default()
251            },
252            endpoint: endpoint.trim_end_matches('/').to_string(),
253            server_path,
254            authorization,
255            root,
256            user_metadata_prefix: self
257                .config
258                .user_metadata_prefix
259                .unwrap_or_else(|| DEFAULT_USER_METADATA_PREFIX.to_string()),
260            user_metadata_uri: self
261                .config
262                .user_metadata_uri
263                .unwrap_or_else(|| DEFAULT_USER_METADATA_URI.to_string()),
264            disable_create_dir: self.config.disable_create_dir,
265        });
266        Ok(WebdavBackend { core })
267    }
268}

270#[derive(Clone, Debug)]
271pub struct WebdavBackend {
272    pub(crate) core: Arc<WebdavCore>,
273}
274
275impl Service for WebdavBackend {
276    type Reader = oio::StreamReader<WebdavReader>;
277    type Writer = oio::OneShotWriter<WebdavWriter>;
278    type Lister = oio::PageLister<WebdavLister>;
279    type Deleter = oio::OneShotDeleter<WebdavDeleter>;
280    type Copier = oio::OneShotCopier;
281    type Composer = ();
282
283    fn info(&self) -> ServiceInfo {
284        self.core.info.clone()
285    }
286
287    fn capability(&self) -> Capability {
288        self.core.capability
289    }
290
291    async fn create_dir(
292        &self,
293        ctx: &OperationContext,
294        path: &str,
295        _: OpCreateDir,
296    ) -> Result<RpCreateDir> {
297        self.core.webdav_mkcol(ctx, path).await?;
298        Ok(RpCreateDir::default())
299    }
300
301    async fn stat(&self, ctx: &OperationContext, path: &str, _: OpStat) -> Result<RpStat> {
302        let metadata = self.core.webdav_stat(ctx, path).await?;
303        Ok(RpStat::new(metadata))
304    }
305    fn read(&self, ctx: &OperationContext, path: &str, args: OpRead) -> Result<Self::Reader> {
306        let output: oio::StreamReader<WebdavReader> = {
307            Ok(oio::StreamReader::new(WebdavReader::new(
308                self.clone(),
309                ctx.clone(),
310                path,
311                args,
312            )))
313        }?;
314
315        Ok(output)
316    }
317
318    fn write(&self, ctx: &OperationContext, path: &str, args: OpWrite) -> Result<Self::Writer> {
319        if args.if_version_match().is_some() || args.if_version_not_match().is_some() {
320            return Err(Error::new(
321                ErrorKind::Unsupported,
322                "WebDAV does not support write version conditions",
323            ));
324        }
325
326        Ok(oio::OneShotWriter::new(WebdavWriter::new(
327            self.core.clone(),
328            ctx.clone(),
329            args,
330            path.to_string(),
331        )))
332    }
333
334    fn delete(&self, ctx: &OperationContext) -> Result<Self::Deleter> {
335        let output: oio::OneShotDeleter<WebdavDeleter> = {
336            Ok(oio::OneShotDeleter::new(WebdavDeleter::new(
337                self.core.clone(),
338                ctx.clone(),
339            )))
340        }?;
341
342        Ok(output)
343    }
344
345    fn list(&self, ctx: &OperationContext, path: &str, args: OpList) -> Result<Self::Lister> {
346        if args.recursive() && !self.core.capability.list_with_recursive {
347            return Err(Error::new(
348                ErrorKind::Unsupported,
349                "WebDAV recursive listing with Depth: infinity is disabled",
350            ));
351        }
352
353        Ok(oio::PageLister::new(WebdavLister::new(
354            self.core.clone(),
355            ctx.clone(),
356            path,
357            args,
358        )))
359    }
360
361    fn copy(
362        &self,
363        ctx: &OperationContext,
364        from: &str,
365        to: &str,
366        args: OpCopy,
367    ) -> Result<Self::Copier> {
368        let backend = self.clone();
369        let core = self.core.clone();
370        let ctx = ctx.clone();
371        let from = from.to_string();
372        let to = to.to_string();
373        let source_content_length_hint = args.source_content_length_hint();
374
375        Ok(oio::OneShotCopier::new_with(move || {
376            let backend = backend.clone();
377            let core = core.clone();
378            let ctx = ctx.clone();
379            let from = from.clone();
380            let to = to.clone();
381
382            async move {
383                let source_size = match source_content_length_hint {
384                    Some(size) => size,
385                    None => backend
386                        .stat(&ctx, &from, OpStat::default())
387                        .await?
388                        .into_metadata()
389                        .content_length(),
390                };
391
392                let resp = core.webdav_copy(&ctx, &from, &to).await?;
393                let status = resp.status();
394
395                match status {
396                    StatusCode::CREATED | StatusCode::NO_CONTENT => {
397                        Ok(MetadataBuilder::file(source_size).build())
398                    }
399                    _ => Err(parse_error(
400                        ErrorContext::new(ServiceOperation("Copy")),
401                        resp,
402                    )),
403                }
404            }
405        }))
406    }
407
408    async fn rename(
409        &self,
410        ctx: &OperationContext,
411        from: &str,
412        to: &str,
413        args: OpRename,
414    ) -> Result<RpRename> {
415        self.core.webdav_stat(ctx, from).await?;
416        self.core.webdav_mkcol(ctx, get_parent(to)).await?;
417        let resp = self
418            .core
419            .webdav_move(ctx, from, to, args.if_not_exists())
420            .await?;
421
422        let status = resp.status();
423        match status {
424            StatusCode::CREATED | StatusCode::NO_CONTENT | StatusCode::OK => {
425                Ok(RpRename::default())
426            }
427            _ => Err(parse_error(
428                ErrorContext::new(ServiceOperation("Move")),
429                resp,
430            )),
431        }
432    }
433
434    async fn presign(
435        &self,
436        _ctx: &OperationContext,
437        _path: &str,
438        _args: OpPresign,
439    ) -> Result<RpPresign> {
440        Err(Error::new(
441            ErrorKind::Unsupported,
442            "operation is not supported",
443        ))
444    }
445}

458    #[derive(Clone, Debug, PartialEq, Eq)]
459    struct RecordedRequest {
460        method: String,
461        path: String,
462        headers: http::HeaderMap,
463    }
464
465    #[derive(Clone, Default)]
466    struct RecordingHttpTransport {
467        requests: Arc<Mutex<Vec<RecordedRequest>>>,
468    }
469
470    impl RecordingHttpTransport {
471        fn requests(&self) -> Vec<RecordedRequest> {
472            self.requests.lock().unwrap().clone()
473        }
474
475        fn methods_and_paths(&self) -> Vec<(String, String)> {
476            self.requests()
477                .into_iter()
478                .map(|req| (req.method, req.path))
479                .collect()
480        }
481    }
482
483    impl HttpTransport for RecordingHttpTransport {
484        async fn fetch(&self, req: Request<Buffer>) -> Result<Response<HttpBody>> {
485            let method = req.method().as_str().to_string();
486            let path = req.uri().path().to_string();
487            let headers = req.headers().clone();
488            self.requests.lock().unwrap().push(RecordedRequest {
489                method: method.clone(),
490                path: path.clone(),
491                headers,
492            });
493
494            let (status, body) = match (method.as_str(), path.as_str()) {
495                ("PROPFIND", _) => (StatusCode::NOT_FOUND, ""),
496                ("MKCOL", _) | ("PUT", _) => (StatusCode::CREATED, ""),
497                ("DELETE", _) => (StatusCode::NO_CONTENT, ""),
498                _ => (StatusCode::METHOD_NOT_ALLOWED, ""),
499            };
500
501            let data = Bytes::from_static(body.as_bytes());
502            let size = data.len() as u64;
503            Ok(Response::builder()
504                .status(status)
505                .body(HttpBody::new(
506                    stream::iter([Ok(Buffer::from(data))]),
507                    Some(size),
508                ))
509                .unwrap())
510        }
511    }
