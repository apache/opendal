35impl WebdavLister {
36    pub fn new(core: Arc<WebdavCore>, ctx: OperationContext, path: &str, args: OpList) -> Self {
37        Self {
38            core,
39            ctx,
40            path: path.to_string(),
41            args,
42        }
43    }
44
45    fn resolve_href(&self, href: &str, is_dir: bool) -> Result<String> {
46        let raw_path = extract_href_path(href);
47        if !raw_path.starts_with('/') {
48            return Err(Error::new(
49                ErrorKind::Unexpected,
50                "webdav propfind href is not an absolute path or URI",
51            )
52            .with_context("href", href));
53        }
54
55        let stripped = if self.core.server_path.is_empty() {
56            raw_path
57        } else if raw_path == self.core.server_path {
58            ""
59        } else if let Some(rest) = raw_path.strip_prefix(&self.core.server_path)
60            && rest.starts_with('/')
61        {
62            rest
63        } else {
64            return Err(Error::new(
65                ErrorKind::Unexpected,
66                "webdav propfind href does not start with server path",
67            )
68            .with_context("href", href)
69            .with_context("server_path", &self.core.server_path));
70        };
71
72        let mut path = percent_decode_path(stripped);
73
74        // Append `/` to path if it's a dir
75        if !path.ends_with('/') && is_dir {
76            path.push('/');
77        }
78
79        if path == self.core.root {
80            return Ok("/".to_owned());
81        }
82
83        match path.strip_prefix(&self.core.root) {
84            Some(rel) => Ok(rel.to_owned()),
85            None => Err(Error::new(
86                ErrorKind::Unexpected,
87                "webdav propfind href is outside configured root",
88            )
89            .with_context("href", href)
90            .with_context("root", &self.core.root)),
91        }
92    }
93}

95/// Extract the URI path component from a WebDAV `<D:href>` value.
96///
97/// Per RFC 4918 §8.3, `href` may be either a `path-absolute` (e.g. `/dav/file`)
98/// or an `absolute-URI` (e.g. `https://host/dav/file`).
99fn extract_href_path(href: &str) -> &str {
100    if let Some((scheme, rest)) = href.split_once("://")
101        && scheme
102            .as_bytes()
103            .first()
104            .is_some_and(u8::is_ascii_alphabetic)
105        && scheme
106            .bytes()
107            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.'))
108    {
109        let authority_and_path = rest.split(['?', '#']).next().unwrap_or("");
110        return match authority_and_path.find('/') {
111            Some(idx) => &authority_and_path[idx..],
112            None => "/",
113        };
114    }
115
116    href
117}

119impl oio::PageList for WebdavLister {
120    async fn next_page(&self, ctx: &mut oio::PageContext) -> Result<()> {
121        // PROPFIND enumerates a collection. Resolve a partial name in its parent.
122        let directory = if self.path.ends_with('/') {
123            self.path.as_str()
124        } else {
125            get_parent(&self.path)
126        };
127        let resp = self
128            .core
129            .webdav_list(&self.ctx, directory, &self.args)
130            .await?;
131
132        // jfrog artifactory's webdav services have some strange behavior.
133        // We add this flag to check if the server is jfrog artifactory.
134        //
135        // Example: `"x-jfrog-version": "Artifactory/7.77.5 77705900"`
136        let is_jfrog_artifactory = if let Some(v) = resp.headers().get("x-jfrog-version") {
137            v.to_str().unwrap_or_default().starts_with("Artifactory")
138        } else {
139            false
140        };
141
142        let bs = if resp.status().is_success() {
143            resp.into_body()
144        } else if resp.status() == StatusCode::NOT_FOUND {
145            ctx.done = true;
146            return Ok(());
147        } else {
148            return Err(parse_error(
149                ErrorContext::new(ServiceOperation("Propfind")),
150                resp,
151            ));
152        };
153
154        let result: Multistatus = deserialize_multistatus(&bs.to_bytes())?;
155
156        for res in result.response {
157            let meta = parse_propstats(&res.propstat)?;
158            let normalized_path = self.resolve_href(&res.href, meta.is_dir())?;
159
160            if directory != self.path && !normalized_path.starts_with(&self.path) {
161                continue;
162            }
163
164            // HACKS! HACKS! HACKS!
165            //
166            // jfrog artifactory will generate a virtual checksum file for each file.
167            // The checksum file can't be stated, but can be listed and read.
168            // We ignore the checksum files to avoid listing unexpected files.
169            if is_jfrog_artifactory && meta.content_type() == Some("application/x-checksum") {
170                continue;
171            }
172
173            ctx.entries
174                .push_back(oio::Entry::new(&normalized_path, meta))
175        }
176        ctx.done = true;
177
178        Ok(())
179    }
180}

191    struct StaticPropfindTransport {
192        body: String,
193    }

195    impl HttpTransport for StaticPropfindTransport {
196        async fn fetch(&self, _req: Request<Buffer>) -> Result<Response<HttpBody>> {
197            let data = Bytes::from(self.body.clone());
198            let size = data.len() as u64;
199            Ok(Response::builder()
200                .status(StatusCode::MULTI_STATUS)
201                .body(HttpBody::new(
202                    stream::iter([Ok(Buffer::from(data))]),
203                    Some(size),
204                ))
205                .unwrap())
206        }
207    }

209    fn operator_with_propfind(endpoint: &str, root: &str, xml: &str) -> Result<Operator> {
210        Ok(
211            Operator::new(WebdavBuilder::default().endpoint(endpoint).root(root))?.with_context(
212                OperationContext::new().with_http_transport(HttpTransporter::new(
213                    StaticPropfindTransport {
214                        body: xml.to_string(),
215                    },
216                )),
217            ),
218        )
219    }

314    #[tokio::test]
315    async fn test_list_rejects_short_empty_or_out_of_root_hrefs_without_panicking() -> Result<()> {
316        let cases = [
317            // Shorter than root
318            ("http://localhost:8080", "/sub/", "<D:href>/</D:href>", true),
319            // Empty href under non-root
320            ("http://localhost:8080", "/sub/", "<D:href/>", false),
321            // Empty href under root even when marked as collection
322            ("http://localhost:8080", "/", "<D:href/>", true),
323            // Sibling prefix sharing string prefix without path boundary
324            (
325                "http://localhost:8080",
326                "/sub/",
327                "<D:href>/sub2/file.txt</D:href>",
328                false,
329            ),
330            // Absolute URI outside root
331            (
332                "http://localhost:8080",
333                "/sub/",
334                "<D:href>http://localhost:8080/other/file.txt</D:href>",
335                false,
336            ),
337            // Outside server_path (shares prefix without slash boundary)
338            (
339                "http://localhost:8080/dav",
340                "/",
341                "<D:href>/dav2/file.txt</D:href>",
342                false,
343            ),
344            // Non-absolute relative href
345            (
346                "http://localhost:8080",
347                "/sub/",
348                "<D:href>sub/file.txt</D:href>",
349                false,
350            ),
351        ];
352
353        for (endpoint, root, href_xml, is_dir) in cases {
354            let prop_xml = if is_dir {
355                "<D:resourcetype><D:collection/></D:resourcetype>"
356            } else {
357                "<D:getcontentlength>1</D:getcontentlength><D:resourcetype/>"
358            };
359            let xml = format!(
360                r#"<?xml version="1.0" encoding="utf-8"?>
361                <D:multistatus xmlns:D="DAV:">
362                  <D:response>
363                    {href_xml}
364                    <D:propstat>
365                      <D:prop>
366                        <D:getlastmodified>Sat, 07 May 2022 06:39:47 GMT</D:getlastmodified>
367                        {prop_xml}
368                      </D:prop>
369                      <D:status>HTTP/1.1 200 OK</D:status>
370                    </D:propstat>
371                  </D:response>
372                </D:multistatus>"#
373            );
374
375            let op = operator_with_propfind(endpoint, root, &xml)?;
376            let err = op.list("").await.expect_err(&format!(
377                "href {href_xml} under {endpoint} / {root} must fail"
378            ));
379            assert_eq!(err.kind(), ErrorKind::Unexpected, "{href_xml}");
380            assert!(
381                err.to_string().contains("href"),
382                "expected href error for {href_xml}, got: {err}"
383            );
384        }
385
386        Ok(())
387    }
